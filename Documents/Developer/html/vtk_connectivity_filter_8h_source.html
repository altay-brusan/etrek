307inline const char* vtkConnectivityFilter::GetExtractionModeAsString()
308{
309 if (this->ExtractionMode == VTK_EXTRACT_POINT_SEEDED_REGIONS)
310 {
311 return "ExtractPointSeededRegions";
312 }
313 else if (this->ExtractionMode == VTK_EXTRACT_CELL_SEEDED_REGIONS)
314 {
315 return "ExtractCellSeededRegions";
316 }
317 else if (this->ExtractionMode == VTK_EXTRACT_SPECIFIED_REGIONS)
318 {
319 return "ExtractSpecifiedRegions";
320 }
321 else if (this->ExtractionMode == VTK_EXTRACT_ALL_REGIONS)
322 {
323 return "ExtractAllRegions";
324 }
325 else if (this->ExtractionMode == VTK_EXTRACT_CLOSEST_POINT_REGION)
326 {
327 return "ExtractClosestPointRegion";
328 }
329 else
330 {
331 return "ExtractLargestRegion";
332 }
333}
