908 template <typename T>
909 static OFBool
910 safeSubtract(T minuend, T subtrahend, T& difference)
911 {
912 assert(!OFnumeric_limits<T>::is_signed);
913 if (minuend < subtrahend) {
914 return OFFalse;
915 } else {
916 difference = minuend - subtrahend;
917 return OFTrue;
918 }
919 }

930 template <typename T>
931 static OFBool
932 safeAdd(T a, T b, T& sum)
933 {
934 assert(!OFnumeric_limits<T>::is_signed);
935 if ((OFnumeric_limits<T>::max)() - a < b) {
936 return OFFalse;
937 } else {
938 sum = a + b;
939 return OFTrue;
940 }
941 }

952 template <typename T>
953 static OFBool safeMult(T a, T b, T& product)
954 {
955 assert(!OFnumeric_limits<T>::is_signed);
956 T x = a * b;
957 if (a != 0 && x / a != b) {
958 return OFFalse;
959 }
960 product = x;
961 return OFTrue;
962 }
