20/* Flag definitions for H5LTopen_file_image() */
21#define H5LT_FILE_IMAGE_OPEN_RW 0x0001 /* Open image for read-write */
22#define H5LT_FILE_IMAGE_DONT_COPY 0x0002 /* The HDF5 lib won't copy */
23/* user supplied image buffer. The same image is open with the core driver. */
24#define H5LT_FILE_IMAGE_DONT_RELEASE 0x0004 /* The HDF5 lib won't */
25/* deallocate user supplied image buffer. The user application is responsible */
26/* for doing so. */
