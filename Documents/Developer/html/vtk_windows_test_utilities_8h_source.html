16inline LONG WINAPI vtkWindowsTestUlititiesExceptionHandler(EXCEPTION_POINTERS* ExceptionInfo)
17{
18 switch (ExceptionInfo->ExceptionRecord->ExceptionCode)
19 {
20 case EXCEPTION_ACCESS_VIOLATION:
21 vtkLog(ERROR, << "Error: EXCEPTION_ACCESS_VIOLATION\n");
22 break;
23 case EXCEPTION_ARRAY_BOUNDS_EXCEEDED:
24 vtkLog(ERROR, << "Error: EXCEPTION_ARRAY_BOUNDS_EXCEEDED\n");
25 break;
26 case EXCEPTION_BREAKPOINT:
27 vtkLog(ERROR, << "Error: EXCEPTION_BREAKPOINT\n");
28 break;
29 case EXCEPTION_DATATYPE_MISALIGNMENT:
30 vtkLog(ERROR, << "Error: EXCEPTION_DATATYPE_MISALIGNMENT\n");
31 break;
32 case EXCEPTION_FLT_DENORMAL_OPERAND:
33 vtkLog(ERROR, << "Error: EXCEPTION_FLT_DENORMAL_OPERAND\n");
34 break;
35 case EXCEPTION_FLT_DIVIDE_BY_ZERO:
36 vtkLog(ERROR, << "Error: EXCEPTION_FLT_DIVIDE_BY_ZERO\n");
37 break;
38 case EXCEPTION_FLT_INEXACT_RESULT:
39 vtkLog(ERROR, << "Error: EXCEPTION_FLT_INEXACT_RESULT\n");
40 break;
41 case EXCEPTION_FLT_INVALID_OPERATION:
42 vtkLog(ERROR, << "Error: EXCEPTION_FLT_INVALID_OPERATION\n");
43 break;
44 case EXCEPTION_FLT_OVERFLOW:
45 vtkLog(ERROR, << "Error: EXCEPTION_FLT_OVERFLOW\n");
46 break;
47 case EXCEPTION_FLT_STACK_CHECK:
48 vtkLog(ERROR, << "Error: EXCEPTION_FLT_STACK_CHECK\n");
49 break;
50 case EXCEPTION_FLT_UNDERFLOW:
51 vtkLog(ERROR, << "Error: EXCEPTION_FLT_UNDERFLOW\n");
52 break;
53 case EXCEPTION_ILLEGAL_INSTRUCTION:
54 vtkLog(ERROR, << "Error: EXCEPTION_ILLEGAL_INSTRUCTION\n");
55 break;
56 case EXCEPTION_IN_PAGE_ERROR:
57 vtkLog(ERROR, << "Error: EXCEPTION_IN_PAGE_ERROR\n");
58 break;
59 case EXCEPTION_INT_DIVIDE_BY_ZERO:
60 vtkLog(ERROR, << "Error: EXCEPTION_INT_DIVIDE_BY_ZERO\n");
61 break;
62 case EXCEPTION_INT_OVERFLOW:
63 vtkLog(ERROR, << "Error: EXCEPTION_INT_OVERFLOW\n");
64 break;
65 case EXCEPTION_INVALID_DISPOSITION:
66 vtkLog(ERROR, << "Error: EXCEPTION_INVALID_DISPOSITION\n");
67 break;
68 case EXCEPTION_NONCONTINUABLE_EXCEPTION:
69 vtkLog(ERROR, << "Error: EXCEPTION_NONCONTINUABLE_EXCEPTION\n");
70 break;
71 case EXCEPTION_PRIV_INSTRUCTION:
72 vtkLog(ERROR, << "Error: EXCEPTION_PRIV_INSTRUCTION\n");
73 break;
74 case EXCEPTION_SINGLE_STEP:
75 vtkLog(ERROR, << "Error: EXCEPTION_SINGLE_STEP\n");
76 break;
77 case EXCEPTION_STACK_OVERFLOW:
78 vtkLog(ERROR, << "Error: EXCEPTION_STACK_OVERFLOW\n");
79 break;
80 default:
81 vtkLog(ERROR, << "Error: Unrecognized Exception\n");
82 break;
83 }
84
85 std::string stack = vtksys::SystemInformation::GetProgramStack(0, 0);
86
87 vtkLog(ERROR, << stack);
88
89 return EXCEPTION_CONTINUE_SEARCH;
90}
91
92inline void vtkWindowsTestUtilitiesSetupForTesting()
93{
94 SetUnhandledExceptionFilter(vtkWindowsTestUlititiesExceptionHandler);
95}
