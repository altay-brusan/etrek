181 unsigned int FindDichotomic(double value, unsigned char dim, double tol) const
182 {
183 unsigned int maxIdx = this->GetDimensions()[dim] - 1;
184 if (value < (this->Origin[dim] - tol) ||
185 value > (this->Origin[dim] + tol + this->GridScale[dim] * maxIdx))
186 {
187 return std::numeric_limits<unsigned int>::max();
188 }
189
190 long idx = std::round((value - this->Origin[dim]) / this->GridScale[dim]);
191 return std::min(std::max(idx, 0l), static_cast<long>(maxIdx));
192 }
