112class vtkPermuteOptions

114 using Permutation = std::vector<size_t>;

116 struct Value
117 {
118 Value(const std::string& name, std::function<void(ObjType*)> setter)
119 : Name(name)
120 , Setter(setter)
121 {
122 }
123
124 void Apply(ObjType* obj) const { this->Setter(obj); }
125
126 std::string Name; // user-readable option name
127 std::function<void(ObjType*)> Setter; // Sets the option to a single values
128 };

130 struct Option
131 {
132 Option(const std::string& name)
133 : Name(name)
134 {
135 }
136 std::string Name; // user-readable option name
137 std::vector<Value> Values; // list of values to test for this option
138 };
139
140 std::vector<Option> Options;
141 std::vector<Permutation> Permutations;
142 size_t CurrentPermutation;
143 vtkTimeStamp OptionTime;
144 vtkTimeStamp PermutationTime;
145
146 Option& FindOrCreateOption(const std::string& name)
147 {
148 for (Option& opt : this->Options)
149 {
150 if (opt.Name == name)
151 {
152 return opt;
153 }
154 }
155
156 this->Options.emplace_back(name);
157 return this->Options.back();
158 }
159
160 void RecursePermutations(Permutation& perm, size_t level)
161 {
162 const size_t maxIdx = this->Options[level].Values.size();
163
164 if (level == 0) // base case
165 {
166 for (size_t i = 0; i < maxIdx; ++i)
167 {
168 perm[0] = i;
169 this->Permutations.push_back(perm);
170 }
171 }
172 else // recursive case
173 {
174 for (size_t i = 0; i < maxIdx; ++i)
175 {
176 perm[level] = i;
177 this->RecursePermutations(perm, level - 1);
178 }
179 }
180 }
181
182 void RebuildPermutations()
183 {
184 this->Permutations.clear();
185
186 const size_t numOptions = this->Options.size();
187 Permutation perm(numOptions, 0);
188 this->RecursePermutations(perm, numOptions - 1);
189 this->PermutationTime.Modified();
190 }
191
192 void Apply(ObjType* obj, const Permutation& perm) const
193 {
194 const size_t numOpts = this->Options.size();
195 assert("Sane permutation" && perm.size() == numOpts);
196
197 for (size_t i = 0; i < numOpts; ++i)
198 {
199 size_t valIdx = perm[i];
200 assert("ValueIdx in range" && valIdx < this->Options[i].Values.size());
201 this->Options[i].Values[valIdx].Apply(obj);
202 }
203 }
204
205 std::string NamePermutation(const Permutation& perm) const
206 {
207 const size_t numOpts = this->Options.size();
208 assert("Sane permutation" && perm.size() == numOpts);
209
210 std::ostringstream out;
211 for (size_t i = 0; i < numOpts; ++i)
212 {
213 size_t valIdx = perm[i];
214 assert("ValueIdx in range" && valIdx < this->Options[i].Values.size());
215
216 out << (i != 0 ? "-" : "") << this->Options[i].Name << "."
217 << this->Options[i].Values[valIdx].Name;
218 }
219
220 return out.str();
221 }

223public:
224 vtkPermuteOptions()

229 template <typename SetterType, typename ValueType>
230 void AddOptionValue(
231 const std::string& optionName, SetterType setter, const std::string& valueName, ValueType value)
232 {
233 using std::placeholders::_1;
234
235 std::function<void(ObjType*)> func = [setter, value](ObjType* obj) { (obj->*setter)(value); };
236 Option& opt = this->FindOrCreateOption(optionName);
237 opt.Values.emplace_back(valueName, func);
238 this->OptionTime.Modified();
239 }

242 void AddOptionValues(

244 {
245 this->AddOptionValue(optionName, setter, valueName, value);
246 }

249 void AddOptionValues(const std::string& optionName, SetterType setter,
250 const std::string& valueName, ValueType value, Tail... tail)
251 {
252 this->AddOptionValue(optionName, setter, valueName, value);
253 this->AddOptionValues(optionName, setter, tail...);
254 }
255
256 void InitPermutations()
257 {
258 if (this->OptionTime > this->PermutationTime)
259 {
260 this->RebuildPermutations();
261 }
262
263 this->CurrentPermutation = 0;
264 }
265
266 bool IsDoneWithPermutations() const
267 {
268 assert("Modified options without resetting permutations." &&
269 this->PermutationTime > this->OptionTime);
270
271 return this->CurrentPermutation >= this->Permutations.size();
272 }
273
274 void GoToNextPermutation()
275 {
276 assert("Modified options without resetting permutations." &&
277 this->PermutationTime > this->OptionTime);
278 assert("Invalid permutation." && !this->IsDoneWithPermutations());
279
280 ++this->CurrentPermutation;
281 }
282
283 void ApplyCurrentPermutation(ObjType* obj) const
284 {
285 assert("Modified options without resetting permutations." &&
286 this->PermutationTime > this->OptionTime);
287 assert("Invalid permutation." && !this->IsDoneWithPermutations());
288
289 this->Apply(obj, this->Permutations[this->CurrentPermutation]);
290 }
291
292 std::string GetCurrentPermutationName() const
293 {
294 assert("Modified options without resetting permutations." &&
295 this->PermutationTime > this->OptionTime);
296 assert("Invalid permutation." && !this->IsDoneWithPermutations());
297 return this->NamePermutation(this->Permutations[this->CurrentPermutation]);
298 }
