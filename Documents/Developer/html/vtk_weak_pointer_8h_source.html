46 // These static asserts only fire when the function calling CheckTypes is
47 // used. Thus, this smart pointer class may still be used as a member variable
48 // with a forward declared T, so long as T is defined by the time the calling
49 // function is used.
50 template <typename U = T>
51 static void CheckTypes() noexcept
52 {
53 static_assert(vtk::detail::IsComplete<T>::value,
54 "vtkWeakPointer<T>'s T type has not been defined. Missing "
55 "include?");
56 static_assert(vtk::detail::IsComplete<U>::value,
57 "Cannot store an object with undefined type in "
58 "vtkWeakPointer. Missing include?");
59 static_assert(std::is_base_of<T, U>::value,
60 "Argument type is not compatible with vtkWeakPointer<T>'s "
61 "T type.");
62 static_assert(std::is_base_of<vtkObjectBase, T>::value,
63 "vtkWeakPointer can only be used with subclasses of "
64 "vtkObjectBase.");
65 }

208 // Work-around for HP and IBM overload resolution bug. Since
209 // NullPointerOnly is a private type the only pointer value that can
210 // be passed by user code is a null pointer. This operator will be
211 // chosen by the compiler when comparing against null explicitly and
212 // avoid the bogus ambiguous overload error.
213#if defined(__HP_aCC) || defined(__IBMCPP__)
214#define VTK_WEAK_POINTER_DEFINE_OPERATOR_WORKAROUND(op) \
215 bool operator op(NullPointerOnly*) const { return ::operator op(*this, 0); }
