363 template <class Container>
364 static OFCondition copyFromUint8Array(DcmElement* elem, Container& destination)
365 {
366 if (elem == NULL)
367 return EC_IllegalParameter;
368
369 size_t numValues = elem->getNumberOfValues();
370 OFCondition result;
371 for (size_t count = 0; (count < numValues) && result.good(); count++)
372 {
373 Uint8 value;
374 result = elem->getUint8(value, OFstatic_cast(unsigned long, count));
375 if (result.good())
376 {
377 destination.push_back(value);
378 }
379 }
380 if (result.bad())
381 {
382 destination.clear();
383 }
384 return result;
385 }

420 template <class Container>
421 static OFCondition readSubSequence(DcmSequenceOfItems* source,
422 const DcmTagKey& seqKey,
423 Container& destination,
424 const OFString& cardinality,
425 const OFString& type,
426 const OFString& module)
427 {
428 OFCondition result;
429 OFCondition exists = EC_Normal;
430 if (!source)
431 exists = EC_TagNotFound;
432
433 /* Check sequence and report errors as warnings, read anyway */
434 checkElementValue(source, seqKey, cardinality, type, exists, module.c_str());
435 if (source)
436 {
437 DcmItem* item = OFstatic_cast(DcmItem*, source->nextInContainer(NULL));
438 size_t count = 0;
439 while (item != NULL)
440 {
441 if (item != NULL)
442 {
443 // define the element type
444 typedef typename OFremove_pointer<typename Container::value_type>::type Element;
445 Element* newElem = new Element();
446 // read into container item (clears old data first)
447 if (newElem != NULL)
448 {
449 result = (newElem)->read(*item, OFFalse /* no need to delete anything */);
450 if (result.good())
451 {
452 destination.push_back(newElem);
453 }
454 else
455 {
456 delete newElem;
457 DCMIOD_WARN("Could not read item #" << count << " from "
458 << DcmTag(source->getTag()).getTagName()
459 << " (skipping item): " << result.text());
460 }
461 }
462 else
463 {
464 DCMIOD_ERROR("Could not store data from item #" << count << " from "
465 << DcmTag(source->getTag()).getTagName()
466 << " (skipping item): Memory exhausted?");
467 }
468 }
469 else
470 {
471 DCMIOD_WARN("Could not get item #" << count << " from " << DcmTag(source->getTag()).getTagName()
472 << " (malformed data or internal error), skipping item");
473 }
474 item = OFstatic_cast(DcmItem*, source->nextInContainer(item));
475 count++;
476 }
477 }
478 else
479 {
480 result = EC_IllegalParameter;
481 }
482 return result;
483 }

612 template <class Container>
613 static void writeSubSequence(OFCondition& result,
614 const DcmTagKey& seqKey,
615 Container& source,
616 DcmItem& destination,
617 const OFString& cardinality,
618 const OFString& type,
619 const OFString& module)
620 {
621 if (result.good())
622 {
623 // Delete old data
624 destination.findAndDeleteElement(seqKey);
625
626 // If we have meaningful data, write it
627 OFBool haveData = source.size() > 0;
628 if (haveData)
629 {
630 destination.insertEmptyElement(seqKey);
631 Uint32 count = 0;
632 typename Container::iterator it = source.begin();
633 while (it != source.end() && result.good())
634 {
635 if ((*it) != NULL)
636 {
637 DcmItem* localItem = NULL;
638 // If item is NULL result.bad() is always true.
639 result = destination.findOrCreateSequenceItem(seqKey, localItem, -2 /* append new */);
640 if (result.good())
641 {
642 result = (*it)->write(*localItem);
643 if (result.bad())
644 {
645 destination.findAndDeleteSequenceItem(seqKey, -1 /* last */);
646 DCMIOD_ERROR("Could not write item #" << count << " in " << DcmTag(seqKey).getTagName()
647 << ": " << result.text());
648 }
649 }
650 else
651 {
652 DCMIOD_ERROR("Could not insert item data #" << count << " of "
653 << DcmTag(seqKey).getTagName()
654 << " (internal error), ignoring");
655 }
656 }
657 else
658 {
659 DCMIOD_ERROR("Found uninitialized container item (NULL value) for sequence "
660 << DcmTag(seqKey).getTagName() << " (internal error, skipping)");
661 }
662 count++;
663 it++;
664 }
665 }
666 // If we do not have data, create empty sequence if type 2
667 else if (type == "2")
668 {
669 destination.insertEmptyElement(seqKey);
670 }
671
672 // Check result
673 checkSubSequence(result, destination, seqKey, cardinality, type, module, dcmtk::log4cplus::ERROR_LOG_LEVEL);
674
675 // Clean up if we did not have success */
676 if (result.bad())
677 {
678 destination.findAndDeleteElement(seqKey);
679 }
680 }
681 }

729 template <class Container>
730 static void writeSingleItem(OFCondition& result,
731 const DcmTagKey& seqKey,
732 Container& source,
733 DcmItem& destination,
734 const OFString& type,
735 const OFString& module)
736 {
737 if (result.good())
738 {
739 // Delete old data
740 destination.findAndDeleteElement(seqKey);
741
742 /* If we have data, write it */
743 OFCondition haveData = source.check(OFTrue /* Be quiet */);
744 if (haveData.good())
745 {
746 DcmItem* localItem = NULL;
747 // If item is NULL result.bad() is always true.
748 result = destination.findOrCreateSequenceItem(seqKey, localItem, 0);
749 if (result.good())
750 {
751 result = source.write(*localItem);
752 // It can happen that check() returns OK but no elements have to be
753 // written at all (e.g.\ if it contains only type 3 attributes). In that
754 // case a sequence with a single empty item is written which must be removed
755 // afterwards.
756 if (result.good() && (localItem->card() == 0))
757 {
758 destination.findAndDeleteElement(seqKey);
759 }
760 }
761 }
762 /* If we do not have data, insert empty for type 2 */
763 else if (type == "2")
764 {
765 destination.insertEmptyElement(seqKey);
766 }
767 else if (type == "1C")
768 {
769 DCMIOD_TRACE("Skipping type 1C sequence " << seqKey << ": No data or incomplete data available");
770 }
771 else if (type == "3")
772 {
773 DCMIOD_TRACE("Skipping type 3 sequence " << seqKey << ": No data or incomplete data available");
774 }
775 /* Check outcome */
776 checkSubSequence(result, destination, seqKey, "1", type, module, dcmtk::log4cplus::ERROR_LOG_LEVEL);
777 }
778 }

861 template <class Container>
862 static OFCondition copyContainer(const Container& src, Container& dst)
863 {
864 typedef typename OFremove_pointer<typename Container::value_type>::type Element;
865 typename Container::const_iterator it = src.begin();
866 while (it != src.end())
867 {
868 if (*it != NULL)
869 {
870 Element* elem = new Element(**it);
871 if (elem == NULL)
872 {
873 freeContainer(dst);
874 return EC_MemoryExhausted;
875 }
876 dst.push_back(elem);
877 }
878 else
879 {
880 DCMIOD_ERROR("Could not copy element while copying container: Element in source is NULL, skipping");
881 }
882 it++;
883 }
884 return EC_Normal;
885 }

895 template <class Container>
896 static int compareContainer(const Container& src, const Container& dst)
897 {
898 size_t card = src.size();
899 if (card != dst.size())
900 return -1;
901 typename Container::const_iterator it = src.begin();
902 typename Container::const_iterator it2 = dst.begin();
903 while (it != src.end())
904 {
905 if ((*it)->compare(**it2) != 0)
906 return 1;
907 it++;
908 it2++;
909 }
910 return 0;
911 }
