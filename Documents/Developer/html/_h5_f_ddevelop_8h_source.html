28/* Map "fractal heap" header blocks to 'ohdr' type file memory, since its
29 * a fair amount of work to add a new kind of file memory and they are similar
30 * enough to object headers and probably too minor to deserve their own type.
31 *
32 * Map "fractal heap" indirect blocks to 'ohdr' type file memory, since they
33 * are similar to fractal heap header blocks.
34 *
35 * Map "fractal heap" direct blocks to 'lheap' type file memory, since they
36 * will be replacing local heaps.
37 *
38 * Map "fractal heap" 'huge' objects to 'draw' type file memory, since they
39 * represent large objects that are directly stored in the file.
40 *
41 * -QAK
42 */
43#define H5FD_MEM_FHEAP_HDR H5FD_MEM_OHDR
44#define H5FD_MEM_FHEAP_IBLOCK H5FD_MEM_OHDR
45#define H5FD_MEM_FHEAP_DBLOCK H5FD_MEM_LHEAP
46#define H5FD_MEM_FHEAP_HUGE_OBJ H5FD_MEM_DRAW
47
48/* Map "free space" header blocks to 'ohdr' type file memory, since its
49 * a fair amount of work to add a new kind of file memory and they are similar
50 * enough to object headers and probably too minor to deserve their own type.
51 *
52 * Map "free space" serialized sections to 'lheap' type file memory, since they
53 * are similar enough to local heap info.
54 *
55 * -QAK
56 */
57#define H5FD_MEM_FSPACE_HDR H5FD_MEM_OHDR
58#define H5FD_MEM_FSPACE_SINFO H5FD_MEM_LHEAP
59
60/* Map "shared object header message" master table to 'ohdr' type file memory,
61 * since its a fair amount of work to add a new kind of file memory and they are
62 * similar enough to object headers and probably too minor to deserve their own
63 * type.
64 *
65 * Map "shared object header message" indices to 'btree' type file memory,
66 * since they are similar enough to B-tree nodes.
67 *
68 * -QAK
69 */
70#define H5FD_MEM_SOHM_TABLE H5FD_MEM_OHDR
71#define H5FD_MEM_SOHM_INDEX H5FD_MEM_BTREE

73/* Map "extensible array" header blocks to 'ohdr' type file memory, since its
74 * a fair amount of work to add a new kind of file memory and they are similar
75 * enough to object headers and probably too minor to deserve their own type.
76 *
77 * Map "extensible array" index blocks to 'ohdr' type file memory, since they
78 * are similar to extensible array header blocks.
79 *
80 * Map "extensible array" super blocks to 'btree' type file memory, since they
81 * are similar enough to B-tree nodes.
82 *
83 * Map "extensible array" data blocks & pages to 'lheap' type file memory, since
84 * they are similar enough to local heap info.
85 *
86 * -QAK
87 */
88#define H5FD_MEM_EARRAY_HDR H5FD_MEM_OHDR
89#define H5FD_MEM_EARRAY_IBLOCK H5FD_MEM_OHDR
90#define H5FD_MEM_EARRAY_SBLOCK H5FD_MEM_BTREE
91#define H5FD_MEM_EARRAY_DBLOCK H5FD_MEM_LHEAP
92#define H5FD_MEM_EARRAY_DBLK_PAGE H5FD_MEM_LHEAP
93
94/* Map "fixed array" header blocks to 'ohdr' type file memory, since its
95 * a fair amount of work to add a new kind of file memory and they are similar
96 * enough to object headers and probably too minor to deserve their own type.
97 *
98 * Map "fixed array" data blocks & pages to 'lheap' type file memory, since
99 * they are similar enough to local heap info.
100 *
101 */
102#define H5FD_MEM_FARRAY_HDR H5FD_MEM_OHDR
103#define H5FD_MEM_FARRAY_DBLOCK H5FD_MEM_LHEAP
104#define H5FD_MEM_FARRAY_DBLK_PAGE H5FD_MEM_LHEAP

106/*
107 * A free-list map which maps all types of allocation requests to a single
108 * free list. This is useful for drivers that don't really care about
109 * keeping different requests segregated in the underlying file and which
110 * want to make most efficient reuse of freed memory. The use of the
111 * H5FD_MEM_SUPER free list is arbitrary.
112 */
113#define H5FD_FLMAP_SINGLE \
114 { \
115 H5FD_MEM_SUPER, /*default*/ \
116 H5FD_MEM_SUPER, /*super*/ \
117 H5FD_MEM_SUPER, /*btree*/ \
118 H5FD_MEM_SUPER, /*draw*/ \
119 H5FD_MEM_SUPER, /*gheap*/ \
120 H5FD_MEM_SUPER, /*lheap*/ \
121 H5FD_MEM_SUPER /*ohdr*/ \
122 }
123
124/*
125 * A free-list map which segregates requests into `raw' or `meta' data
126 * pools.
127 */
128#define H5FD_FLMAP_DICHOTOMY \
129 { \
130 H5FD_MEM_SUPER, /*default*/ \
131 H5FD_MEM_SUPER, /*super*/ \
132 H5FD_MEM_SUPER, /*btree*/ \
133 H5FD_MEM_DRAW, /*draw*/ \
134 H5FD_MEM_DRAW, /*gheap*/ \
135 H5FD_MEM_SUPER, /*lheap*/ \
136 H5FD_MEM_SUPER /*ohdr*/ \
137 }
138
139/*
140 * The default free list map which causes each request type to use it's own
141 * free-list.
142 */
143#define H5FD_FLMAP_DEFAULT \
144 { \
145 H5FD_MEM_DEFAULT, /*default*/ \
146 H5FD_MEM_DEFAULT, /*super*/ \
147 H5FD_MEM_DEFAULT, /*btree*/ \
148 H5FD_MEM_DEFAULT, /*draw*/ \
149 H5FD_MEM_DEFAULT, /*gheap*/ \
150 H5FD_MEM_DEFAULT, /*lheap*/ \
151 H5FD_MEM_DEFAULT /*ohdr*/ \
152 }

207/*
208 * The main datatype for each driver. Public fields common to all drivers
209 * are declared here and the driver appends private fields in memory.
210 */
211struct H5FD_t {
212 hid_t driver_id; /*driver ID for this file */
213 const H5FD_class_t *cls; /*constant class info */
214 unsigned long fileno; /* File 'serial' number */
215 unsigned access_flags; /* File access flags (from create or open) */
216 unsigned long feature_flags; /* VFL Driver feature Flags */
217 haddr_t maxaddr; /* For this file, overrides class */
218 haddr_t base_addr; /* Base address for HDF5 data w/in file */
219
220 /* Space allocation management fields */
221 hsize_t threshold; /* Threshold for alignment */
222 hsize_t alignment; /* Allocation alignment */
223 hbool_t paged_aggr; /* Paged aggregation for file space is enabled or not */
224};
