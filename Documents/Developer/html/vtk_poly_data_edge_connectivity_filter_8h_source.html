402inline const char* vtkPolyDataEdgeConnectivityFilter::GetExtractionModeAsString()
403{
404 if (this->ExtractionMode == VTK_EXTRACT_POINT_SEEDED_REGIONS)
405 {
406 return "ExtractPointSeededRegions";
407 }
408 else if (this->ExtractionMode == VTK_EXTRACT_CELL_SEEDED_REGIONS)
409 {
410 return "ExtractCellSeededRegions";
411 }
412 else if (this->ExtractionMode == VTK_EXTRACT_SPECIFIED_REGIONS)
413 {
414 return "ExtractSpecifiedRegions";
415 }
416 else if (this->ExtractionMode == VTK_EXTRACT_ALL_REGIONS)
417 {
418 return "ExtractAllRegions";
419 }
420 else if (this->ExtractionMode == VTK_EXTRACT_CLOSEST_POINT_REGION)
421 {
422 return "ExtractClosestPointRegion";
423 }
424 else if (this->ExtractionMode == VTK_EXTRACT_LARGE_REGIONS)
425 {
426 return "ExtractLargeRegions";
427 }
428 else
429 {
430 return "ExtractLargestRegion";
431 }
432}
