26/* Flags for H5Pset_fapl_log() */
27/* Flags for tracking 'meta' operations (truncate) */
28#define H5FD_LOG_TRUNCATE 0x00000001
29#define H5FD_LOG_META_IO (H5FD_LOG_TRUNCATE)
30/* Flags for tracking where reads/writes/seeks occur */
31#define H5FD_LOG_LOC_READ 0x00000002
32#define H5FD_LOG_LOC_WRITE 0x00000004
33#define H5FD_LOG_LOC_SEEK 0x00000008
34#define H5FD_LOG_LOC_IO (H5FD_LOG_LOC_READ | H5FD_LOG_LOC_WRITE | H5FD_LOG_LOC_SEEK)
35/* Flags for tracking number of times each byte is read/written */
36#define H5FD_LOG_FILE_READ 0x00000010
37#define H5FD_LOG_FILE_WRITE 0x00000020
38#define H5FD_LOG_FILE_IO (H5FD_LOG_FILE_READ | H5FD_LOG_FILE_WRITE)
39/* Flag for tracking "flavor" (type) of information stored at each byte */
40#define H5FD_LOG_FLAVOR 0x00000040
41/* Flags for tracking total number of reads/writes/seeks/truncates */
42#define H5FD_LOG_NUM_READ 0x00000080
43#define H5FD_LOG_NUM_WRITE 0x00000100
44#define H5FD_LOG_NUM_SEEK 0x00000200
45#define H5FD_LOG_NUM_TRUNCATE 0x00000400
46#define H5FD_LOG_NUM_IO (H5FD_LOG_NUM_READ | H5FD_LOG_NUM_WRITE | H5FD_LOG_NUM_SEEK | H5FD_LOG_NUM_TRUNCATE)
47/* Flags for tracking time spent in open/stat/read/write/seek/truncate/close */
48#define H5FD_LOG_TIME_OPEN 0x00000800
49#define H5FD_LOG_TIME_STAT 0x00001000
50#define H5FD_LOG_TIME_READ 0x00002000
51#define H5FD_LOG_TIME_WRITE 0x00004000
52#define H5FD_LOG_TIME_SEEK 0x00008000
53#define H5FD_LOG_TIME_TRUNCATE 0x00010000
54#define H5FD_LOG_TIME_CLOSE 0x00020000
55#define H5FD_LOG_TIME_IO \
56 (H5FD_LOG_TIME_OPEN | H5FD_LOG_TIME_STAT | H5FD_LOG_TIME_READ | H5FD_LOG_TIME_WRITE | \
57 H5FD_LOG_TIME_SEEK | H5FD_LOG_TIME_TRUNCATE | H5FD_LOG_TIME_CLOSE)
58/* Flags for tracking allocation/release of space in file */
59#define H5FD_LOG_ALLOC 0x00040000
60#define H5FD_LOG_FREE 0x00080000
61#define H5FD_LOG_ALL \
62 (H5FD_LOG_FREE | H5FD_LOG_ALLOC | H5FD_LOG_TIME_IO | H5FD_LOG_NUM_IO | H5FD_LOG_FLAVOR | \
63 H5FD_LOG_FILE_IO | H5FD_LOG_LOC_IO | H5FD_LOG_META_IO)
