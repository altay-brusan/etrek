141inline std::string RegularExpressionMatch::match(int n) const
142{
143 if (this->startp[n] == nullptr) {
144 return std::string();
145 } else {
146 return std::string(
147 this->startp[n],
148 static_cast<std::string::size_type>(this->endp[n] - this->startp[n]));
149 }
