593inline const char* vtkVolumeProperty::GetInterpolationTypeAsString()
594{
595 if (this->InterpolationType == VTK_NEAREST_INTERPOLATION)
596 {
597 return "Nearest Neighbor";
598 }
599 if (this->InterpolationType == VTK_LINEAR_INTERPOLATION)
600 {
601 return "Linear";
602 }
603 return "Unknown";
604}
