288// Macro to create the interface "C" functions used in
289// a dll or shared library that contains a VTK object factory.
290// Put this function in the .cxx file of your object factory,
291// and pass in the name of the factory sub-class that you want
292// the dll to create.

294 extern "C" VTK_FACTORY_INTERFACE_EXPORT const char* vtkGetFactoryVersion() \
295 { \
296 return VTK_SOURCE_VERSION; \
297 } \
298 extern "C" VTK_FACTORY_INTERFACE_EXPORT vtkObjectFactory* vtkLoad() \
299 { \
300 return factoryName ::New(); \
301 }

303// Macro to implement the body of the object factory form of the New() method.
304#define VTK_OBJECT_FACTORY_NEW_BODY(thisClass) \
305 vtkObject* ret = vtkObjectFactory::CreateInstance(#thisClass, false); \
306 if (ret) \
307 { \
308 return static_cast<thisClass*>(ret); \
309 } \
310 auto result = new thisClass; \
311 result->InitializeObjectBase(); \
312 return result
313
314// Macro to implement the body of the abstract object factory form of the New()
315// method, i.e. an abstract base class that can only be instantiated if the
316// object factory overrides it.
317#define VTK_ABSTRACT_OBJECT_FACTORY_NEW_BODY(thisClass) \
318 vtkObject* ret = vtkObjectFactory::CreateInstance(#thisClass, true); \
319 if (ret) \
320 { \
321 return static_cast<thisClass*>(ret); \
322 } \
323 vtkGenericWarningMacro("Error: no override found for '" #thisClass "'."); \
324 return nullptr
325
326// Macro to implement the body of the standard form of the New() method.
327#if defined(VTK_ALL_NEW_OBJECT_FACTORY)
328#define VTK_STANDARD_NEW_BODY(thisClass) VTK_OBJECT_FACTORY_NEW_BODY(thisClass)
329#else
330#define VTK_STANDARD_NEW_BODY(thisClass) \
331 auto result = new thisClass; \
332 result->InitializeObjectBase(); \
333 return result
334#endif
335
336// Macro to implement the standard form of the New() method.
337#define vtkStandardNewMacro(thisClass) \
338 thisClass* thisClass::New() \
339 { \
340 VTK_STANDARD_NEW_BODY(thisClass); \
341 }
342
343// Macro to implement the ExtendedNew() to create an object in a memkind extended memory space. If
344// VTK is not compiled with VTK_USE_MEMKIND this is equivalent to New()
345#define vtkStandardExtendedNewMacro(thisClass) \
346 thisClass* thisClass::ExtendedNew() \
347 { \
348 auto mkhold = vtkMemkindRAII(true); \
349 (void)mkhold; \
350 return thisClass::New(); \
351 }

353// Macro to implement the object factory form of the New() method.
354#define vtkObjectFactoryNewMacro(thisClass) \
355 thisClass* thisClass::New() \
356 { \
357 VTK_OBJECT_FACTORY_NEW_BODY(thisClass); \
358 }
