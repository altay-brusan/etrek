149/*
150 * Tag data type information.
151 *
152 * Note: RATIONALs are the ratio of two 32-bit integer values.
153 *--:
154 * Note2: TIFF_IFD8 data type is used in tiffFields[]-tag definition in order to
155 distinguish the write-handling of those tags between ClassicTIFF and BigTiff:
156 For ClassicTIFF libtiff writes a 32-bit value and the TIFF_IFD
157 type-id into the file For BigTIFF libtiff writes a 64-bit value and the
158 TIFF_IFD8 type-id into the file
159 */
160typedef enum
161{
162 TIFF_NOTYPE = 0, /* placeholder */
163 TIFF_BYTE = 1, /* 8-bit unsigned integer */
164 TIFF_ASCII = 2, /* 8-bit bytes w/ last byte null */
165 TIFF_SHORT = 3, /* 16-bit unsigned integer */
166 TIFF_LONG = 4, /* 32-bit unsigned integer */
167 TIFF_RATIONAL = 5, /* 64-bit unsigned fraction */
168 TIFF_SBYTE = 6, /* !8-bit signed integer */
169 TIFF_UNDEFINED = 7, /* !8-bit untyped data */
170 TIFF_SSHORT = 8, /* !16-bit signed integer */
171 TIFF_SLONG = 9, /* !32-bit signed integer */
172 TIFF_SRATIONAL = 10, /* !64-bit signed fraction */
173 TIFF_FLOAT = 11, /* !32-bit IEEE floating point */
174 TIFF_DOUBLE = 12, /* !64-bit IEEE floating point */
175 TIFF_IFD = 13, /* %32-bit unsigned integer (offset) */
176 TIFF_LONG8 = 16, /* BigTIFF 64-bit unsigned integer */
177 TIFF_SLONG8 = 17, /* BigTIFF 64-bit signed integer */
178 TIFF_IFD8 = 18 /* BigTIFF 64-bit unsigned integer (offset) */
179} TIFFDataType;
