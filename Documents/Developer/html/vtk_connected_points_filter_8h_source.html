256inline const char* vtkConnectedPointsFilter::GetExtractionModeAsString()
257{
258 if (this->ExtractionMode == VTK_EXTRACT_POINT_SEEDED_REGIONS)
259 {
260 return "ExtractPointSeededRegions";
261 }
262 else if (this->ExtractionMode == VTK_EXTRACT_SPECIFIED_REGIONS)
263 {
264 return "ExtractSpecifiedRegions";
265 }
266 else if (this->ExtractionMode == VTK_EXTRACT_ALL_REGIONS)
267 {
268 return "ExtractAllRegions";
269 }
270 else if (this->ExtractionMode == VTK_EXTRACT_CLOSEST_POINT_REGION)
271 {
272 return "ExtractClosestPointRegion";
273 }
274 else
275 {
276 return "ExtractLargestRegion";
277 }
278}
