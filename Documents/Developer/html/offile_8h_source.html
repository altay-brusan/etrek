404 {

410 result = wfopen(filename.getWideCharPointer(), wmodes);

414#endif

744 int fseek(offile_off_t off, int whence)
745 {
746 int result;
747#ifdef _WIN32
748 // Windows does not have a 64-bit fseek.
749 // We emulate fseek through fsetpos, which does exist on Windows.
750 // fpos_t is (hopefully always) defined as __int64 on this platform
751 offile_fpos_t off2 = off;
752 fpos_t pos;
753 struct _stati64 buf;
754 switch (whence)
755 {
756 case SEEK_END:
757 // flush write buffer, if any, so that the file size is correct
758 :: fflush(file_);
759#if 0
760 // Python implementation based on _lseeki64(). May be unsafe because
761 // there is no guarantee that fflush also empties read buffers.
762 :: fflush(file_);
763#ifdef fileno
764 if (_lseeki64( fileno(file_), 0, 2) == -1)
765#else
766 if (_lseeki64(:: fileno(file_), 0, 2) == -1)
767#endif
768 {
769 storeLastError();
770 return -1;
771 }
772 // fall through
773#else
774 // determine file size (using underlying file descriptor). This should be safe.
775#ifdef fileno
776 if (_fstati64( fileno(file_), &buf) == -1)
777#else
778 if (_fstati64(:: fileno(file_), &buf) == -1)
779#endif
780 {
781 storeLastError();
782 return -1;
783 }
784
785 // fsetpos position is offset + file size.
786 off2 += buf.st_size;
787 break;
788#endif
789 case SEEK_CUR:
790 if (:: fgetpos(file_, &pos) != 0)
791 {
792 storeLastError();
793 return -1;
794 }
795
796 off2 += pos;
797 break;
798 case SEEK_SET:
799 /* do nothing */
800 break;
801 }
802 result = this->fsetpos(&off2);
803#else
804#ifdef HAVE_FSEEKO
805#ifdef EXPLICIT_LFS_64
806 result = :: fseeko64(file_, off, whence);
807#else
808 result = :: fseeko(file_, off, whence);
809#endif
810#else
811 result = :: fseek(file_, off, whence);
812#endif
813#endif
814 if (result) storeLastError();
815 return result;
816 }

821 offile_off_t ftell()
822 {
823#ifdef _WIN32
824 // Windows does not have a 64-bit ftell, and _telli64 cannot be used
825 // because it operates on file descriptors and ignores FILE buffers.
826 // We emulate ftell through fgetpos, which does exist on Windows.
827 // fpos_t is (hopefully always) defined as __int64 on this platform.
828 offile_fpos_t pos;

842#endif
