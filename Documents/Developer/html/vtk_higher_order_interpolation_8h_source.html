125template <int N>
126int vtkHigherOrderInterpolation::NumberOfIntervals(const int order[N])
127{
128 int ni = 1;
129 for (int n = 0; n < N; ++n)
130 {
131 ni *= order[n];
132 }
133 return ni;
134}
