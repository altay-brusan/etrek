41 void add_element(size_t element_id) const
42 {
43 assert(element_id != 0);

66 // NOTE: Not used at all by `Face` or `FaceGenerator` class, but are used by

88 struct IOSS_EXPORT FaceEqual
89 {
90 bool operator()(const Face &left, const Face &right) const
91 {
92 if (left.hashId_ != right.hashId_) {
93 return false;
94 }
95 // Hash (hashId_) is equal
96 // Check whether same vertices (can be in different order)
97 // Most (All?) of the time, there are no hashId_ collisions, so this test will not
98 // find a difference and the function will return 'true'
99 // However, for some reason, removing this check does not change the execution time
100 // appreiciably...
101
102 // TODO: Loop can probably be replaced by std::all_of...
103 for (auto lvert : left.connectivity_) {
104 if (std::find(right.connectivity_.cbegin(), right.connectivity_.cend(), lvert) ==
105 right.connectivity_.cend()) {
106 // Not found, therefore not the same.
107 return false;
108 }
109 }
110 return true;
111 }
112 };
