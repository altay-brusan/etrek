33// in order to test or use the System V message queue implementation
34// on platforms such as Linux that support both Posix and System V,
35// uncomment the following line:
36// #undef HAVE_MQUEUE_H
37
38// in order to test or use the message queue implementation based
39// on unix domain sockets and a separate queue handler thread,
40// uncomment the following line:
41// #define DCMTK_USE_UNIX_SOCKET_QUEUE
42
43// Android has <mqueue.h> and <sys/msg.h> but no implementation
44// of Posix or System V message queues
45#ifdef __ANDROID__
46#define DCMTK_USE_UNIX_SOCKET_QUEUE
47#endif
48
49// GNU Hurd has <mqueue.h> but no working implementation
50// of Posix message queues
51#ifdef __gnu_hurd__
52#define DCMTK_USE_UNIX_SOCKET_QUEUE
53#endif
54
55// There may be other platforms where we have threads and
56// Unix domain sockets but neither Posix nor System V
57// message queues. Use the Unix domain socket based implementation
58// in this case.
59#if defined(WITH_THREADS) && defined(HAVE_SYS_UN_H) && !defined(_WIN32) && !defined(HAVE_MQUEUE_H) && !defined(HAVE_SYS_MSG_H)
60#define DCMTK_USE_UNIX_SOCKET_QUEUE
61#endif
62
63// Note: On FreeBSD, Posix message queues are available since FreeBSD 7,
64// but must be explicitly enabled by loading the "mqueuefs" kernel module
65// (i.e. by calling "kldload mqueuefs" as root user).
66// System V message queues, however, are always available. Therefore,
67// we prefer these when compiling on FreeBSD.
68
69#if defined(HAVE_MQUEUE_H) && !defined(__FreeBSD__)
70BEGIN_EXTERN_C
71#include <mqueue.h> /* for mqd_t */
72END_EXTERN_C
73#endif
