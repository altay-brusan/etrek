23/***** Macros for One linked collective IO case. *****/
24/* The default value to do one linked collective IO for all chunks.
25 If the average number of chunks per process is greater than this value,
26 the library will create an MPI derived datatype to link all chunks to do collective IO.
27 The user can set this value through an API. */
28
29#define H5D_ONE_LINK_CHUNK_IO_THRESHOLD 0
30/***** Macros for multi-chunk collective IO case. *****/
31/* The default value of the threshold to do collective IO for this chunk.
32 If the average percentage of processes per chunk is greater than the default value,
33 collective IO is done for this chunk.
34*/
35
36#define H5D_MULTI_CHUNK_IO_COL_THRESHOLD 60
