96 // Compute a 32-bit hash of a string.
97 // Unlike the original, this version handles embedded null characters so that
98 // unicode multi-byte sequences can be hashed.
99 inline static constexpr uint32_t hash_32_fnv1a_const(
100 const char* const str, std::size_t size, const uint32_t value = hash32a_const) noexcept
101 {
102 return (!str || size <= 0)
103 ? value
104 : hash_32_fnv1a_const(&str[1], size - 1, (value ^ uint32_t(str[0])) * hash32b_const);
105 }
