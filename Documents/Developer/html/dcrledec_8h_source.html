71 inline OFCondition decompress(void *compressedData, size_t compressedSize)
72 {
73 // we allow a call for zero bytes
74 if (compressedSize == 0) return EC_Normal;
75
76 OFCondition result = EC_IllegalCall;
77
78 // check parameters passed by caller
79 if (compressedData == NULL) fail_ = 1;
80
81 if (! fail_) // if fail_ is true, just ignore input
82 {
83 result = EC_Normal;
84 unsigned char ch;
85 unsigned char nbytes;
86 unsigned char *cp = OFstatic_cast(unsigned char *, compressedData);
87
88 // check if we suspended last time, clean up
89 if (suspendInfo_ > 128)
90 {
91 // DICOM packbit scheme uses 257 - nbytes to represent replicate runs
92 nbytes = OFstatic_cast(unsigned char, 257 - suspendInfo_);
93
94 // suspended replicate run. compressedSize cannot be zero now.
95 suspendInfo_ = 128;
96
97 ch = *cp++;
98 --compressedSize;
99 replicate(ch, nbytes);
100 }
101 else if (suspendInfo_ < 128)
102 {
103 // suspended literal run
104 nbytes = OFstatic_cast(unsigned char, (suspendInfo_ & 0x7f) + 1);
105 suspendInfo_ = 128;
106 if (compressedSize < nbytes)
107 {
108 // we're going to suspend again (oops?), prepare everything
109 suspendInfo_ = OFstatic_cast(unsigned char, nbytes - compressedSize - 1);
110 nbytes = OFstatic_cast(unsigned char, compressedSize);
111 result = EC_StreamNotifyClient;
112 }
113
114 literal(cp, nbytes);
115 compressedSize -= nbytes;
116 cp += nbytes;
117 }
118
119 // continue with ordinary RLE decompression
120 while (compressedSize && (! fail_))
121 {
122 ch = *cp++;
123 --compressedSize;
124
125 if (ch & 0x80)
126 {
127 // replicate run
128 if (compressedSize)
129 {
130 // DICOM packbit scheme uses 257 - nbytes to represent replicate runs
131 nbytes = OFstatic_cast(unsigned char, 257 - ch);
132 ch = *cp++;
133 --compressedSize;
134 replicate(ch, nbytes);
135 }
136 else
137 {
138 // suspension: replicate run but second byte is in next block
139 suspendInfo_ = ch;
140 result = EC_StreamNotifyClient;
141 }
142 }
143 else
144 {
145 // literal run
146 nbytes = OFstatic_cast(unsigned char, (ch & 0x7f) + 1);
147 if (compressedSize < nbytes)
148 {
149 // we're going to suspend, prepare everything
150 suspendInfo_ = OFstatic_cast(unsigned char, nbytes - compressedSize - 1);
151 nbytes = OFstatic_cast(unsigned char, compressedSize);
152 result = EC_StreamNotifyClient;
153 }
154
155 literal(cp, nbytes);
156 compressedSize -= nbytes;
157 cp += nbytes;
158 }
159 }
160
161 // failure status at this point means output buffer overflow
162 if (fail_) result = EC_CorruptedData;
163 }
164
165 return result;
166 }
