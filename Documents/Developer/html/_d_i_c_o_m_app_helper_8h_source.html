213 return 1;
214 }
215
216 //
217 // DICOM standard says that spaces (0x20) are to
218 // be ignored for CS types. We don't handle this
219 // well yet.

221 std::string str1(*this->PhotometricInterpretation);
222 std::string rgb("RGB ");

225 {
226 return 3;
227 }
228 else
229 {
230 return 1;
231 }
