18//
19// Pure virtual class that specifies the interface
20// for a DICOMCallback.
21//
22// The DICOMParser allows a vector of callbacks to
23// be specified for each group element tag. When
24// a group, element with a registered callback is
25// encountered, the callback is called and passed
26// the group, element, type, data, and data length.
27//
28
29VTK_ABI_NAMESPACE_BEGIN
30class DICOM_EXPORT DICOMCallback
31{
32public:
33 virtual ~DICOMCallback() = default;
34 virtual void Execute(DICOMParser* parser, doublebyte group, doublebyte element,
35 DICOMParser::VRTypes type, unsigned char* val, quadbyte len) = 0;
36};
37
38//
39// Subclass of DICOMCallback which can be used
40// with member functions.
41//
42template <class T>
43class DICOMMemberCallback : public DICOMCallback
44{
45public:
46 typedef void (T::*TMemberFunctionPointer)(DICOMParser* parser, doublebyte group,
47 doublebyte element, DICOMParser::VRTypes type, unsigned char* val, quadbyte len);
48
49 //
50 // Method to set the object and member function pointers
51 // that will be called in the callback.
52 //
53 void SetCallbackFunction(T* object, TMemberFunctionPointer memberFunction)
54 {
55 ObjectThis = object;
56 MemberFunction = memberFunction;
57 }
58
59 //
60 // Execute method implementation from DICOMCallback.
61 //
62 void Execute(DICOMParser* parser, doublebyte group, doublebyte element, DICOMParser::VRTypes type,
63 unsigned char* val, quadbyte len) override
64 {
65 if (MemberFunction)
66 {
67 ((*ObjectThis).*(MemberFunction))(parser, group, element, type, val, len);
68 }
69 }
70
71protected:
72 T* ObjectThis;
73 TMemberFunctionPointer MemberFunction;
74};
