209// In-lined for performance
210inline void vtkIdList::InsertId(const vtkIdType i, const vtkIdType vtkid)
211{
212 if (i >= this->Size)
213 {
214 this->Resize(i + 1);
215 }
216 this->Ids[i] = vtkid;
217 if (i >= this->NumberOfIds)
218 {
219 this->NumberOfIds = i + 1;
220 }
221}

223// In-lined for performance
224inline vtkIdType vtkIdList::InsertNextId(const vtkIdType vtkid)
225{
226 if (this->NumberOfIds >= this->Size)
227 {
228 if (!this->Resize(2 * this->NumberOfIds + 1)) // grow by factor of 2
229 {
230 return this->NumberOfIds - 1;
231 }
232 }
233 this->Ids[this->NumberOfIds++] = vtkid;
234 return this->NumberOfIds - 1;
235}

237inline vtkIdType vtkIdList::IsId(vtkIdType vtkid) VTK_FUTURE_CONST
238{
239 vtkIdType *ptr, i;
240 for (ptr = this->Ids, i = 0; i < this->NumberOfIds; i++, ptr++)
