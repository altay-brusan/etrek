58 static int calculateCoefficients(const T1 *x,
59 const T2 *y,
60 const unsigned int n,
61 const unsigned int o,
62 T3 *c)
63 {
64 int result = 0;
65 if ((x != NULL) && (y != NULL) && (c !=NULL) && (n > 0))
66 {
67 const unsigned int order = o + 1;
68 const unsigned int order2 = order * order;
69 T3 *basis = new T3[order * n];
70 T3 *alpha = new T3[order2];
71 T3 *beta = new T3[order];
72 if ((basis != NULL) && (alpha != NULL) && (beta != NULL))
73 {
74 unsigned int i;
75 unsigned int j;
76 unsigned int k;
77 for (i = 0; i < order; ++i)
78 {
79 for (j = 0; j < n; ++j)
80 {
81 k = i + j * order;
82 if (i == 0)
83 basis[k] = 1;
84 else
85 basis[k] = OFstatic_cast(T3, x[j]) * basis[k - 1];
86 }
87 }
88 T3 sum;
89 for (i = 0; i < order; ++i)
90 {
91 const unsigned int i_order = i * order;
92 for (j = 0; j <= i; ++j)
93 {
94 sum = 0;
95 for (k = 0; k < n; ++k)
96 sum += basis[i + k * order] * basis[j + k * order];
97 alpha[i + j * order] = sum;
98 if (i != j)
99 alpha[j + i_order] = sum;
100 }
101 }
102 for (i = 0; i < order; ++i)
103 {
104 sum = 0;
105 for (j = 0; j < n; ++j)
106 sum += OFstatic_cast(T3, y[j]) * basis[i + j * order];
107 beta[i] = sum;
108 }
109 if (solve(alpha, beta, order))
110 {
111 for (i = 0; i < order; ++i)
112 c[i] = beta[i];
113 result = 1;
114 }
115 }
116 delete[] basis;
117 delete[] alpha;
118 delete[] beta;
119 }
120 return result;
121 }

139 static int calculateValues(const T1 xs,
140 const T1 xe,
141 T2 *y,
142 const unsigned int n,
143 const unsigned int o,
144 const T3 *c)
145 {
146 int result = 0;
147 if ((y != NULL) && (c != NULL) && (n > 0) && (xe > xs))
148 {
149 unsigned int i;
150 unsigned int j;
151 T3 x;
152 T3 x2;
153 T3 w;
154 const T3 xo = OFstatic_cast(T3, xs);
155 const T3 xi = OFstatic_cast(T3, (OFstatic_cast(T3, xe) - OFstatic_cast(T3, xs)) / (n - 1));
156 for (i = 0; i < n; ++i)
157 {
158 x = xo + OFstatic_cast(T3, i) * xi;
159 x2 = 1;
160 w = 0;
161 for (j = 0; j <= o; ++j)
162 {
163 w += c[j] * x2;
164 x2 *= x;
165 }
166 convertValue(w, y[i]); // cut value if necessary
167 }
168 result = 1;
169 }
170 return result;
171 }

245 static int solve(T3 *a,
246 T3 *b,
247 const unsigned int n)
248 {
249 int result = 0;
250 if ((a != NULL) && (b != NULL) && (n > 0))
251 {
252 unsigned int i;
253 unsigned int j;
254 unsigned int k;
255 signed int pivot;
256 T3 mag;
257 T3 mag2;
258 T3 temp;
259 for (i = 0; i < n; ++i)
260 {
261 mag = 0;
262 pivot = -1;
263 for (j = i; j < n; ++j)
264 {
265 mag2 = fabs(a[i + j * n]);
266 if (mag2 > mag)
267 {
268 mag = mag2;
269 pivot = j;
270 }
271 }
272 if ((pivot == -1) || (mag == 0))
273 break;
274 else
275 {
276 const unsigned int piv = OFstatic_cast(unsigned int, pivot);
277 const unsigned int i_n = i * n;
278 if (piv != i)
279 {
280 const unsigned int piv_n = piv * n;
281 for (j = i; j < n; ++j)
282 {
283 temp = a[j + i_n];
284 a[j + i_n] = a[j + piv_n];
285 a[j + piv_n] = temp;
286 }
287 temp = b[i];
288 b[i] = b[piv];
289 b[piv] = temp;
290 }
291 mag = a[i + i_n];
292 for (j = i; j < n; ++j)
293 a[j + i_n] /= mag;
294 b[i] /= mag;
295 for (j = 0; j < n; ++j)
296 {
297 if (i == j)
298 continue;
299 const unsigned int j_n = j * n;
300 mag2 = a[i + j_n];
301 for (k = i; k < n; ++k)
302 a[k + j_n] -= mag2 * a[k + i_n];
303 b[j] -= mag2 * b[i];
304 }
305 result = 1;
306 }
307
308 }
309 }
310 return result;
311 }
