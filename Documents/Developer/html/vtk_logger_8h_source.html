189 enum Verbosity
190 {
191 // Used to mark an invalid verbosity. Do not log to this level.
192 VERBOSITY_INVALID = -10, // Never do LOG_F(INVALID)
193
194 // You may use VERBOSITY_OFF on g_stderr_verbosity, but for nothing else!
195 VERBOSITY_OFF = -9, // Never do LOG_F(OFF)
196
197 VERBOSITY_ERROR = -2,
198 VERBOSITY_WARNING = -1,
199
200 // Normal messages. By default written to stderr.
201 VERBOSITY_INFO = 0,
202
203 // Same as VERBOSITY_INFO in every way.
204 VERBOSITY_0 = 0,
205
206 // Verbosity levels 1-9 are generally not written to stderr, but are written to file.
207 VERBOSITY_1 = +1,
208 VERBOSITY_2 = +2,
209 VERBOSITY_3 = +3,
210 VERBOSITY_4 = +4,
211 VERBOSITY_5 = +5,
212 VERBOSITY_6 = +6,
213 VERBOSITY_7 = +7,
214 VERBOSITY_8 = +8,
215 VERBOSITY_9 = +9,
216
217 // trace level, same as VERBOSITY_9
218 VERBOSITY_TRACE = +9,
219
220 // Don not use higher verbosity levels, as that will make grepping log files harder.
221 VERBOSITY_MAX = +9,
222 };
