124inline void VTKCOMMONCORE_EXPORT from_json(const nlohmann::json& jj, vtkStringToken& tt)
125{
126 if (jj.is_number_integer())
127 {
128 tt = vtkStringToken(jj.get<vtkStringToken::Hash>());
129 }
130 else if (jj.is_string())
131 {
132 tt = jj.get<std::string>();
133 }
134 else

137 }
138}
