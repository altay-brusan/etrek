455inline vtkFFT::ComplexNumber operator+(
456 const vtkFFT::ComplexNumber& lhs, const vtkFFT::ComplexNumber& rhs)
457{
458 return vtkFFT::ComplexNumber{ lhs.r + rhs.r, lhs.i + rhs.i };
459}
460inline vtkFFT::ComplexNumber operator-(
461 const vtkFFT::ComplexNumber& lhs, const vtkFFT::ComplexNumber& rhs)
462{
463 return vtkFFT::ComplexNumber{ lhs.r - rhs.r, lhs.i - rhs.i };
464}
465inline vtkFFT::ComplexNumber operator*(
466 const vtkFFT::ComplexNumber& lhs, const vtkFFT::ComplexNumber& rhs)
467{
468 return vtkFFT::ComplexNumber{ (lhs.r * rhs.r) - (lhs.i * rhs.i),
469 (lhs.r * rhs.i) + (lhs.i * rhs.r) };
470}
471inline vtkFFT::ComplexNumber operator*(
472 const vtkFFT::ComplexNumber& lhs, const vtkFFT::ScalarNumber& rhs)
473{
474 return vtkFFT::ComplexNumber{ lhs.r * rhs, lhs.i * rhs };
475}
476inline vtkFFT::ComplexNumber operator/(
477 const vtkFFT::ComplexNumber& lhs, const vtkFFT::ComplexNumber& rhs)
478{
479 const double divisor = rhs.r * rhs.r + rhs.i * rhs.i;
480 return vtkFFT::ComplexNumber{ ((lhs.r * rhs.r) + (lhs.i * rhs.i)) / divisor,
481 ((lhs.i * rhs.r) - (lhs.r * rhs.i)) / divisor };
482}
483inline vtkFFT::ComplexNumber operator/(
484 const vtkFFT::ComplexNumber& lhs, const vtkFFT::ScalarNumber& rhs)
485{
486 return vtkFFT::ComplexNumber{ lhs.r / rhs, lhs.i / rhs };
487}

529 return 0.42 - 0.5 * cosin + 0.08 * (2.0 * cosin * cosin - 1.0);
530}

538//------------------------------------------------------------------------------
539template <typename T>
540void vtkFFT::GenerateKernel1D(T* kernel, std::size_t n, WindowGenerator generator)
541{
542 std::size_t half = (n / 2) + (n % 2);
543 for (std::size_t i = 0; i < half; ++i)
544 {
545 kernel[i] = kernel[n - 1 - i] = generator(i, n);
546 }
547}
548
549//------------------------------------------------------------------------------
550template <typename T>
551void vtkFFT::GenerateKernel2D(T* kernel, std::size_t n, std::size_t m, WindowGenerator generator)
552{
553 const std::size_t halfX = (n / 2) + (n % 2);
554 const std::size_t halfY = (m / 2) + (m % 2);
555 for (std::size_t i = 0; i < halfX; ++i)
556 {
557 for (std::size_t j = 0; j < halfY; ++j)
558 {
559 // clang-format off
560 kernel[i][j]
561 = kernel[n - 1 - i][j]
562 = kernel[i][m - 1 - j]
563 = kernel[n - 1 - i][m - 1 - j]
564 = generator(i, n) * generator(j, m);
565 // clang-format on
566 }
567 }
568}
