492 template <typename T>
493 bool verify_field_exists_in_entity_group(const std::string &field_name,
494 const std::vector<T *> &entity_container)
495 {
496 bool found = false;
497 Ioss::Field::RoleType role = Ioss::Field::RoleType::INTERNAL;
498
499 for (const T *entity : entity_container) {
500 if (entity->field_exists(field_name)) {
501 Ioss::Field field = entity->get_field(field_name);
502
503 if (found && field.get_role() != role) {
504 std::ostringstream errmsg;
505 // Would be nice to use fmt:: here, but we need to avoid using fmt includes in public
506 // headers...
507 errmsg << "ERROR: Field " << field.get_name() << " with role " << field.role_string()
508 << " on entity " << entity->name() << " does not match previously found role "
509 << Ioss::Field::role_string(role) << ".\n";
510 IOSS_ERROR(errmsg);
511 }
512
513 found = true;
514 role = field.get_role();
515 }
516 }
517
518 return found;
519 }

522 template <typename T>
523 size_t get_field_data_count_for_entities(const std::string &field_name,
524 const std::vector<T *> &entity_container)
525 {
526 size_t count = 0;
527
528 for (const T *entity : entity_container) {
529 if (entity->field_exists(field_name)) {
530 Ioss::Field field = entity->get_field(field_name);
531
532 count += entity->entity_count() * field.raw_storage()->component_count();
533 }
534 }
535
536 return count;
537 }

541template <typename T, typename U>
542std::vector<size_t> Ioss::Region::get_entity_field_data(const std::string &field_name,
543 const std::vector<T *> &entity_container,
544 std::vector<U> &field_data) const
545{
546 bool field_exists = verify_field_exists_in_entity_group(field_name, entity_container);
547
548 if (!field_exists) {
549 return std::vector<size_t>(entity_container.size() + 1, 0);
550 }
551
552 size_t field_count = impl::get_field_data_count_for_entities(field_name, entity_container);
553
554 field_data.resize(field_count);
555 size_t data_size = field_count * sizeof(U);
556
557 std::vector<size_t> offsets =
558 internal_get_entity_field_data(field_name, entity_container, Data(field_data), data_size);
559
560 assert(offsets.size() == (entity_container.size() + 1));
561 assert(offsets[entity_container.size()] == field_count);
562
563 return offsets;
564}
