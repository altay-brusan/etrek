400template<typename T>
401DSRTree<T>::DSRTree(const DSRTree<T> &tree)
402 : DSRTreeNodeCursor<T>(),
403 RootNode(NULL)
404{
405 if (!tree.isEmpty())
406 {
407 E_AddMode addMode = AM_afterCurrent;
408 T *newNode = NULL;
409 T *nodeCursor = tree.getRoot();
410 OFStack<T *> nodeCursorStack;
411 /* perform a "deep search", just like DSRTreeNodeCursor<>::iterate() */
412 while (nodeCursor != NULL)
413 {
414 /* clone current node and add it to the tree */
415 if (addNode(newNode = nodeCursor->clone(), addMode) == 0)
416 {
417 /* failed to add node, so delete it and exit the loop */
418 delete newNode;
419 break;
420 }
421 /* then goto to the next node to be copied */
422 if (nodeCursor->Down != NULL)
423 {
424 /* go one level down to the first child node */
425 nodeCursorStack.push(nodeCursor);
426 nodeCursor = nodeCursor->getDown();
427 addMode = AM_belowCurrent;
428 }
429 else if (nodeCursor->Next != NULL)
430 {
431 /* proceed with the following sibling */
432 nodeCursor = nodeCursor->getNext();
433 addMode = AM_afterCurrent;
434 } else {
435 /* check whether there are any siblings on higher levels */
436 do {
437 if (!nodeCursorStack.empty())
438 {
439 nodeCursor = nodeCursorStack.top();
440 nodeCursorStack.pop();
441 this->goUp();
442 } else
443 nodeCursor = NULL;
444 } while ((nodeCursor != NULL) && (nodeCursor->Next == NULL));
445 if (nodeCursor != NULL)
446 {
447 nodeCursor = nodeCursor->getNext();
448 addMode = AM_afterCurrent;
449 }
450 }
451 }
452 /* initialize the cursor */
453 gotoRoot();
454 }
455}

468template<typename T>
469DSRTree<T>::DSRTree(const DSRTreeNodeCursor<T> &startCursor,
470 size_t stopAfterNodeID)
471 : DSRTreeNodeCursor<T>(),
472 RootNode(NULL)
473{
474 T *nodeCursor = startCursor.getNode();
475 /* since we start from a particular node, we need to check it first */
476 if (nodeCursor != NULL)
477 {
478 E_AddMode addMode = AM_afterCurrent;
479 T *newNode = NULL;
480 OFStack<T *> nodeCursorStack;
481 /* use current node if none was specified */
482 if (stopAfterNodeID == 0)
483 stopAfterNodeID = nodeCursor->getIdent();
484 /* perform a "deep search", just like DSRTreeNodeCursor<>::iterate() */
485 while (nodeCursor != NULL)
486 {
487 /* clone current node and add it to the tree */
488 if (addNode(newNode = nodeCursor->clone(), addMode) == 0)
489 {
490 /* failed to add node, so delete it and exit the loop */
491 delete newNode;
492 break;
493 }
494 /* then goto to the next node to be copied */
495 if (nodeCursor->Down != NULL)
496 {
497 /* go one level down to the first child node */
498 nodeCursorStack.push(nodeCursor);
499 nodeCursor = nodeCursor->getDown();
500 addMode = AM_belowCurrent;
501 }
502 else if (nodeCursor->Next != NULL)
503 {
504 /* check whether the last node has been processed */
505 if (nodeCursor->getIdent() == stopAfterNodeID)
506 {
507 /* exit the loop */
508 nodeCursor = NULL;
509 } else {
510 /* proceed with the following sibling */
511 nodeCursor = nodeCursor->getNext();
512 addMode = AM_afterCurrent;
513 }
514 } else {
515 /* check whether there are any siblings on higher levels */
516 do {
517 if (!nodeCursorStack.empty())
518 {
519 nodeCursor = nodeCursorStack.top();
520 nodeCursorStack.pop();
521 this->goUp();
522 /* check whether the last node has been processed */
523 if ((nodeCursor != NULL) && (nodeCursor->getIdent() == stopAfterNodeID))
524 {
525 /* exit the loop */
526 nodeCursor = NULL;
527 }
528 } else
529 nodeCursor = NULL;
530 } while ((nodeCursor != NULL) && (nodeCursor->Next == NULL));
531 if (nodeCursor != NULL)
532 {
533 nodeCursor = nodeCursor->getNext();
534 addMode = AM_afterCurrent;
535 }
536 }
537 }
538 /* initialize the cursor */
539 gotoRoot();
540 }

686template<typename T>
687size_t DSRTree<T>::addNode(T *node,
688 const E_AddMode addMode)
689{
690 size_t nodeID = 0;
691 /* make sure that 'node' points to a single node or to the "root" of a subtree */
692 if ((node != NULL) && (node->Prev == NULL))
693 {
694 if (this->NodeCursor != NULL)
695 {
696 DSRTreeNode *lastNode = node;
697 /* update references based on 'addMode' */
698 switch (addMode)
699 {
700 case AM_afterCurrent:
701 node->Prev = this->NodeCursor;
702 /* goto last node (sibling), if any */
703 while (lastNode->Next != NULL)
704 lastNode = lastNode->Next;
705 lastNode->Next = this->NodeCursor->Next;
706 /* connect to current node */
707 if (this->NodeCursor->Next != NULL)
708 (this->NodeCursor->Next)->Prev = lastNode;
709 this->NodeCursor->Next = node;
710 ++this->Position;
711 break;
712 case AM_beforeCurrent:
713 node->Prev = this->NodeCursor->Prev;
714 /* goto last node (sibling), if any */
715 while (lastNode->Next != NULL)
716 lastNode = lastNode->Next;
717 lastNode->Next = this->NodeCursor;
718 /* connect to current node */
719 if ((this->NodeCursor->Prev != NULL) && (this->Position > 1))
720 (this->NodeCursor->Prev)->Next = node;
721 else if (!this->NodeCursorStack.empty() && (this->Position == 1))
722 this->NodeCursorStack.top()->Down = node;
723 this->NodeCursor->Prev = lastNode;
724 /* check whether root node has been replaced */
725 if (this->NodeCursor == this->RootNode)
726 this->RootNode = node;
727 break;
728 case AM_belowCurrent:
729 /* store old position */
730 this->Position.goDown();
731 this->NodeCursorStack.push(this->NodeCursor);
732 /* parent node has already child nodes */
733 if (this->NodeCursor->Down != NULL)
734 {
735 DSRTreeNode *tempNode = this->NodeCursor->Down;
736 /* goto last node (sibling) */
737 while (tempNode->Next != NULL)
738 {
739 tempNode = tempNode->Next;
740 ++this->Position;
741 }
742 /* connect to last child */
743 tempNode->Next = node;
744 node->Prev = tempNode;
745 ++this->Position;
746 } else
747 this->NodeCursor->Down = node;
748 break;
749 case AM_belowCurrentBeforeFirstChild:
750 /* store old position */
751 this->Position.goDown();
752 this->NodeCursorStack.push(this->NodeCursor);
753 /* parent node has already child nodes */
754 if (this->NodeCursor->Down != NULL)
755 {
756 /* goto last node (sibling), if any */
757 while (lastNode->Next != NULL)
758 lastNode = lastNode->Next;
759 /* connect to (current) first child */
760 lastNode->Next = this->NodeCursor->Down;
761 (this->NodeCursor->Down)->Prev = lastNode;
762 }
763 this->NodeCursor->Down = node;
764 break;
765 }
766 this->NodeCursor = node;
767 } else {
768 /* originally, the tree was empty */
769 this->RootNode = this->NodeCursor = node;
770 this->Position.initialize();
771 }
772 nodeID = this->NodeCursor->getIdent();
773 }
774 return nodeID;
775}

778template<typename T>
779size_t DSRTree<T>::replaceNode(T *node)
780{
781 size_t nodeID = 0;
782 /* make sure that 'node' points to a single node or to the "root" of a subtree */
783 if ((node != NULL) && (node->Prev == NULL))
784 {
785 if (this->NodeCursor != NULL)
786 {
787 /* connect to previous node */
788 if (this->NodeCursor->Prev != NULL)
789 {
790 (this->NodeCursor->Prev)->Next = node;
791 /* remove reference to former sibling */
792 this->NodeCursor->Prev = NULL;
793 } else {
794 /* is there any direct parent node? */
795 if (!this->NodeCursorStack.empty())
796 {
797 DSRTreeNode *parent = this->NodeCursorStack.top();
798 if (parent != NULL)
799 parent->Down = node;
800 }
801 }
802 /* connect to next node */
803 if (this->NodeCursor->Next != NULL)
804 {
805 DSRTreeNode *lastNode = node;
806 /* goto last node (sibling), if any */
807 while (lastNode->Next != NULL)
808 lastNode = lastNode->Next;
809 (this->NodeCursor->Next)->Prev = lastNode;
810 lastNode->Next = this->NodeCursor->Next;
811 /* remove reference to former sibling */
812 this->NodeCursor->Next = NULL;
813 }
814 /* check whether root node has been replaced */
815 if (this->NodeCursor == this->RootNode)
816 this->RootNode = node;
817 /* free memory of old (now replaced) node */
818 deleteTreeFromRootNode(this->NodeCursor);
819 /* set cursor to new node */
820 this->NodeCursor = node;
821 nodeID = node->getIdent();
822 }
823 }
824 return nodeID;
825}

828template<typename T>
829T *DSRTree<T>::extractNode()
830{
831 T *cursor = this->NodeCursor;
832 /* extract current node (incl. subtree) from tree */
833 if (cursor != NULL)
834 {
835 /* are there any siblings? */
836 if ((cursor->Prev != NULL) || (cursor->Next != NULL))
837 {
838 /* connect to previous node */
839 if (cursor->Prev != NULL)
840 (cursor->Prev)->Next = cursor->Next;
841 else
842 {
843 /* is there any direct parent node? */
844 if (!this->NodeCursorStack.empty())
845 {
846 DSRTreeNode *parent = this->NodeCursorStack.top();
847 if (parent != NULL)
848 parent->Down = cursor->Next;
849 }
850 }
851 /* connect to next node */
852 if (cursor->Next != NULL)
853 {
854 (cursor->Next)->Prev = cursor->Prev;
855 if (this->NodeCursor == this->RootNode)
856 this->RootNode = cursor->getNext(); // old root node deleted
857 this->NodeCursor = cursor->getNext();
858 } else {
859 /* set cursor to previous node since there is no next node */
860 this->NodeCursor = cursor->getPrev();
861 --this->Position;
862 }
863 } else {
864 /* no siblings: check for child nodes */
865 if (!this->NodeCursorStack.empty())
866 {
867 this->NodeCursor = this->NodeCursorStack.top();
868 this->NodeCursorStack.pop();
869 this->Position.goUp();
870 /* should never be NULL, but ... */
871 if (this->NodeCursor != NULL)
872 this->NodeCursor->Down = NULL;
873 else
874 {
875 this->RootNode = NULL; // tree is now empty
876 this->Position.clear();
877 }
878 } else {
879 this->RootNode = this->NodeCursor = NULL; // tree is now empty
880 this->Position.clear();
881 }
882 }
883 /* remove references to former siblings */
884 cursor->Prev = NULL;
885 cursor->Next = NULL;
886 }
887 return cursor;
888}

901template<typename T>
902size_t DSRTree<T>::removeNode()
903{
904 size_t nodeID = 0;
905 /* extract current node (incl. subtree) from tree */
906 T *cursor = extractNode();
907 if (cursor != NULL)
908 {
909 /* delete all nodes from extracted subtree */
910 /* (could also use the "new" DSRTreeNodeCursor class) */
911 T *delNode = NULL;
912 OFStack<T *> cursorStack;
913 while (cursor != NULL)
914 {
915 delNode = cursor;
916 if (cursor->Down != NULL)
917 {
918 if (cursor->Next != NULL)
919 cursorStack.push(cursor->getNext());
920 cursor = cursor->getDown();
921 }
922 else if (cursor->Next != NULL)
923 cursor = cursor->getNext();
924 else if (!cursorStack.empty())
925 {
926 cursor = cursorStack.top();
927 cursorStack.pop();
928 } else
929 cursor = NULL;
930 delete delNode;
931 }
932 /* return identifier of (new) current node */
933 if (this->NodeCursor != NULL)
934 nodeID = this->NodeCursor->getIdent();
935 }
936 return nodeID;
937}
