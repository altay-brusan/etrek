272inline const char* vtkPolyDataConnectivityFilter::GetExtractionModeAsString()
273{
274 if (this->ExtractionMode == VTK_EXTRACT_POINT_SEEDED_REGIONS)
275 {
276 return "ExtractPointSeededRegions";
277 }
278 else if (this->ExtractionMode == VTK_EXTRACT_CELL_SEEDED_REGIONS)
279 {
280 return "ExtractCellSeededRegions";
281 }
282 else if (this->ExtractionMode == VTK_EXTRACT_SPECIFIED_REGIONS)
283 {
284 return "ExtractSpecifiedRegions";
285 }
286 else if (this->ExtractionMode == VTK_EXTRACT_ALL_REGIONS)
287 {
288 return "ExtractAllRegions";
289 }
290 else if (this->ExtractionMode == VTK_EXTRACT_CLOSEST_POINT_REGION)
291 {
292 return "ExtractClosestPointRegion";
293 }
294 else
295 {
296 return "ExtractLargestRegion";
297 }
298}
