52// We have been relying on the assumption that calling `.data()` on an empty vector
53// will return `nullptr`. However, according to cppreference (based on the standard):
54//
55// `If size() is 0, data() may or may not return a null pointer.`
56//
57// We don't have any systems on which we have found that (yet?), but this is proactive
58// in removing our use of `.data()` on potentially empty vectors...
59template <typename T> IOSS_NODISCARD constexpr T *Data(std::vector<T> &vec)
60{
61 if (vec.empty()) {
62 return nullptr;
63 }
64 return vec.data();
65}

184 template <typename T> static void generate_index(std::vector<T> &index)
185 {
186 T sum = 0;

188 T cnt = index[i];

195 template <typename T>
196 IOSS_NODISCARD static T find_index_location(T node, const std::vector<T> &index)
197 {
198 // 0-based node numbering
199 // index[p] = first node (0-based) on processor p
200
201#if 1
202 // Assume data coherence. I.e., a new search will be close to the
203 // previous search.
204 static size_t prev = 1;
205
206 size_t nproc = index.size();
207 if (prev < nproc && index[prev - 1] <= node && index[prev] > node) {
208 return prev - 1;
209 }
210
211 for (size_t p = 1; p < nproc; p++) {
212 if (index[p] > node) {
213 prev = p;
214 return p - 1;
215 }
216 }
217 std::ostringstream errmsg;
218 errmsg << "FATAL ERROR: find_index_location. Searching for " << node << " in:\n";
219 for (auto idx : index) {
220 errmsg << idx << ", ";
221 }
222 errmsg << "\n";
223 IOSS_ERROR(errmsg);
224#else
225 return std::distance(index.begin(), std::upper_bound(index.begin(), index.end(), node)) - 1;
226#endif
227 }

265 IOSS_NODISCARD static constexpr int number_width(const size_t number, bool use_commas = false)
266 {
267 if (number == 0) {
268 return 1;
269 }
270 int width = static_cast<int>(std::floor(std::log10(number))) + 1;
271 if (use_commas) {
272 width += ((width - 1) / 3);
273 }
274 return width;

277 IOSS_NODISCARD static constexpr int power_2(int count)
278 {
279 // Return the power of two which is equal to or greater than `count`
280 // count = 15 -> returns 16
281 // count = 16 -> returns 16
282 // count = 17 -> returns 32
283
284 // Use brute force...
285 int pow2 = 1;
286 while (pow2 < count) {
287 pow2 *= 2;
288 }
289 return pow2;
290 }

584 static void insert_sort_and_unique(const Ioss::NameList &src, Ioss::NameList &dest)
585 {
586 dest.insert(dest.end(), src.begin(), src.end());
587 std::sort(dest.begin(), dest.end(), std::less<>());
588 auto endIter = std::unique(dest.begin(), dest.end());
589 dest.resize(endIter - dest.begin());

593 // SEE: http://lemire.me/blog/2017/04/10/removing-duplicates-from-lists-quickly
594 template <typename T> static size_t unique(std::vector<T> &out, bool skip_first)
595 {
596 if (out.empty()) {
597 return 0;
598 }
599 size_t i = 1;
600 size_t pos = 1;
601 T oldv = out[0];
602 if (skip_first) {
603 i = 2;
604 pos = 2;
605 oldv = out[1];
606 }
607 for (; i < out.size(); ++i) {
608 T newv = out[i];
609 out[pos] = newv;
610 pos += (newv != oldv);
611 oldv = newv;
612 }
613 return pos;
614 }
