172inline char* vtkTestUtilities::ExpandDataFileName(
173 int argc, char* argv[], const char* fname, int slash)
174{
175#ifdef __EMSCRIPTEN__
176 // determine where the file is located on the host file system using args/env.
177 char* hostPath = vtkTestUtilities::ExpandFileNameWithArgOrEnvOrDefault(
178 "-D", argc, argv, "VTK_DATA_ROOT", "../../../../VTKData", fname, slash);
179 // preload file from host into the sandbox
180 const size_t n = strlen(fname) + 1;
181 char* sandboxPath = new char[n];
182 strncpy(sandboxPath, fname, n);
183 vtkEmscriptenTestUtilities::PreloadDataFile(hostPath, sandboxPath);
184 delete[] hostPath;
185 return sandboxPath;
186#else
187 return vtkTestUtilities::ExpandFileNameWithArgOrEnvOrDefault(
188 "-D", argc, argv, "VTK_DATA_ROOT", "../../../../VTKData", fname, slash);
189#endif
190}

192inline char* vtkTestUtilities::GetArgOrEnvOrDefault(
193 const char* arg, int argc, char* argv[], const char* env, const char* def)
194{
195 int index = -1;
196
197 for (int i = 0; i < argc; i++)
198 {
199 if (strcmp(arg, argv[i]) == 0 && i < argc - 1)
200 {
201 index = i + 1;
202 }
203 }
204
205 char* value;
206
207 if (index != -1)
208 {
209 value = new char[strlen(argv[index]) + 1];
210 strcpy(value, argv[index]);
211 }
212 else
213 {
214 char* foundenv = getenv(env);
215 if (foundenv)
216 {
217 value = new char[strlen(foundenv) + 1];
218 strcpy(value, foundenv);
219 }
220 else if (def)
221 {
222 value = new char[strlen(def) + 1];
223 strcpy(value, def);
224 }
225 else
226 {
227 value = nullptr;
228 }
229 }
230
231 return value;
232}

234inline char* vtkTestUtilities::ExpandFileNameWithArgOrEnvOrDefault(const char* arg, int argc,
235 char* argv[], const char* env, const char* def, const char* fname, int slash)
236{
237 char* fullName;
238
239 char* value = vtkTestUtilities::GetArgOrEnvOrDefault(arg, argc, argv, env, def);
240 if (value)
241 {
242 fullName = new char[strlen(value) + strlen(fname) + 2 + static_cast<size_t>(slash ? 1 : 0)];
243 fullName[0] = 0;
244 strcat(fullName, value);
245 size_t len = strlen(fullName);
246 fullName[len] = '/';
247 fullName[len + 1] = 0;
248 strcat(fullName, fname);
249 }
250 else
251 {
252 fullName = new char[strlen(fname) + 1 + static_cast<size_t>(slash ? 1 : 0)];
253 strcpy(fullName, fname);
254 }
255
256 if (slash)
257 {
258 strcat(fullName, "/");
259 }
260
261 delete[] value;
262
263 return fullName;
264}
