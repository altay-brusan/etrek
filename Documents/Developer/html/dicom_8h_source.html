99/*
100 * Useful Types
101 */
102
103/* these types are for backwards compatibility */
104typedef Uint16 U_SHORT; /* normal unsigned short*/
105typedef Uint32 U_LONG; /* normal unsigned long */
106typedef Uint32 MASK_32; /* for bit masks */
107
108
109#define DICOM_STDAPPLICATIONCONTEXT UID_StandardApplicationContext
110
111/* network type constants as used by the DUL layer */
112#define DICOM_APPLICATION_ACCEPTOR 0x01
113#define DICOM_APPLICATION_REQUESTOR 0x02
114
115/*
116** The following defines are used in some levels of the network code.
117*/
118
119#define DIC_US_LEN sizeof(DIC_US)
120#define DIC_UL_LEN sizeof(DIC_UL)
121#define DIC_UI_LEN 64
122#define DIC_AE_LEN 16
123#define DIC_SH_LEN 16
124#define DIC_PN_LEN 64
125#define DIC_LO_LEN 64
126#define DIC_CS_LEN 16
127#define DIC_DS_LEN 16
128#define DIC_IS_LEN 12
129#define DIC_AT_LEN (2*sizeof(DIC_US))
130
131/* a valid DNS name can be up to 253 characters, and we need to add
132 * a port number. This should be sufficient.
133 */
134#define DIC_NODENAME_LEN 270
135
136#define DICOM_UI_LENGTH DIC_UI_LEN
137
138typedef Uint16 DIC_US;
139typedef Uint32 DIC_UL;
140typedef char DIC_UI[DIC_UI_LEN + 1];
141typedef char DIC_AE[DIC_AE_LEN + 1];
142typedef char DIC_SH[DIC_SH_LEN + 1];
143typedef char DIC_PN[DIC_PN_LEN + 1];
144typedef char DIC_LO[DIC_LO_LEN + 1];
145typedef char DIC_CS[DIC_CS_LEN + 1];
146typedef char DIC_DS[DIC_DS_LEN + 1];
147typedef char DIC_IS[DIC_IS_LEN + 1];
148typedef char DIC_NODENAME[DIC_NODENAME_LEN + 1];

151/*
152 * Useful Macros
153 */
154
155#ifndef MAX
156#define MAX(x, y) (((x) < (y)) ? (y) : (x))
157#endif
158#ifndef MIN
159#define MIN(x, y) (((x) < (y)) ? (x) : (y))
160#endif
161#ifndef IS_EVEN
162#define IS_EVEN(i) (~(i) & 0x01)
163#endif
164#ifndef DIM_OF
165#define DIM_OF(a) (sizeof(a) / sizeof(a[0]))
166#endif
167#ifndef IN_RANGE
168#define IN_RANGE(n, lo, hi) ((lo) <= n && (n) <= (hi))
169#endif
