150 enum class GradientModes : int
151 {
152 // Background color is used at the bottom, Background2 color is used at the top.
153 VTK_GRADIENT_VERTICAL,
154 // Background color on the left, Background2 color on the right.
155 VTK_GRADIENT_HORIZONTAL,
156 // Background color in the center, Background2 color on and beyond the ellipse edge.
157 // Ellipse touches all sides of the viewport. The ellipse is a circle for viewports with equal
158 // width and height.
159 VTK_GRADIENT_RADIAL_VIEWPORT_FARTHEST_SIDE,
160 // Background color in the center, Background2 color on and beyond the ellipse edge.
161 // Ellipse touches all corners of the viewport. The ellipse is a circle for viewports with equal
162 // width and height.
163 VTK_GRADIENT_RADIAL_VIEWPORT_FARTHEST_CORNER,
164 };
