531 out[1] = ((in[1] < 0.0) ? (static_cast<unsigned int>(-in[1] * VTKKW_FP_SCALE + 0.5))

537inline void vtkFixedPointVolumeRayCastMapper::FixedPointIncrement(
538 unsigned int position[3], unsigned int increment[3])
539{
540 if (increment[0] & 0x80000000)
541 {
542 position[0] += (increment[0] & 0x7fffffff);
543 }
544 else
545 {
546 position[0] -= increment[0];
547 }
548 if (increment[1] & 0x80000000)
549 {
550 position[1] += (increment[1] & 0x7fffffff);
551 }
552 else
553 {
554 position[1] -= increment[1];
555 }
556 if (increment[2] & 0x80000000)
557 {
558 position[2] += (increment[2] & 0x7fffffff);
559 }
560 else
561 {
562 position[2] -= increment[2];
563 }
564}

589inline int vtkFixedPointVolumeRayCastMapper::CheckMinMaxVolumeFlag(unsigned int mmpos[3], int c)
590{
591 vtkIdType offset = static_cast<vtkIdType>(this->MinMaxVolumeSize[3]) *
592 (mmpos[2] * static_cast<vtkIdType>(this->MinMaxVolumeSize[0] * this->MinMaxVolumeSize[1]) +
593 mmpos[1] * static_cast<vtkIdType>(this->MinMaxVolumeSize[0]) + mmpos[0]) +
594 static_cast<vtkIdType>(c);
595
596 return ((*(this->MinMaxVolume + 3 * offset + 2)) & 0x00ff);
597}

599inline int vtkFixedPointVolumeRayCastMapper::CheckMIPMinMaxVolumeFlag(
600 unsigned int mmpos[3], int c, unsigned short maxIdx, int flip)
601{
602 vtkIdType offset = static_cast<vtkIdType>(this->MinMaxVolumeSize[3]) *
603 (mmpos[2] * static_cast<vtkIdType>(this->MinMaxVolumeSize[0] * this->MinMaxVolumeSize[1]) +
604 mmpos[1] * static_cast<vtkIdType>(this->MinMaxVolumeSize[0]) + mmpos[0]) +
605 static_cast<vtkIdType>(c);
606
607 if ((*(this->MinMaxVolume + 3 * offset + 2) & 0x00ff))
608 {
609 if (flip)
610 {
611 return (*(this->MinMaxVolume + 3 * offset) < maxIdx);
612 }
613 else
614 {
615 return (*(this->MinMaxVolume + 3 * offset + 1) > maxIdx);
616 }
617 }
618 else
619 {
620 return 0;
621 }
622}

624inline void vtkFixedPointVolumeRayCastMapper::LookupColorUC(unsigned short* colorTable,
625 unsigned short* scalarOpacityTable, unsigned short index, unsigned char color[4])
626{
627 unsigned short alpha = scalarOpacityTable[index];
628 color[0] = static_cast<unsigned char>(
629 (colorTable[3 * index] * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
630 color[1] = static_cast<unsigned char>(
631 (colorTable[3 * index + 1] * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
632 color[2] = static_cast<unsigned char>(
633 (colorTable[3 * index + 2] * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
634 color[3] = static_cast<unsigned char>(alpha >> (VTKKW_FP_SHIFT - 8));
635}

637inline void vtkFixedPointVolumeRayCastMapper::LookupDependentColorUC(unsigned short* colorTable,
638 unsigned short* scalarOpacityTable, unsigned short index[4], int components,
639 unsigned char color[4])
640{
641 unsigned short alpha;
642 switch (components)
643 {
644 case 2:
645 alpha = scalarOpacityTable[index[1]];
646 color[0] = static_cast<unsigned char>(
647 (colorTable[3 * index[0]] * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
648 color[1] = static_cast<unsigned char>(
649 (colorTable[3 * index[0] + 1] * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
650 color[2] = static_cast<unsigned char>(
651 (colorTable[3 * index[0] + 2] * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
652 color[3] = static_cast<unsigned char>(alpha >> (VTKKW_FP_SHIFT - 8));
653 break;
654 case 4:
655 alpha = scalarOpacityTable[index[3]];
656 color[0] = static_cast<unsigned char>((index[0] * alpha + 0x7fff) >> VTKKW_FP_SHIFT);
657 color[1] = static_cast<unsigned char>((index[1] * alpha + 0x7fff) >> VTKKW_FP_SHIFT);
658 color[2] = static_cast<unsigned char>((index[2] * alpha + 0x7fff) >> VTKKW_FP_SHIFT);
659 color[3] = static_cast<unsigned char>(alpha >> (VTKKW_FP_SHIFT - 8));
660 break;
661 }
662}

664inline void vtkFixedPointVolumeRayCastMapper::LookupAndCombineIndependentColorsUC(
665 unsigned short* colorTable[4], unsigned short* scalarOpacityTable[4], unsigned short index[4],
666 float weights[4], int components, unsigned char color[4])
667{
668 unsigned int tmp[4] = { 0, 0, 0, 0 };
669
670 for (int i = 0; i < components; i++)
671 {
672 unsigned short alpha =
673 static_cast<unsigned short>(static_cast<float>(scalarOpacityTable[i][index[i]]) * weights[i]);
674 tmp[0] += static_cast<unsigned char>(
675 ((colorTable[i][3 * index[i]]) * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
676 tmp[1] += static_cast<unsigned char>(
677 ((colorTable[i][3 * index[i] + 1]) * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
678 tmp[2] += static_cast<unsigned char>(
679 ((colorTable[i][3 * index[i] + 2]) * alpha + 0x7fff) >> (2 * VTKKW_FP_SHIFT - 8));
680 tmp[3] += static_cast<unsigned char>(alpha >> (VTKKW_FP_SHIFT - 8));
681 }
682
683 color[0] = static_cast<unsigned char>((tmp[0] > 255) ? (255) : (tmp[0]));
684 color[1] = static_cast<unsigned char>((tmp[1] > 255) ? (255) : (tmp[1]));
685 color[2] = static_cast<unsigned char>((tmp[2] > 255) ? (255) : (tmp[2]));
686 color[3] = static_cast<unsigned char>((tmp[3] > 255) ? (255) : (tmp[3]));
687}

689inline int vtkFixedPointVolumeRayCastMapper::CheckIfCropped(unsigned int pos[3])
690{
691 int idx;
692
693 if (pos[2] < this->FixedPointCroppingRegionPlanes[4])
694 {
695 idx = 0;
696 }
697 else if (pos[2] > this->FixedPointCroppingRegionPlanes[5])
698 {
699 idx = 18;
700 }
701 else
702 {
703 idx = 9;
704 }
705
706 if (pos[1] >= this->FixedPointCroppingRegionPlanes[2])
707 {
708 if (pos[1] > this->FixedPointCroppingRegionPlanes[3])
709 {
710 idx += 6;
711 }
712 else
713 {
714 idx += 3;
715 }
716 }
717
718 if (pos[0] >= this->FixedPointCroppingRegionPlanes[0])
719 {
720 if (pos[0] > this->FixedPointCroppingRegionPlanes[1])
721 {
722 idx += 2;
723 }
724 else
725 {
726 idx += 1;
727 }
728 }
729
730 return !(static_cast<unsigned int>(this->CroppingRegionFlags) & this->CroppingRegionMask[idx]);
731}
