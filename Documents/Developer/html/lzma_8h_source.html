33/*****************************
34 * Required standard headers *
35 *****************************/
36
37/*
38 * liblzma API headers need some standard types and macros. To allow
39 * including lzma.h without requiring the application to include other
40 * headers first, lzma.h includes the required standard headers unless
41 * they already seem to be included already or if LZMA_MANUAL_HEADERS
42 * has been defined.
43 *
44 * Here's what types and macros are needed and from which headers:
45 * - stddef.h: size_t, NULL
46 * - stdint.h: uint8_t, uint32_t, uint64_t, UINT32_C(n), uint64_C(n),
47 * UINT32_MAX, UINT64_MAX
48 *
49 * However, inttypes.h is a little more portable than stdint.h, although
50 * inttypes.h declares some unneeded things compared to plain stdint.h.
51 *
52 * The hacks below aren't perfect, specifically they assume that inttypes.h
53 * exists and that it typedefs at least uint8_t, uint32_t, and uint64_t,
54 * and that, in case of incomplete inttypes.h, unsigned int is 32-bit.
55 * If the application already takes care of setting up all the types and
56 * macros properly (for example by using gnulib's stdint.h or inttypes.h),
57 * we try to detect that the macros are already defined and don't include
58 * inttypes.h here again. However, you may define LZMA_MANUAL_HEADERS to
59 * force this file to never include any system headers.
60 *
61 * Some could argue that liblzma API should provide all the required types,
62 * for example lzma_uint64, LZMA_UINT64_C(n), and LZMA_UINT64_MAX. This was
63 * seen as an unnecessary mess, since most systems already provide all the
64 * necessary types and macros in the standard headers.
65 *
66 * Note that liblzma API still has lzma_bool, because using stdbool.h would
67 * break C89 and C++ programs on many systems. sizeof(bool) in C99 isn't
68 * necessarily the same as sizeof(bool) in C++.
69 */
70
71#include "vtk_lzma_mangle.h"
72
73#ifndef LZMA_MANUAL_HEADERS
74 /*
75 * I suppose this works portably also in C++. Note that in C++,
76 * we need to get size_t into the global namespace.
77 */
78# include <stddef.h>
79
80 /*
81 * Skip inttypes.h if we already have all the required macros. If we
82 * have the macros, we assume that we have the matching typedefs too.
83 */
84# if !defined(UINT32_C) || !defined(UINT64_C) \
85 || !defined(UINT32_MAX) || !defined(UINT64_MAX)
86 /*
87 * MSVC versions older than 2013 have no C99 support, and
88 * thus they cannot be used to compile liblzma. Using an
89 * existing liblzma.dll with old MSVC can work though(*),
90 * but we need to define the required standard integer
91 * types here in a MSVC-specific way.
92 *
93 * (*) If you do this, the existing liblzma.dll probably uses
94 * a different runtime library than your MSVC-built
95 * application. Mixing runtimes is generally bad, but
96 * in this case it should work as long as you avoid
97 * the few rarely-needed liblzma functions that allocate
98 * memory and expect the caller to free it using free().
99 */
100# if defined(_WIN32) && defined(_MSC_VER) && _MSC_VER < 1800
101 typedef unsigned __int8 uint8_t;
102 typedef unsigned __int32 uint32_t;
103 typedef unsigned __int64 uint64_t;
104# else
105 /* Use the standard inttypes.h. */
106# ifdef __cplusplus
107 /*
108 * C99 sections 7.18.2 and 7.18.4 specify
109 * that C++ implementations define the limit
110 * and constant macros only if specifically
111 * requested. Note that if you want the
112 * format macros (PRIu64 etc.) too, you need
113 * to define __STDC_FORMAT_MACROS before
114 * including lzma.h, since re-including
115 * inttypes.h with __STDC_FORMAT_MACROS
116 * defined doesn't necessarily work.
117 */
118# ifndef __STDC_LIMIT_MACROS
119# define __STDC_LIMIT_MACROS 1
120# endif
121# ifndef __STDC_CONSTANT_MACROS
122# define __STDC_CONSTANT_MACROS 1
123# endif
124# endif
125
126# include <inttypes.h>
127# endif
128
129 /*
130 * Some old systems have only the typedefs in inttypes.h, and
131 * lack all the macros. For those systems, we need a few more
132 * hacks. We assume that unsigned int is 32-bit and unsigned
133 * long is either 32-bit or 64-bit. If these hacks aren't
134 * enough, the application has to setup the types manually
135 * before including lzma.h.
136 */
137# ifndef UINT32_C
138# if defined(_WIN32) && defined(_MSC_VER)
139# define UINT32_C(n) n ## UI32
140# else
141# define UINT32_C(n) n ## U
142# endif
143# endif
144
145# ifndef UINT64_C
146# if defined(_WIN32) && defined(_MSC_VER)
147# define UINT64_C(n) n ## UI64
148# else
149 /* Get ULONG_MAX. */
150# include <limits.h>
151# if ULONG_MAX == 4294967295UL
152# define UINT64_C(n) n ## ULL
153# else
154# define UINT64_C(n) n ## UL
155# endif
156# endif
157# endif
158
159# ifndef UINT32_MAX
160# define UINT32_MAX (UINT32_C(4294967295))
161# endif
162
163# ifndef UINT64_MAX
164# define UINT64_MAX (UINT64_C(18446744073709551615))
165# endif
166# endif
167#endif /* ifdef LZMA_MANUAL_HEADERS */

170/******************
171 * LZMA_API macro *
172 ******************/
173
174/*
175 * Some systems require that the functions and function pointers are
176 * declared specially in the headers. LZMA_API_IMPORT is for importing
177 * symbols and LZMA_API_CALL is to specify the calling convention.
178 *
179 * By default it is assumed that the application will link dynamically
180 * against liblzma. #define LZMA_API_STATIC in your application if you
181 * want to link against static liblzma. If you don't care about portability
182 * to operating systems like Windows, or at least don't care about linking
183 * against static liblzma on them, don't worry about LZMA_API_STATIC. That
184 * is, most developers will never need to use LZMA_API_STATIC.
185 *
186 * The GCC variants are a special case on Windows (Cygwin and MinGW-w64).
187 * We rely on GCC doing the right thing with its auto-import feature,
188 * and thus don't use __declspec(dllimport). This way developers don't
189 * need to worry about LZMA_API_STATIC. Also the calling convention is
190 * omitted on Cygwin but not on MinGW-w64.
191 */
192#ifndef LZMA_API_IMPORT
193# if !defined(LZMA_API_STATIC) && defined(_WIN32) && !defined(__GNUC__)
194# define LZMA_API_IMPORT __declspec(dllimport)
195# else
196# define LZMA_API_IMPORT
197# endif
198#endif
199
200#ifndef LZMA_API_CALL
201# if defined(_WIN32) && !defined(__CYGWIN__)
202# define LZMA_API_CALL __cdecl
203# else
204# define LZMA_API_CALL
205# endif
206#endif
207
208#ifndef LZMA_API
209# define LZMA_API(type) LZMA_API_IMPORT type LZMA_API_CALL
210#endif

214 * nothrow *
215 ***********/
216
217/*
218 * None of the functions in liblzma may throw an exception. Even
219 * the functions that use callback functions won't throw exceptions,
220 * because liblzma would break if a callback function threw an exception.
221 */
222#ifndef lzma_nothrow
223# if defined(__cplusplus)
224# if __cplusplus >= 201103L || (defined(_MSVC_LANG) \
225 && _MSVC_LANG >= 201103L)
226# define lzma_nothrow noexcept
227# else
228# define lzma_nothrow throw()
229# endif
230# elif defined(__GNUC__) && (__GNUC__ > 3 \
231 || (__GNUC__ == 3 && __GNUC_MINOR__ >= 3))
232# define lzma_nothrow __attribute__((__nothrow__))
233# else
234# define lzma_nothrow
235# endif
236#endif

244 * GNU C extensions are used conditionally in the public API. It doesn't
245 * break anything if these are sometimes enabled and sometimes not, only
246 * affects warnings and optimizations.

254# ifndef lzma_attr_warn_unused_result
255# if __GNUC__ == 3 && __GNUC_MINOR__ < 4
256# define lzma_attr_warn_unused_result
257# endif
258# endif

261# ifndef lzma_attribute
262# define lzma_attribute(attr)
263# endif
264#endif
