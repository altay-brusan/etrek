186 inline int getMinMaxWindow(const int idx,
187 double &center,
188 double &width)
189 {
190 int result = 0;
191 if ((idx >= 0) && (idx <= 1))
192 {
193 if ((idx == 1) && (MinValue[1] == 0) && (MaxValue[1] == 0))
194 determineMinMax(0, 0, 0x2); // determine on demand
195 /* suppl. 33: "A Window Center of 2^n-1 and a Window Width of 2^n
196 selects the range of input values from 0 to 2^n-1."
197 */
198 center = (OFstatic_cast(double, MinValue[idx]) + OFstatic_cast(double, MaxValue[idx]) + 1) / 2; // type cast to avoid overflows !
199 width = OFstatic_cast(double, MaxValue[idx]) - OFstatic_cast(double, MinValue[idx]) + 1;
200 result = (width > 0); // check for valid value
201 }
202 return result;
203 }

219 virtual int getRoiWindow(const unsigned long left_pos,
220 const unsigned long top_pos,
221 const unsigned long width,
222 const unsigned long height,
223 const unsigned long columns,
224 const unsigned long rows,
225 const unsigned long frame,
226 double &voiCenter,
227 double &voiWidth)
228 {
229 int result = 0;
230 if ((Data != NULL) && (left_pos < columns) && (top_pos < rows))
231 {
232 T *p = Data + (columns * rows * frame) + (top_pos * columns) + left_pos;
233 const unsigned long right_pos = (left_pos + width < columns) ? left_pos + width : columns;
234 const unsigned long bottom = (top_pos + height < rows) ? top_pos + height : rows;
235 const unsigned long skip_x = left_pos + (columns - right_pos);
236 unsigned long x;
237 unsigned long y;
238 T value = 0;
239 T min = *p; // get first pixel as initial value for min ...
240 T max = min; // ... and max
241 for (y = top_pos; y < bottom; ++y)
242 {
243 for (x = left_pos; x < right_pos; ++x)
244 {
245 value = *(p++);
246 if (value < min)
247 min = value;
248 else if (value > max)
249 max = value;
250 }
251 p += skip_x; // skip rest of current line and beginning of next
252 }
253 /* suppl. 33: "A Window Center of 2^n-1 and a Window Width of 2^n
254 selects the range of input values from 0 to 2^n-1."
255 */
256 voiCenter = (OFstatic_cast(double, min) + OFstatic_cast(double, max) + 1) / 2; // type cast to avoid overflows !
257 voiWidth = OFstatic_cast(double, max) - OFstatic_cast(double, min) + 1;
258 result = (width > 0); // check for valid value
259 }
260 return result;
261 }

271 int getHistogramWindow(const double thresh, // could be optimized if necessary (see diinpxt.h)!
272 double &center,
273 double &width)
274 {
275 if ((Data != NULL) && (MinValue[0] < MaxValue[0]))
276 {
277 const Uint32 count = OFstatic_cast(Uint32, MaxValue[0] - MinValue[0] + 1);
278 Uint32 *quant = new Uint32[count];
279 if (quant != NULL)
280 {
281 unsigned long i;
282 OFBitmanipTemplate<Uint32>::zeroMem(quant, count); // initialize array
283 for (i = 0; i < Count; ++i)
284 {
285 if ((Data[i] >= MinValue[0]) && (Data[i] <= MaxValue[0])) // only for stability !
286 ++quant[OFstatic_cast(Uint32, Data[i] - MinValue[0])]; // count values
287#ifdef DEBUG
288 else
289 DCMIMGLE_WARN("invalid value (" << Data[i] << ") in DiMonoPixelTemplate<T>::getHistogramWindow()");
290#endif
291 }
292 const Uint32 threshvalue = OFstatic_cast(Uint32, thresh * OFstatic_cast(double, Count));
293 Uint32 t = 0;
294 i = 0;
295 while ((i < count) && (t < threshvalue))
296 t += quant[i++];
297 const T minvalue = (i < count) ? OFstatic_cast(T, MinValue[0] + i) : 0;
298 t = 0;
299 i = count;
300 while ((i > 0) && (t < threshvalue))
301 t += quant[--i];
302 const T maxvalue = (i > 0) ? OFstatic_cast(T, MinValue[0] + i) : 0;
303 delete[] quant;
304 if (minvalue < maxvalue)
305 {
306 /* suppl. 33: "A Window Center of 2^n-1 and a Window Width of 2^n
307 selects the range of input values from 0 to 2^n-1."
308 */
309 center = (OFstatic_cast(double, minvalue) + OFstatic_cast(double, maxvalue) + 1) / 2;
310 width = OFstatic_cast(double, maxvalue) - OFstatic_cast(double, minvalue) + 1;
311 return (width > 0);
312 }
313 }
314 }
315 return 0;
316 }

360 void determineMinMax(T minvalue = 0,
361 T maxvalue = 0,
362 const int mode = 0x1)
363 {
364 if (Data != NULL)
365 {
366 if (mode & 0x1)
367 {
368 if ((minvalue == 0) && (maxvalue == 0))
369 {
370 DCMIMGLE_DEBUG("determining global minimum and maximum pixel values for monochrome image");
371 T *p = Data;
372 T value = *p;
373 unsigned long i;
374 minvalue = value;
375 maxvalue = value;
376 for (i = Count; i > 1; --i) // could be optimized if necessary (see diinpxt.h) !
377 {
378 value = *(++p);
379 if (value < minvalue)
380 minvalue = value;
381 else if (value > maxvalue)
382 maxvalue = value;
383 }
384 }
385 MinValue[0] = minvalue; // global minimum
386 MaxValue[0] = maxvalue; // global maximum
387 MinValue[1] = 0; // invalidate value
388 MaxValue[1] = 0;
389 } else {
390 minvalue = MinValue[0];
391 maxvalue = MaxValue[0];
392 }
393 if (mode & 0x2)
394 {
395 DCMIMGLE_DEBUG("determining next minimum and maximum pixel values for monochrome image");
396 T *p = Data;
397 T value;
398 int firstmin = 1;
399 int firstmax = 1;
400 unsigned long i;
401 for (i = Count; i != 0; --i) // could be optimized if necessary (see diinpxt.h) !
402 {
403 value = *(p++);
404 if ((value > minvalue) && ((value < MinValue[1]) || firstmin))
405 {
406 MinValue[1] = value;
407 firstmin = 0;
408 }
409 if ((value < maxvalue) && ((value > MaxValue[1]) || firstmax))
410 {
411 MaxValue[1] = value;
412 firstmax = 0;
413 }
414 }
415 }
416 }
