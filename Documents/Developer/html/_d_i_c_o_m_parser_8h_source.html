93 //
94 // Defined DICOM types.
95 //
96 enum VRTypes
97 {
98 VR_UNKNOWN = 0x0,
99 VR_OB = 0x424f, // Other byte string (string of bytes, insensitive to byte order)
100 VR_AW = 0x5741,
101 VR_AE = 0x4541, // Application Entity (char string)
102 VR_AS = 0x5341, // Age string (char string)
103 VR_CS = 0x5343, // Code string (char string, leading/trailing spaces insignificant)
104 VR_UI = 0x4955, // UID (character string)
105 VR_DA = 0x4144, // Date (char string)
106 VR_DS = 0x5344, // Decimal string (char string representing a fixed/floating pt number)
107 VR_DT = 0x5444, // Date Time (char string)
108 VR_IS = 0x5349, // Integer string (char string representing an integer)
109 VR_LO = 0x4f4c, // Long string (character string padded with leading/trailing spaces)
110 VR_LT = 0x544c, // Long text (character string with one or more paragraphs.)
111 VR_OW = 0x574f, // Other word string (sensitive to byte order, transfer syntax)
112 VR_PN = 0x4e50, // Person name (character string)
113 VR_ST = 0x5453, // Short text (character string, one or more paragraphs)
114 VR_TM = 0x4d54, // Time (character string)
115 VR_UN = 0x4e55,
116 VR_UT = 0x5455, // Unlimited text (character string)
117 VR_SQ = 0x5153, // Sequence of items
118 VR_SH = 0x4853, // Short string (character string with leading/trailing spaces).
119 VR_FL = 0x4c46, // Floating point single precision
120 VR_SL = 0x4c53, // Signed long, (signed 32bits, 2's complement form.)
121 VR_AT = 0x5441, // Attribute tag (ordered pair 16 bit unsigned)
122 VR_UL = 0x4c55, // Unsigned long (unsigned 32 bits)
123 VR_US = 0x5355, // Unsigned short (unsigned 16 bits)
124 VR_SS = 0x5353, // Signed short, (signed 16bits, 2's complement)
125 VR_FD = 0x4446 // Floating point double precision
126 };
