18 // An enum that defines the neighboring orientation which is stored in a
19 // 3-tuple vtkStructuredNeighbor::Orientation. In each dimension, there
20 // is a high and low end, the orientation tuple defines how to grow ghost
21 // layers along each dimension.
22 enum NeighborOrientation
23 {
24 SUBSET_LO = -2, // The grid is a subset of the neighboring grid and the
25 // ghost layers are pointing away from the low end
26 LO = -1, // The grid partially overlap with its neighbor on the
27 // low end, thus, ghost layers are pointing away from
28 // the low end
29 ONE_TO_ONE = 0, // grids abut 1-to-1 in both HI and LO, the
30 // cardinality of both grids is the same in the
31 // corresponding dimension.
32 HI = 1, // The grid partially overlaps with its neighbor on the
33 // high end, thus, ghost layers are pointing away from
34 // the high end
35 SUBSET_HI = 2, // The grid is a subset of the neighboring grid and the
36 // ghost layers are pointing away from the high end
37 SUBSET_BOTH = 3, // The grid is a subset of the neighboring grid and the
38 // ghost layers grow from both low and high ends.
39 SUPERSET = 4, // grid is a superset of the neighboring grid in the
40 // given direction.
41 UNDEFINED = 5 // the neighboring relationship is undefined, e.g., if
42 // we are checking 2D data, the neighbor orientation
43 // in the 3rd dimension is undefined.
44 };
45
46 // Class Member Variables made public for easier access
47 int NeighborID; // The registered ID of the neighboring grid
48 int OverlapExtent[6]; // The extent at which the grids overlap
49 int SendExtent[6]; // The extent that we send to this neighbor
50 int RcvExtent[6]; // The extent that we receive from this neighbor
51 int Orientation[3]; // Defines how we are neighboring with this grid, see
52 // NeighborOrientation enum above.

85 vtkStructuredNeighbor& operator=(const vtkStructuredNeighbor& N)
86 {
87 if (this != &N)
88 {
89 this->Orientation[0] = N.Orientation[0];
90 this->Orientation[1] = N.Orientation[1];
91 this->Orientation[2] = N.Orientation[2];
92 this->NeighborID = N.NeighborID;
93 for (int i = 0; i < 6; ++i)
94 {
95 this->SendExtent[i] = N.SendExtent[i];
96 this->RcvExtent[i] = N.RcvExtent[i];
97 this->OverlapExtent[i] = N.OverlapExtent[i];
98 } // END for
99 } // END if
100 return *this;
101 }
