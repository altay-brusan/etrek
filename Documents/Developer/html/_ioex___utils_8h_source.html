55 using NameTopoKey = std::pair<std::string, const Ioss::ElementTopology *>;
56 struct IOEX_EXPORT NameTopoKeyCompare
57 {
58 IOSS_NODISCARD bool operator()(const NameTopoKey &lhs, const NameTopoKey &rhs) const
59 {
60 assert(lhs.second != nullptr);
61 assert(rhs.second != nullptr);
62 return lhs.first < rhs.first ||
63 (!(rhs.first < lhs.first) && lhs.second->name() < rhs.second->name());
64 }
65 };
66
67 struct IOEX_EXPORT NameTopoKeyHash
68 {
69 IOSS_NODISCARD size_t operator()(const NameTopoKey &name_topo) const
70 {
71 return std::hash<std::string>{}(name_topo.first) +
72 std::hash<size_t>{}((size_t)name_topo.second);
73 }
74 };
