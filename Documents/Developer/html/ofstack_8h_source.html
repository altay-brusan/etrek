194template <class T>
195class OFStack : private OFStackBase
196{
197
198public:
199
201 OFStack() {};
202
204 OFStack(const OFStack<T> & x) : OFStackBase()
205 {
206 copy(x);
207 }
208
210 OFStack<T> &operator=(const OFStack<T> &x)
211 {
212 if (this != &x)
213 {
214 while(!OFStackBase::base_empty())
215 OFStackBase::base_pop();
216 copy(x);
217 }
218 return *this;
219 }
220
224 OFBool empty() const { return OFStackBase::base_empty(); }
225
229 size_t size() const { return OFStackBase::base_size(); }
230
235 T & top()
236 {
237 return (OFstatic_cast(OFStackLink<T> *, OFStackBase::base_top()))->info;
238 }
239
244 const T & top() const
245 {
246 return (OFstatic_cast(const OFStackLink<T> *, OFStackBase::base_top()))->info;
247 }
248
253 void push(const T & x)
254 {
255 OFStackBase::base_push(new OFStackLink<T>(x));
256 }
257
261 void pop() { OFStackBase::base_pop(); }
262
263private:
264
269 int copy(const OFStack<T> & x)
270 {
271 stackSize = x.size();
272 if (stackSize)
273 {
274 head = new OFStackLink<T>((OFstatic_cast(OFStackLink<T>*, x.head))->info);
275 OFStackLinkBase * newPtr = head;
276 OFStackLinkBase * oldPtr = x.head->next;
277 while (oldPtr)
278 {
279 newPtr->next = new OFStackLink<T>((OFstatic_cast(OFStackLink<T>*, oldPtr))->info);
280 oldPtr = oldPtr->next;
281 newPtr = newPtr->next;
282 }
283 }
284 return 0;
285 }
