227 iterator insert(iterator it, const T& v)
228 {
229 size_type idx = it - begin();
230
231 if (size_ == allocated_) {

370 void reserve(size_type n)
371 {
372 T* old_values = values_;
373 T* new_values;
374
375 if (n == 0)
376 n = 1;
377 if (n <= allocated_)
378 return;
379
380 // While we are at it, let's reserve some extra space
381 n += 10;
382
383 new_values = new T[n];
384 if (old_values)
385 {
386 for (size_type i = 0; i < size_; i++)
387 new_values[i] = old_values[i];
388 delete[] old_values;
389 }
390
391 values_ = new_values;
392 allocated_ = n;
393 }
