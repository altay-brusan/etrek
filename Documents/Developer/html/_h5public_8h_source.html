264/* The signed version of size_t
265 *
266 * ssize_t is POSIX and not defined in any C standard. It's used in some
267 * public HDF5 API calls so this work-around will define it if it's not
268 * present.
269 *
270 * Use of ssize_t should be discouraged in new code.
271 */
272#if H5_SIZEOF_SSIZE_T == 0
273/* Undefine this size, we will re-define it in one of the sections below */
274#undef H5_SIZEOF_SSIZE_T
275#if H5_SIZEOF_SIZE_T == H5_SIZEOF_INT
276typedef int ssize_t;
277#define H5_SIZEOF_SSIZE_T H5_SIZEOF_INT
278#elif H5_SIZEOF_SIZE_T == H5_SIZEOF_LONG
279typedef long ssize_t;
280#define H5_SIZEOF_SSIZE_T H5_SIZEOF_LONG
281#elif H5_SIZEOF_SIZE_T == H5_SIZEOF_LONG_LONG
282typedef long long ssize_t;
283#define H5_SIZEOF_SSIZE_T H5_SIZEOF_LONG_LONG
284#else /* Can't find matching type for ssize_t */
285#error "nothing appropriate for ssize_t"
286#endif
287#endif
