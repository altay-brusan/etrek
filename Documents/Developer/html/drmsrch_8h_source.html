40 template<typename IOD, typename Predicate>
41 static inline OFCondition findItem(IOD& iod, Predicate& pred)
42 {
43 OFCondition cond = iod.gotoFirstItem();
44
45 while (cond.good()) {
46 if (pred(iod.getCurrentItem()))
47 break;
48 cond = iod.gotoNextItem();
49 }
50
51 return cond;
52 }
