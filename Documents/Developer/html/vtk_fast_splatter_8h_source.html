152void vtkFastSplatterClamp(T* array, vtkIdType arraySize, T minValue, T maxValue)
153{
154 for (vtkIdType i = 0; i < arraySize; i++)
155 {
156 if (array[i] < minValue)
157 array[i] = minValue;
158 if (array[i] > maxValue)
159 array[i] = maxValue;
160 }
161}

165template <class T>
166void vtkFastSplatterScale(T* array, int numComponents, vtkIdType numTuples, T minValue, T maxValue,
167 double* dataMinValue, double* dataMaxValue)
168{
169 T* a;
170 T min, max;
171 *dataMinValue = 0;
172 *dataMaxValue = 0;
173 vtkIdType t;
174 for (int c = 0; c < numComponents; c++)
175 {
176 // Find the min and max values in the array.
177 a = array + c;
178 min = max = *a;
179 a += numComponents;
180 for (t = 1; t < numTuples; t++, a += numComponents)
181 {
182 if (min > *a)
183 min = *a;
184 if (max < *a)
185 max = *a;
186 }
187
188 // Bias everything so that 0 is really the minimum.
189 if (min != 0)
190 {
191 for (t = 0, a = array + c; t < numTuples; t++, a += numComponents)
192 {
193 *a -= min;
194 }
195 }
196
197 // Scale the values.
198 if (max != min)
199 {
200 for (t = 0, a = array + c; t < numTuples; t++, a += numComponents)
201 {
202 *a = ((maxValue - minValue) * (*a)) / (max - min);
203 }
204 }
205
206 // Bias everything again so that it lies in the correct range.
207 if (minValue != 0)
208 {
209 for (t = 0, a = array + c; t < numTuples; t++, a += numComponents)
210 {
211 *a += minValue;
212 }
213 }
214 if (c == 0)
215 {
216 *dataMinValue = min;
217 *dataMaxValue = max;
218 }
219 }
220}

224template <class T>
225void vtkFastSplatterFrozenScale(
226 T* array, int numComponents, vtkIdType numTuples, T minValue, T maxValue, double min, double max)
227{
228 T* a;
229
230 vtkIdType t;
231 for (int c = 0; c < numComponents; c++)
232 {
233 // Bias everything so that 0 is really the minimum.
234 if (min != 0)
235 {
236 for (t = 0, a = array + c; t < numTuples; t++, a += numComponents)
237 {
238 *a -= static_cast<T>(min);
239 }
240 }
241
242 // Scale the values.
243 if (max != min)
244 {
245 for (t = 0, a = array + c; t < numTuples; t++, a += numComponents)
246 {
247 *a = static_cast<T>(((maxValue - minValue) * (*a)) / (max - min));
248 }
249 }
250
251 // Bias everything again so that it lies in the correct range.
252 if (minValue != 0)
253 {
254 for (t = 0, a = array + c; t < numTuples; t++, a += numComponents)
255 {
256 *a += minValue;
257 }
258 }
259 }
