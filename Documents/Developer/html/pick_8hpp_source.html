64diy::
65distance(const Bounds& bounds, const Point& p)
66{
67 Out res = 0;
68 for (int i = 0; i < p.size(); ++i)
69 {
70 // avoids all the annoying case logic by finding
71 // diff = max(bounds.min[i] - p[i], 0, p[i] - bounds.max[i])
72 Out diff = 0, d;
73
74 d = bounds.min[i] - p[i];
75 if (d > diff) diff = d;

187// wraps block bounds
188// wrap dir is the wrapping direction from original block to wrapped neighbor block
189// overall domain bounds and dimensionality are also needed
190template<class Bounds>
191void
192diy::
193wrap_bounds(Bounds& bounds, Direction wrap_dir, const Bounds& domain)
194{
195 for (int i = 0; i < bounds.min.size(); ++i) // assume min, max of bounds, domain have same size
196 {
197 bounds.min[i] += wrap_dir[i] * (domain.max[i] - domain.min[i]);
198 bounds.max[i] += wrap_dir[i] * (domain.max[i] - domain.min[i]);
199 }
200}
