334template <typename T>
335void FillRegion(T* pArray, const vtkAMRBox& arrayRegion, const vtkAMRBox& destRegion, T fillValue)
336{
337 // Convert regions to array index space. VTK arrays
338 // always start with 0,0,0.
339 int ofs[3];
340 ofs[0] = -arrayRegion.GetLoCorner()[0];
341 ofs[1] = -arrayRegion.GetLoCorner()[1];
342 ofs[2] = -arrayRegion.GetLoCorner()[2];
343 vtkAMRBox arrayDims(arrayRegion);
344 arrayDims.Shift(ofs);
345 vtkAMRBox destDims(destRegion);
346 destDims.Shift(ofs);
347 // Quick sanity check.
348 if (!arrayRegion.Contains(destRegion))
349 {
350 vtkGenericWarningMacro(<< "ERROR: Array must enclose the destination region. "
351 << "Aborting the fill.");
352 }
353 // Get the bounds of the indices we fill.
354 const int* destLo = destDims.GetLoCorner();
355 int destHi[3];
356 destDims.GetValidHiCorner(destHi);
357 // Get the array dimensions.
358 int arrayHi[3];
359 arrayDims.GetNumberOfCells(arrayHi);
360 // Fill.
361 for (int k = destLo[2]; k <= destHi[2]; ++k)
362 {
363 vtkIdType kOfs = k * arrayHi[0] * arrayHi[1];
364 for (int j = destLo[1]; j <= destHi[1]; ++j)
365 {
366 vtkIdType idx = kOfs + j * arrayHi[0] + destLo[0];
367 for (int i = destLo[0]; i <= destHi[0]; ++i)
368 {
369 pArray[idx] = fillValue;
370 ++idx;
371 }
372 }
373 }
375}
