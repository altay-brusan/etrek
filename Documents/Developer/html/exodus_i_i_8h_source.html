348typedef struct ex_basis
349{
350 /*
351 clang-format off
352 *
353 * subc_dim: dimension of the subcell associated with the specified DoF ordinal
354 * -- 0 node, 1 edge, 2 face, 3 volume [Range: 0..3]
355 * subc_ordinal: ordinal of the subcell relative to its parent cell
356 * -- 0..n for each ordinal with the same subc dim [Range: <= DoF ordinal]
357 * subc_dof_ordinal: ordinal of the DoF relative to the subcell
358 * subc_num_dof: cardinality of the DoF set associated with this subcell.
359 * xi, eta, mu (ξ, η, ζ): Parametric coordinate location of the DoF
360 * -- (Only first ndim values are valid)
361 *
362 clang-format on
363 */
364
365 char name[EX_MAX_NAME + 1];
366 int cardinality; /* number of `basis` points == dimension of non-null subc_*, xi, eta, mu */
367 int *subc_dim;
368 int *subc_ordinal;
369 int *subc_dof_ordinal;
370 int *subc_num_dof;
371 double *xi;
372 double *eta;
373 double *zeta;
374} ex_basis;

378 char name[EX_MAX_NAME + 1];
379 int cardinality; /* Number of quadrature points */
380 int dimension; /* 1,2,3 -- spatial dimension of points */

382 double *

387} ex_quadrature;
