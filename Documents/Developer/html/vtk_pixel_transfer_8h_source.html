84//-----------------------------------------------------------------------------
85template <typename SOURCE_TYPE, typename DEST_TYPE>
86int vtkPixelTransfer::Blit(const vtkPixelExtent& srcWholeExt, const vtkPixelExtent& srcSubset,
87 const vtkPixelExtent& destWholeExt, const vtkPixelExtent& destSubset, int nSrcComps,
88 SOURCE_TYPE* srcData, int nDestComps, DEST_TYPE* destData)
89{
90 if ((srcData == nullptr) || (destData == nullptr))
91 {
92 return -1;
93 }
94 if ((srcWholeExt == srcSubset) && (destWholeExt == destSubset) && (nSrcComps == nDestComps))
95 {
96 // buffers are contiguous
97 size_t n = srcWholeExt.Size() * nSrcComps;
98 for (size_t i = 0; i < n; ++i)
99 {
100 destData[i] = static_cast<DEST_TYPE>(srcData[i]);
101 }
102 }
103 else
104 {
105 // buffers are not contiguous
106 int tmp[2];
107
108 // get the dimensions of the arrays
109 srcWholeExt.Size(tmp);
110 int swnx = tmp[0];
111
112 destWholeExt.Size(tmp);
113 int dwnx = tmp[0];
114
115 // move from logical extent to memory extent
116 vtkPixelExtent srcExt(srcSubset);
117 srcExt.Shift(srcWholeExt);
118
119 vtkPixelExtent destExt(destSubset);
120 destExt.Shift(destWholeExt);
121
122 // get size of sub-set to copy (it's the same in src and dest)
123 int nxny[2];
124 srcExt.Size(nxny);
125
126 // use smaller ncomps for loop index to avoid reading/writing
127 // invalid mem
128 int nCopyComps = nSrcComps < nDestComps ? nSrcComps : nDestComps;
129
130 for (int j = 0; j < nxny[1]; ++j)
131 {
132 int sjj = swnx * (srcExt[2] + j) + srcExt[0];
133 int djj = dwnx * (destExt[2] + j) + destExt[0];
134 for (int i = 0; i < nxny[0]; ++i)
135 {
136 int sidx = nSrcComps * (sjj + i);
137 int didx = nDestComps * (djj + i);
138 // copy values from source
139 for (int p = 0; p < nCopyComps; ++p)
140 {
141 destData[didx + p] = static_cast<DEST_TYPE>(srcData[sidx + p]);
142 }
143 // ensure all dest comps are initialized
144 for (int p = nCopyComps; p < nDestComps; ++p)
145 {
146 destData[didx + p] = static_cast<DEST_TYPE>(0);
147 }
148 }
149 }
150 }
151 return 0;
152}
