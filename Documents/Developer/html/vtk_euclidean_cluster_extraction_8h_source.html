232inline const char* vtkEuclideanClusterExtraction::GetExtractionModeAsString()
233{
234 if (this->ExtractionMode == VTK_EXTRACT_POINT_SEEDED_CLUSTERS)
235 {
236 return "ExtractPointSeededClusters";
237 }
238 else if (this->ExtractionMode == VTK_EXTRACT_SPECIFIED_CLUSTERS)
239 {
240 return "ExtractSpecifiedClusters";
241 }
242 else if (this->ExtractionMode == VTK_EXTRACT_ALL_CLUSTERS)
243 {
244 return "ExtractAllClusters";
245 }
246 else if (this->ExtractionMode == VTK_EXTRACT_CLOSEST_POINT_CLUSTER)
247 {
248 return "ExtractClosestPointCluster";
249 }
250 else
251 {
252 return "ExtractLargestCluster";
253 }
254}
