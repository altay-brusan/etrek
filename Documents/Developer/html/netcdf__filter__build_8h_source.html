139/*
140The struct that provides the necessary filter info.
141The combination of version + sort uniquely determines
142the format of the remainder of the struct
143*/
144typedef struct NCZ_codec_t {
145 int version; /* Version number of the struct */
146 int sort; /* Format of remainder of the struct;
147 Currently always NCZ_CODEC_HDF5 */
148 const char* codecid; /* The name/id of the codec */
149 unsigned int hdf5id; /* corresponding hdf5 id */
150 void (*NCZ_codec_initialize)(void);
151 void (*NCZ_codec_finalize)(void);
152 int (*NCZ_codec_to_hdf5)(const char* codec, size_t* nparamsp, unsigned** paramsp);
153 int (*NCZ_hdf5_to_codec)(size_t nparams, const unsigned* params, char** codecp);
154 int (*NCZ_modify_parameters)(int ncid, int varid, size_t* vnparamsp, unsigned** vparamsp, size_t* wnparamsp, unsigned** wparamsp);
155} NCZ_codec_t;
