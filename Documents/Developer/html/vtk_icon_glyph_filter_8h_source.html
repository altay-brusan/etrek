172inline void vtkIconGlyphFilter::IconConvertIndex(int id, int& j, int& k)
173{
174 int dimX = this->IconSheetSize[0] / this->IconSize[0];
175 int dimY = this->IconSheetSize[1] / this->IconSize[1];
176
177 j = id - dimX * (id / dimX);
178 k = dimY - (id / dimX) - 1;
179}
