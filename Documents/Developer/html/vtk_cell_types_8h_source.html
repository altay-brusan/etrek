168inline int vtkCellTypes::IsType(unsigned char type)
169{
170 vtkIdType numTypes = this->GetNumberOfTypes();
171
172 for (vtkIdType i = 0; i < numTypes; i++)
173 {
174 if (type == this->GetCellType(i))
175 {
176 return 1;
177 }
178 }
179 return 0;
180}

182//-----------------------------------------------------------------------------
183inline int vtkCellTypes::IsLinear(unsigned char type)
184{
185 return ((type <= 20) || (type == VTK_CONVEX_POINT_SET) || (type == VTK_POLYHEDRON));
186}
