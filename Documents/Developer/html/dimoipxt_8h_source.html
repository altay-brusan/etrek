54 DiMonoInputPixelTemplate(DiInputPixel *pixel,
55 DiMonoModality *modality)
56 : DiMonoPixelTemplate<T3>(pixel, modality)
57 {
58 if ((pixel != NULL) && (this->Count > 0))
59 {
60 // check whether to apply any modality transform
61 if ((this->Modality != NULL) && this->Modality->hasLookupTable() && (bitsof(T1) <= MAX_TABLE_ENTRY_SIZE))
62 {
63 modlut(pixel);
64 // ignore modality LUT min/max values since the image does not necessarily have to use all LUT entries
65 this->determineMinMax();
66 }
67 else if ((this->Modality != NULL) && this->Modality->hasRescaling())
68 {
69 rescale(pixel, this->Modality->getRescaleSlope(), this->Modality->getRescaleIntercept());
70 this->determineMinMax(OFstatic_cast(T3, this->Modality->getMinValue()), OFstatic_cast(T3, this->Modality->getMaxValue()));
71 } else {
72 rescale(pixel); // "copy" or reference pixel data
73 this->determineMinMax(OFstatic_cast(T3, this->Modality->getMinValue()), OFstatic_cast(T3, this->Modality->getMaxValue()));
74 }
75 /* erase empty part of the buffer (= blacken the background) */
76 if ((this->Data != NULL) && (this->InputCount < this->Count))
77 OFBitmanipTemplate<T3>::zeroMem(this->Data + this->InputCount, this->Count - this->InputCount);
78 }
79 }

122 void modlut(DiInputPixel *input)
123 {
124 const T1 *pixel = OFstatic_cast(const T1 *, input->getData());
125 if ((pixel != NULL) && (this->Modality != NULL))
126 {
127 const DiLookupTable *mlut = this->Modality->getTableData();
128 if (mlut != NULL)
129 {
130 const int useInputBuffer = (sizeof(T1) == sizeof(T3)) && (this->Count <= input->getCount());
131 if (useInputBuffer) // do not copy pixel data, reference them!
132 {
133 DCMIMGLE_DEBUG("re-using input buffer, do not copy pixel data");
134 this->Data = OFstatic_cast(T3 *, input->getDataPtr());
135 input->removeDataReference(); // avoid double deletion
136 } else
137 this->Data = new T3[this->Count];
138 if (this->Data != NULL)
139 {
140 DCMIMGLE_DEBUG("applying modality transformation with LUT (" << mlut->getCount() << " entries)");
141 T2 value = 0;
142 const T2 firstentry = mlut->getFirstEntry(value); // choose signed/unsigned method
143 const T2 lastentry = mlut->getLastEntry(value);
144 const T3 firstvalue = OFstatic_cast(T3, mlut->getFirstValue());
145 const T3 lastvalue = OFstatic_cast(T3, mlut->getLastValue());
146 const T1 *p = pixel + input->getPixelStart();
147 T3 *q = this->Data;
148 unsigned long i;
149 T3 *lut = NULL;
150 const unsigned long ocnt = OFstatic_cast(unsigned long, input->getAbsMaxRange()); // number of LUT entries
151 if (initOptimizationLUT(lut, ocnt))
152 { // use LUT for optimization
153 const T2 absmin = OFstatic_cast(T2, input->getAbsMinimum());
154 q = lut;
155 for (i = 0; i < ocnt; ++i) // calculating LUT entries
156 {
157 value = OFstatic_cast(T2, i) + absmin;
158 if (value <= firstentry)
159 *(q++) = firstvalue;
160 else if (value >= lastentry)
161 *(q++) = lastvalue;
162 else
163 *(q++) = OFstatic_cast(T3, mlut->getValue(value));
164 }
165 const T3 *lut0 = lut - OFstatic_cast(T2, absmin); // points to 'zero' entry
166 q = this->Data;
167 for (i = this->InputCount; i != 0; --i) // apply LUT
168 *(q++) = *(lut0 + (*(p++)));
169 }
170 if (lut == NULL) // use "normal" transformation
171 {
172 for (i = this->InputCount; i != 0; --i)
173 {
174 value = OFstatic_cast(T2, *(p++));
175 if (value <= firstentry)
176 *(q++) = firstvalue;
177 else if (value >= lastentry)
178 *(q++) = lastvalue;
179 else
180 *(q++) = OFstatic_cast(T3, mlut->getValue(value));
181 }
182 }
183 delete[] lut;
184 }
185 }
186 }
187 }

195 void rescale(DiInputPixel *input,
196 const double slope = 1.0,
197 const double intercept = 0.0)
198 {
199 const T1 *pixel = OFstatic_cast(const T1 *, input->getData());
200 if (pixel != NULL)
201 {
202 const int useInputBuffer = (sizeof(T1) == sizeof(T3)) && (this->Count <= input->getCount()) && (input->getPixelStart() == 0);
203 if (useInputBuffer)
204 { // do not copy pixel data, reference them!
205 DCMIMGLE_DEBUG("re-using input buffer, do not copy pixel data");
206 this->Data = OFstatic_cast(T3 *, input->getDataPtr());
207 input->removeDataReference(); // avoid double deletion
208 } else
209 this->Data = new T3[this->Count];
210 if (this->Data != NULL)
211 {
212 T3 *q = this->Data;
213 unsigned long i;
214 if ((slope == 1.0) && (intercept == 0.0))
215 {
216 if (!useInputBuffer)
217 {
218 DCMIMGLE_DEBUG("copying pixel data from input buffer");
219 const T1 *p = pixel + input->getPixelStart();
220 for (i = this->InputCount; i != 0; --i) // copy pixel data: can't use copyMem because T1 isn't always equal to T3
221 *(q++) = OFstatic_cast(T3, *(p++));
222 }
223 } else {
224 DCMIMGLE_DEBUG("applying modality transformation with rescale slope = " << slope << ", intercept = " << intercept);
225 T3 *lut = NULL;
226 const T1 *p = pixel + input->getPixelStart();
227 const unsigned long ocnt = OFstatic_cast(unsigned long, input->getAbsMaxRange()); // number of LUT entries
228 if (initOptimizationLUT(lut, ocnt))
229 { // use LUT for optimization
230 const double absmin = input->getAbsMinimum();
231 q = lut;
232 if (slope == 1.0)
233 {
234 for (i = 0; i < ocnt; ++i) // calculating LUT entries
235 *(q++) = OFstatic_cast(T3, OFstatic_cast(double, i) + absmin + intercept);
236 } else {
237 if (intercept == 0.0)
238 {
239 for (i = 0; i < ocnt; ++i)
240 *(q++) = OFstatic_cast(T3, (OFstatic_cast(double, i) + absmin) * slope);
241 } else {
242 for (i = 0; i < ocnt; ++i)
243 *(q++) = OFstatic_cast(T3, (OFstatic_cast(double, i) + absmin) * slope + intercept);
244 }
245 }
246 const T3 *lut0 = lut - OFstatic_cast(T2, absmin); // points to 'zero' entry
247 q = this->Data;
248 for (i = this->InputCount; i != 0; --i) // apply LUT
249 *(q++) = *(lut0 + (*(p++)));
250 }
251 if (lut == NULL) // use "normal" transformation
252 {
253 if (slope == 1.0)
254 {
255 for (i = this->InputCount; i != 0; --i)
256 *(q++) = OFstatic_cast(T3, OFstatic_cast(double, *(p++)) + intercept);
257 } else {
258 if (intercept == 0.0)
259 {
260 for (i = this->InputCount; i != 0; --i)
261 *(q++) = OFstatic_cast(T3, OFstatic_cast(double, *(p++)) * slope);
262 } else {
263 for (i = this->InputCount; i != 0; --i)
264 *(q++) = OFstatic_cast(T3, OFstatic_cast(double, *(p++)) * slope + intercept);
265 }
266 }
267 }
268 delete[] lut;
269 }
270 }
271 }
