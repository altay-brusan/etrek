146// Microsoft workaround: OFalign and OFalign_typename macros
147// __declspec(align) does not understand integral expressions
148// but instead requires an integral literal, Microsoft says:
149// "Valid entries are integer powers of two from 1 to 8192 (bytes),
150// such as 2, 4, 8, 16, 32, or 64."
151// So this is fundamentally different to the real alignment specifiers
152// from GNU and C++11.
153#define OFalign(T,A) OFdeclspec_align<T>::as<OFalignof_or_identity(A)>::type
154#define OFalign_typename(T,A) OFTypename OFdeclspec_align<T>::template as<OFalignof_or_identity(A)>::type
155
156// The trick / hack: specialize a template for every valid
157// integral expression and use an appropriate integral literal to
158// define the desired type. At least Microsoft allows us to
159// typedef this stuff, but the API is still fundamentally different
160// to the favored "OFalignas".
161// This also requires a specialization for arrays, since making
162// an array from an aligned type is not the same as making an
163// aligned array.
164#define DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( N )\
165template<typename T>\
166struct make_type<T,N> { typedef __declspec(align(N)) T type; };\
167template<typename T,size_t S>\
168struct make_type<T[S],N> { typedef __declspec(align(N)) T type[S]; }
169
170template<typename X>
171class OFdeclspec_align
172{
173 template<typename T,size_t>
174 struct make_type { typedef T type; };
175 template<typename T,size_t S,size_t A>
176 struct make_type<T[S],A> { typedef T type[S]; };
177 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 1 );
178 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 2 );
179 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 4 );
180 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 8 );
181 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 16 );
182 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 32 );
183 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 64 );
184 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 128 );
185 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 256 );
186 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 512 );
187 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 1024 );
188 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 2048 );
189 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 4096 );
190 DCMTK_OFALIGN_HACK_CONSTANT_BY_SPECIALIZATION( 8192 );
191
192public:
193 template<size_t A>
194 struct as { typedef OFTypename make_type<X,A>::type type; };
195};
