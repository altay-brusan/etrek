173 template <typename T>
174 int MY_Alltoallv64(const std::vector<T> &sendbuf, const std::vector<int64_t> &sendcounts,
175 const std::vector<int64_t> &senddisp, std::vector<T> &recvbuf,
176 const std::vector<int64_t> &recvcounts, const std::vector<int64_t> &recvdisp,
177 Ioss_MPI_Comm comm)
178 {
179 int processor_count = 0;
180 int my_processor = 0;
181 MPI_Comm_size(comm, &processor_count);
182 MPI_Comm_rank(comm, &my_processor);
183
184 // Verify that all 'counts' can fit in an integer. Symmetric
185 // communication, so recvcounts are sendcounts on another processor.
186 for (int i = 0; i < processor_count; i++) {
187 int snd_cnt = static_cast<int>(sendcounts[i]);
188 if (static_cast<int64_t>(snd_cnt) != sendcounts[i]) {
189 std::ostringstream errmsg;
190 errmsg << "ERROR: The number of items that must be communicated via MPI calls from\n"
191 << " processor " << my_processor << " to processor " << i << " is "
192 << sendcounts[i]
193 << "\n which exceeds the storage capacity of the integers "
194 "used by MPI functions.\n";
195 IOSS_ERROR(errmsg);
196 }
197 }
198
199 size_t pow_2 = Ioss::Utils::power_2(processor_count);
200
201 for (size_t i = 1; i < pow_2; i++) {
202 MPI_Status status{};
203
204 size_t exchange_proc = i ^ my_processor;
205 if (exchange_proc < static_cast<size_t>(processor_count)) {
206 int snd_cnt = static_cast<int>(
207 sendcounts[exchange_proc]); // Converts from int64_t to int as needed by mpi
208 int rcv_cnt = static_cast<int>(recvcounts[exchange_proc]);
209 int tag = 24713;
210 if (static_cast<size_t>(my_processor) < exchange_proc) {
211 MPI_Send((void *)&sendbuf[senddisp[exchange_proc]], snd_cnt, mpi_type(T(0)),
212 exchange_proc, tag, comm);
213 MPI_Recv(&recvbuf[recvdisp[exchange_proc]], rcv_cnt, mpi_type(T(0)), exchange_proc, tag,
214 comm, &status);
215 }
216 else {
217 MPI_Recv(&recvbuf[recvdisp[exchange_proc]], rcv_cnt, mpi_type(T(0)), exchange_proc, tag,
218 comm, &status);
219 MPI_Send((void *)&sendbuf[senddisp[exchange_proc]], snd_cnt, mpi_type(T(0)),
220 exchange_proc, tag, comm);
221 }
222 }
223 }
224
225 // Take care of this processor's data movement...
226 std::copy(&sendbuf[senddisp[my_processor]],
227 &sendbuf[senddisp[my_processor] + sendcounts[my_processor]],
228 &recvbuf[recvdisp[my_processor]]);
229 return 0;
230 }

232 template <typename T>
233 int MY_Alltoallv(const std::vector<T> &sendbuf, const std::vector<int64_t> &sendcnts,
234 const std::vector<int64_t> &senddisp, std::vector<T> &recvbuf,
235 const std::vector<int64_t> &recvcnts, const std::vector<int64_t> &recvdisp,
236 Ioss_MPI_Comm comm)
237 {
238// Wrapper to handle case where send/recv counts and displacements are 64-bit integers.
239// Two cases:
240// 1) They are of type 64-bit integers, but only storing data in the 32-bit integer range.
241// -- if (sendcnts[#proc-1] + senddisp[#proc-1] < 2^31, then we are ok
242// 2) They are of type 64-bit integers, and storing data in the 64-bit integer range.
243// -- call special alltoallv which does point-to-point sends
244#if IOSS_DEBUG_OUTPUT
245 {
246 Ioss::ParallelUtils utils(comm);
247 int processor_count = utils.parallel_size();
248
249 int max_comm = sendcnts[processor_count - 1] + senddisp[processor_count - 1];
250 std::vector<int> comm_size;
251
252 utils.gather(max_comm, comm_size);
253 int my_rank = utils.parallel_rank();
254 if (my_rank == 0) {
255 fmt::print("Send Communication Size: {}\n", fmt::join(comm_size, ", "));
256 }
257 }
258#endif
259#if 1
260 int processor_count = 0;
261 MPI_Comm_size(comm, &processor_count);
262 size_t max_comm = sendcnts[processor_count - 1] + senddisp[processor_count - 1];
263 size_t one = 1;
264 if (max_comm < one << 31) {
265 // count and displacement data in range, need to copy to integer vector.
266 std::vector<int> send_cnt(sendcnts.begin(), sendcnts.end());
267 std::vector<int> send_dis(senddisp.begin(), senddisp.end());
268 std::vector<int> recv_cnt(recvcnts.begin(), recvcnts.end());
269 std::vector<int> recv_dis(recvdisp.begin(), recvdisp.end());
270 return MPI_Alltoallv((void *)Data(sendbuf), Data(send_cnt), Data(send_dis), mpi_type(T(0)),
271 (void *)Data(recvbuf), Data(recv_cnt), Data(recv_dis), mpi_type(T(0)),
272 comm);
273 }
274 else {
275#endif
276 // Same as if each processor sent a message to every other process with:
277 // MPI_Send(sendbuf+senddisp[i]*sizeof(sendtype),sendcnts[i], sendtype, i, tag, comm);
278 // And received a message from each processor with a call to:
279 // MPI_Recv(recvbuf+recvdisp[i]*sizeof(recvtype),recvcnts[i], recvtype, i, tag, comm);
280 return MY_Alltoallv64(sendbuf, sendcnts, senddisp, recvbuf, recvcnts, recvdisp, comm);
281#if 1
282 }
283#endif
284 }

287 int MY_Alltoallv(const std::vector<T> &sendbuf, const std::vector<int> &sendcnts,
288 const std::vector<int> &senddisp, std::vector<T> &recvbuf,
289 const std::vector<int> &recvcnts, const std::vector<int> &recvdisp,
290 Ioss_MPI_Comm comm)

310 mpi_type(T(0)), comm);

314 template <typename T>
315 void ParallelUtils::global_array_minmax(IOSS_MAYBE_UNUSED std::vector<T> &local_minmax,
316 IOSS_MAYBE_UNUSED MinMax which) const
317 {
318 IOSS_PAR_UNUSED(local_minmax);
319 IOSS_PAR_UNUSED(which);
320#ifdef SEACAS_HAVE_MPI
321 if (parallel_size() > 1 && !local_minmax.empty()) {
322 if (Ioss::SerializeIO::isEnabled() && Ioss::SerializeIO::inBarrier()) {
323 std::ostringstream errmsg;
324 errmsg << "Attempting mpi while in barrier owned by " << Ioss::SerializeIO::getOwner();
325 IOSS_ERROR(errmsg);
326 }
327
328 std::vector<T> maxout(local_minmax.size());
329 MPI_Op oper = MPI_MAX;
330 if (which == Ioss::ParallelUtils::DO_MAX) {
331 oper = MPI_MAX;
332 }
333 else if (which == Ioss::ParallelUtils::DO_MIN) {
334 oper = MPI_MIN;
335 }
336 else if (which == Ioss::ParallelUtils::DO_SUM) {
337 oper = MPI_SUM;
338 }
339
340 const int success =
341 MPI_Allreduce((void *)(Data(local_minmax)), Data(maxout),
342 static_cast<int>(local_minmax.size()), mpi_type(T()), oper, communicator_);
343 if (success != MPI_SUCCESS) {
344 std::ostringstream errmsg;
345 errmsg << "Ioss::ParallelUtils::global_array_minmax - MPI_Allreduce failed";
346 IOSS_ERROR(errmsg);
347 }
348 // Now copy back into passed in array...
349 for (size_t i = 0; i < local_minmax.size(); i++) {
350 local_minmax[i] = maxout[i];
351 }
352 }
353#endif
354 }
