226struct MeasurementLine {
227 QPointF startPoint;
228 QPointF endPoint;
229 double distanceMm = 0.0;
230 double distancePixels = 0.0;
231 bool isComplete = false;
232 int id = -1;
233 bool isSelected = false;
234
238 void calculateDistance(double pixelSpacingX, double pixelSpacingY) {
239 double dx = (endPoint.x() - startPoint.x()) * pixelSpacingX;
240 double dy = (endPoint.y() - startPoint.y()) * pixelSpacingY;
241 distanceMm = std::sqrt(dx * dx + dy * dy);
242
243 double dxPx = endPoint.x() - startPoint.x();
244 double dyPx = endPoint.y() - startPoint.y();
245 distancePixels = std::sqrt(dxPx * dxPx + dyPx * dyPx);
246 }

253 bool isNearStartHandle(const QPointF& pos, double threshold = 10.0) const {
254 double dx = pos.x() - startPoint.x();
255 double dy = pos.y() - startPoint.y();
256 return std::sqrt(dx * dx + dy * dy) < threshold;
257 }
258
264 bool isNearEndHandle(const QPointF& pos, double threshold = 10.0) const {
265 double dx = pos.x() - endPoint.x();
266 double dy = pos.y() - endPoint.y();
267 return std::sqrt(dx * dx + dy * dy) < threshold;
268 }
269
275 bool isNearLine(const QPointF& pos, double threshold = 5.0) const {
276 // Calculate distance from point to line segment
277 double dx = endPoint.x() - startPoint.x();
278 double dy = endPoint.y() - startPoint.y();
279 double lengthSq = dx * dx + dy * dy;
280
281 if (lengthSq < 0.0001) {
282 // Start and end are the same point
283 double pdx = pos.x() - startPoint.x();
284 double pdy = pos.y() - startPoint.y();
285 return std::sqrt(pdx * pdx + pdy * pdy) < threshold;
286 }
287
288 // Project point onto line, clamped to segment
289 double t = ((pos.x() - startPoint.x()) * dx + (pos.y() - startPoint.y()) * dy) / lengthSq;
290 t = std::clamp(t, 0.0, 1.0);
291
292 // Find closest point on segment
293 double closestX = startPoint.x() + t * dx;
294 double closestY = startPoint.y() + t * dy;
295
296 // Calculate distance
297 double distX = pos.x() - closestX;
298 double distY = pos.y() - closestY;
299 return std::sqrt(distX * distX + distY * distY) < threshold;
300 }

337struct MeasurementAngle {
338 QPointF point1;
339 QPointF vertex;
340 QPointF point2;
341 double angleDegrees = 0.0;
342 bool isComplete = false;
343 int id = -1;
344 bool isSelected = false;
345
349 void calculateAngle() {
350 QPointF v1 = point1 - vertex;
351 QPointF v2 = point2 - vertex;
352
353 double dot = v1.x() * v2.x() + v1.y() * v2.y();
354 double mag1 = std::sqrt(v1.x() * v1.x() + v1.y() * v1.y());
355 double mag2 = std::sqrt(v2.x() * v2.x() + v2.y() * v2.y());
356
357 if (mag1 > 0 && mag2 > 0) {
358 double cosAngle = dot / (mag1 * mag2);
359 cosAngle = std::clamp(cosAngle, -1.0, 1.0);
360 angleDegrees = std::acos(cosAngle) * 180.0 / 3.14159265358979323846;
361 }
362 }

453inline ImageFormat detectFormat(const QString& filePath) {
454 QString lower = filePath.toLower();
455 if (lower.endsWith(".dcm") || lower.endsWith(".dicom")) {
456 return ImageFormat::DICOM;
457 } else if (lower.endsWith(".png")) {
458 return ImageFormat::PNG;
459 } else if (lower.endsWith(".jpg") || lower.endsWith(".jpeg")) {
460 return ImageFormat::JPEG;
461 } else if (lower.endsWith(".jp2") || lower.endsWith(".j2k") || lower.endsWith(".j2c")) {
462 return ImageFormat::JPEG2000;
463 } else if (lower.endsWith(".bmp")) {
464 return ImageFormat::BMP;
465 } else if (lower.endsWith(".tif") || lower.endsWith(".tiff")) {
466 return ImageFormat::TIFF;
467 }
468 // Check for DICOM files without extension (common in medical imaging)
469 return ImageFormat::UNKNOWN;
470}
