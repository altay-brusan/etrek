206 OFPair<iterator, bool> insert(const value_type& val)
207 {
208 // If value already exists, return it
209 OFListIterator(value_type) it = find(val.first);
210 if (it != end())
211 return OFMake_pair(it, false);
212
213 // Sorted insertion
214 it = begin();
215 while ( (it != end()) && (val.first > it->first) )
216 it++;
217 if (it == end())
218 it = values_.insert(values_.end(), val);
219 else
220 {
221 // Insert at current position and rewind iterator
222 // to the inserted element
223 values_.insert(it, val);
224 it--;
225 }
226
227 return OFMake_pair(it, true);
228 }
