142 void AddPoint(const T point[2])
143 {
144 // This code is written like this to ensure that adding a point gives
145 // exactly the same result as AddRect(vtkRect(x,y,0,0)
146 if (point[0] < this->GetX())
147 {
148 T dx = this->GetX() - point[0];
149 this->SetX(point[0]);
150 this->SetWidth(dx + this->GetWidth());
151 }
152 else if (point[0] > this->GetX())
153 {
154 // this->GetX() is already correct
155 T dx = point[0] - this->GetX();
156 this->SetWidth(vtkMath::Max(dx, this->GetWidth()));
157 }
159
160 if (point[1] < this->GetY())
161 {
162 T dy = this->GetY() - point[1];
163 this->SetY(point[1]);
164 this->SetHeight(dy + this->GetHeight());
165 }
166 else if (point[1] > this->GetY())
167 {
168 // this->GetY() is already correct
169 T dy = point[1] - this->GetY();

189 void AddRect(const vtkRect<T>& rect)
190 {
191 if (rect.GetX() < this->GetX())
192 {
193 T dx = this->GetX() - rect.GetX();
194 this->SetX(rect.GetX());
195 this->SetWidth(vtkMath::Max(dx + this->GetWidth(), rect.GetWidth()));
196 }
197 else if (rect.GetX() > this->GetX())
198 {
199 T dx = rect.GetX() - this->GetX();
200 // this->GetX() is already correct
201 this->SetWidth(vtkMath::Max(dx + rect.GetWidth(), this->GetWidth()));
202 }
203 else
204 {
205 // this->GetX() is already correct
206 this->SetWidth(vtkMath::Max(rect.GetWidth(), this->GetWidth()));
207 }
209
210 if (rect.GetY() < this->GetY())

235 bool IntersectsWith(const vtkRect<T>& rect) const
236 {
237 bool intersects = true;
238
239 if (rect.GetX() < this->GetX())
240 {
241 T dx = this->GetX() - rect.GetX();
242 intersects &= (dx < rect.GetWidth());
243 }
244 else if (rect.GetX() > this->GetX())
245 {
246 T dx = rect.GetX() - this->GetX();
247 intersects &= (dx < this->GetWidth());
248 }
249
250 if (rect.GetY() < this->GetY())
251 {
252 T dy = this->GetY() - rect.GetY();
253 intersects &= (dy < rect.GetHeight());
254 }
255 else if (rect.GetY() > this->GetY())
256 {
257 T dy = rect.GetY() - this->GetY();
258 intersects &= (dy < this->GetHeight());
259 }
260
261 return intersects;
262 }

282 bool Intersect(const vtkRect<T>& other)
283 {
284 if (this->IntersectsWith(other))
285 {
286 const T left = vtkMath::Max(this->GetLeft(), other.GetLeft());
287 const T bottom = vtkMath::Max(this->GetBottom(), other.GetBottom());
288 const T right = vtkMath::Min(this->GetRight(), other.GetRight());
289 const T top = vtkMath::Min(this->GetTop(), other.GetTop());
290
291 this->Data[0] = left;
292 this->Data[1] = bottom;
293 this->Data[2] = (right - left);
294 this->Data[3] = (top - bottom);
295 return true;
296 }
297 return false;
298 }
