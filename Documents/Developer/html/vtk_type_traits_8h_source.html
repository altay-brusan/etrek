22// Define a macro to simplify trait definitions.
23#define VTK_TYPE_TRAITS(type, macro, isSigned, name, print, format) \
24 template <> \
25 struct vtkTypeTraits<type> \
26 { \
27 /* The type itself. */ \
28 typedef type ValueType; \
29 \
30 /* the value defined for this type in vtkType */ \
31 enum \
32 { \
33 VTK_TYPE_ID = VTK_##macro \
34 }; \
35 static int VTKTypeID() \
36 { \
37 return VTK_##macro; \
38 } \
39 \
40 /* The smallest possible value represented by the type. */ \
41 static type Min() \
42 { \
43 return VTK_##macro##_MIN; \
44 } \
45 \
46 /* The largest possible value represented by the type. */ \
47 static type Max() \
48 { \
49 return VTK_##macro##_MAX; \
50 } \
51 \
52 /* Whether the type is signed. */ \
53 static int IsSigned() \
54 { \
55 return isSigned; \
56 } \
57 \
58 /* An "alias" type that is the same size and signedness. */ \
59 typedef vtkType##name SizedType; \
60 \
61 /* A name for the type indicating its size and signedness. */ \
62 static const char* SizedName() \
63 { \
64 return #name; \
65 } \
66 \
67 /* The common C++ name for the type (e.g. float, unsigned int, etc).*/ \
68 static const char* Name() \
69 { \
70 return #type; \
71 } \
72 \
73 /* A type to use for printing or parsing values in strings. */ \
74 typedef print PrintType; \
75 \
76 /* A format for parsing values from strings. Use with PrintType. */ \
77 static const char* ParseFormat() \
78 { \
79 return format; \
80 } \
81 }

83// Define traits for floating-point types.
84#define VTK_TYPE_NAME_FLOAT float
85#define VTK_TYPE_NAME_DOUBLE double
86#define VTK_TYPE_SIZED_FLOAT FLOAT32
87#define VTK_TYPE_SIZED_DOUBLE FLOAT64
88VTK_TYPE_TRAITS(float, FLOAT, 1, Float32, float, "%f");

91// Define traits for char types.
92// Note the print type is short because not all platforms support formatting integers with char.

105VTK_TYPE_TRAITS(signed char, SIGNED_CHAR, 1, Int8, short, "%hd");
106VTK_TYPE_TRAITS(unsigned char, UNSIGNED_CHAR, 0, UInt8, unsigned short, "%hu");

131VTK_TYPE_TRAITS(unsigned long, UNSIGNED_LONG, 0, UInt32, unsigned long, "%lu");
132#elif VTK_SIZEOF_LONG == 8
133#define VTK_TYPE_SIZED_LONG INT64
134#define VTK_TYPE_SIZED_UNSIGNED_LONG UINT64
135VTK_TYPE_TRAITS(long, LONG, 1, Int64, long, "%ld");
136VTK_TYPE_TRAITS(unsigned long, UNSIGNED_LONG, 0, UInt64, unsigned long, "%lu");
137#else
138#error "Type long is not 4 or 8 bytes in size."
