224// Generally the normal should be normalized
225inline double vtkPlane::Evaluate(double normal[3], double origin[3], double x[3])
226{
227 return normal[0] * (x[0] - origin[0]) + normal[1] * (x[1] - origin[1]) +
228 normal[2] * (x[2] - origin[2]);
229}
230
231// Assumes normal is normalized
232inline double vtkPlane::DistanceToPlane(double x[3], double n[3], double p0[3])
233{
234 return (std::abs(n[0] * (x[0] - p0[0]) + n[1] * (x[1] - p0[1]) + n[2] * (x[2] - p0[2])));
235}
