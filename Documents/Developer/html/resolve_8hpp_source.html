29void
30diy::
31update_links(diy::Master& master, const diy::DynamicAssigner& assigner)
32{
33 // figure out all the gids we need
34 std::vector<int> nbr_gids;
35 for (int i = 0; i < static_cast<int>(master.size()); ++i)
36 {
37 auto* link = master.link(i);
38 for (auto blockid : link->neighbors())
39 nbr_gids.emplace_back(blockid.gid);
40 }
41
42 // keep only unique gids to avoid unnecessary lookups
43 std::sort(nbr_gids.begin(), nbr_gids.end());
44 nbr_gids.resize(std::unique(nbr_gids.begin(), nbr_gids.end()) - nbr_gids.begin());
45
46 // get the neighbor ranks
47 std::vector<int> nbr_procs = assigner.ranks(nbr_gids);
48
49 // convert to a map
50 std::unordered_map<int, int> gid_to_proc;
51 for (size_t i = 0; i < nbr_gids.size(); ++i)
52 gid_to_proc[nbr_gids[i]] = nbr_procs[i];
53
54 // fix the procs in links
55 for (int i = 0; i < static_cast<int>(master.size()); ++i)
56 {
57 auto* link = master.link(i);
58 for (auto& blockid : link->neighbors())
59 blockid.proc = gid_to_proc[blockid.gid];
60 }
61}
