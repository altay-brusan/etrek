70/*
71 * One often needs to temporarily disable automatic error reporting when
72 * trying something that's likely or expected to fail. The code to try can
73 * be nested between calls to H5Eget_auto() and H5Eset_auto(), but it's
74 * easier just to use this macro like:
75 * H5E_BEGIN_TRY {
76 * ...stuff here that's likely to fail...
77 * } H5E_END_TRY;
78 *
79 * Warning: don't break, return, or longjmp() from the body of the loop or
80 * the error reporting won't be properly restored!
81 *
82 * These two macros still use the old API functions for backward compatibility
83 * purpose.
84 */
85#ifndef H5_NO_DEPRECATED_SYMBOLS
86#define H5E_BEGIN_TRY \
87 { \
88 unsigned H5E_saved_is_v2; \
89 union { \
90 H5E_auto1_t efunc1; \
91 H5E_auto2_t efunc2; \
92 } H5E_saved; \
93 void *H5E_saved_edata; \
94 \
95 (void)H5Eauto_is_v2(H5E_DEFAULT, &H5E_saved_is_v2); \
96 if (H5E_saved_is_v2) { \
97 (void)H5Eget_auto2(H5E_DEFAULT, &H5E_saved.efunc2, &H5E_saved_edata); \
98 (void)H5Eset_auto2(H5E_DEFAULT, NULL, NULL); \
99 } \
100 else { \
101 (void)H5Eget_auto1(&H5E_saved.efunc1, &H5E_saved_edata); \
102 (void)H5Eset_auto1(NULL, NULL); \
103 }
104
105#define H5E_END_TRY \
106 if (H5E_saved_is_v2) \
107 (void)H5Eset_auto2(H5E_DEFAULT, H5E_saved.efunc2, H5E_saved_edata); \
108 else \
109 (void)H5Eset_auto1(H5E_saved.efunc1, H5E_saved_edata); \
110 }
111#else /* H5_NO_DEPRECATED_SYMBOLS */
112#define H5E_BEGIN_TRY \
113 { \
114 H5E_auto2_t saved_efunc; \
115 void * H5E_saved_edata; \
116 \
117 (void)H5Eget_auto2(H5E_DEFAULT, &saved_efunc, &H5E_saved_edata); \
118 (void)H5Eset_auto2(H5E_DEFAULT, NULL, NULL);
119
120#define H5E_END_TRY \
121 (void)H5Eset_auto2(H5E_DEFAULT, saved_efunc, H5E_saved_edata); \
122 }
123#endif /* H5_NO_DEPRECATED_SYMBOLS */

128/* Use the Standard C __FILE__ & __LINE__ macros instead of typing them in */
129#define H5Epush_sim(func, cls, maj, min, str) \
130 H5Epush2(H5E_DEFAULT, __FILE__, func, __LINE__, cls, maj, min, str)

132/*
133 * Public API Convenience Macros for Error reporting - Undocumented
134 */
135/* Use the Standard C __FILE__ & __LINE__ macros instead of typing them in */
136/* And return after pushing error onto stack */
137#define H5Epush_ret(func, cls, maj, min, str, ret) \
138 do { \
139 H5Epush2(H5E_DEFAULT, __FILE__, func, __LINE__, cls, maj, min, str); \
140 return (ret); \
141 } while (0)
142
143/* Use the Standard C __FILE__ & __LINE__ macros instead of typing them in
144 * And goto a label after pushing error onto stack.
145 */
146#define H5Epush_goto(func, cls, maj, min, str, label) \
147 { \
148 H5Epush2(H5E_DEFAULT, __FILE__, func, __LINE__, cls, maj, min, str); \
149 goto label; \
150 }
