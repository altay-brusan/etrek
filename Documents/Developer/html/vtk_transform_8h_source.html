200 vtkLinearTransform* GetConcatenatedTransform(int i)
201 {
202 vtkAbstractTransform* t;
203 if (this->Input == nullptr)
204 {
205 t = this->Concatenation->GetTransform(i);
206 }
207 else if (i < this->Concatenation->GetNumberOfPreTransforms())
208 {
209 t = this->Concatenation->GetTransform(i);
210 }
211 else if (i > this->Concatenation->GetNumberOfPreTransforms())
212 {
213 t = this->Concatenation->GetTransform(i - 1);
214 }
215 else if (this->GetInverseFlag())
216 {
217 t = this->Input->GetInverse();
218 }
219 else
220 {
221 t = this->Input;
222 }
223 return static_cast<vtkLinearTransform*>(t);
224 }
