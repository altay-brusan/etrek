325inline OFBool
326DcmTagKey::isPrivateReservation() const
327{
328 // private reservation has element number ranging from 0x0010 to 0x00FF
329 return isPrivate() && (element >= 0x10) && (element <= 0xFF);
330}

332inline OFBool
333DcmTagKey::hasValidGroup() const
334{
335 // group numbers 1, 3, 5, 7 and 0xFFFF are illegal in DICOM
336 if (((group & 1) != 0) && ((group <= 7) || (group == 0xFFFF)))
337 return OFFalse;
338 else
339 return OFTrue;
340}
