518inline int DiOverlayPlane::reset(const unsigned long frame)
519{
520 int result = 0;
521 if (Valid && (Data != NULL))
522 {
523 /* check for multi-frame overlay */
524 const Uint32 frameNumber = (MultiframeOverlay) ? OFstatic_cast(Uint32, FirstFrame + frame) : 0;
525 DCMIMGLE_TRACE("reset overlay plane in group 0x" << STD_NAMESPACE hex << GroupNumber << " to start position");
526 /* special case: single frame overlay for multi-frame image */
527 if (!MultiframeOverlay && (frame > 0))
528 DCMIMGLE_TRACE(" using single frame overlay for multi-frame image (see CP-1974)");
529 DCMIMGLE_TRACE(" frameNumber: " << frameNumber << " (" << FirstFrame << "+" << frame
530 << "), ImageFrameOrigin: " << ImageFrameOrigin << ", NumberOfFrames: " << NumberOfFrames);
531 if ((frameNumber >= ImageFrameOrigin) && (frameNumber < ImageFrameOrigin + NumberOfFrames))
532 {
533 const unsigned long bits = (OFstatic_cast(unsigned long, StartLeft) + OFstatic_cast(unsigned long, StartTop) *
534 OFstatic_cast(unsigned long, Columns) + (frameNumber - ImageFrameOrigin) * OFstatic_cast(unsigned long, Rows) *
535 OFstatic_cast(unsigned long, Columns)) * OFstatic_cast(unsigned long, BitsAllocated);
536 StartBitPos = BitPos = OFstatic_cast(unsigned long, BitPosition) + bits;
537 DCMIMGLE_TRACE(" StartLeft: " << StartLeft << ", StartTop: " << StartTop << ", Columns: " << Columns << ", Rows: " << Rows);
538 DCMIMGLE_TRACE(" StartBitPos: " << StartBitPos << ", BitPosition: " << BitPosition << ", BitsAllocated: " << BitsAllocated << ", bits: " << bits);
539 /* distinguish between embedded and separate overlay data */
540 if (BitsAllocated == 16)
541 StartPtr = Ptr = Data + (bits >> 4);
542 else
543 StartPtr = Data;
544 result = (getRight() > 0) && (getBottom() > 0);
545 } else
546 DCMIMGLE_TRACE(" -> overlay plane does not apply to this frame");
547 }
548 return result;
549}

552inline int DiOverlayPlane::getNextBit()
553{
554 int result;
555 if (BitsAllocated == 16) // optimization
556 result = *(Ptr++) & (1 << BitPosition);
557 else
558 {
559 Ptr = StartPtr + (BitPos >> 4); // div 16
560 result = *Ptr & (1 << (BitPos & 0xf)); // mod 16
561 BitPos += BitsAllocated; // next bit
562 }
563 return result;
564}

567inline void DiOverlayPlane::setStart(const Uint16 x,
568 const Uint16 y,
569 const OFBool useOrigin)
570{
571 if (useOrigin)
572 {
573 if (BitsAllocated == 16)
574 Ptr = StartPtr + OFstatic_cast(unsigned long, y - Top) * OFstatic_cast(unsigned long, Columns) +
575 OFstatic_cast(unsigned long, x - Left);
576 else
577 BitPos = StartBitPos + (OFstatic_cast(unsigned long, y - Top) * OFstatic_cast(unsigned long, Columns) +
578 OFstatic_cast(unsigned long, x - Left)) * OFstatic_cast(unsigned long, BitsAllocated);
579 } else {
580 if (BitsAllocated == 16)
581 Ptr = StartPtr + OFstatic_cast(unsigned long, y) * OFstatic_cast(unsigned long, Columns) +
582 OFstatic_cast(unsigned long, x);
583 else
584 BitPos = StartBitPos + (OFstatic_cast(unsigned long, y) * OFstatic_cast(unsigned long, Columns) +
585 OFstatic_cast(unsigned long, x)) * OFstatic_cast(unsigned long, BitsAllocated);
586 }
