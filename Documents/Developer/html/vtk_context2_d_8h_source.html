568inline int vtkContext2D::FloatToInt(float x)
569{
570 // Use a tolerance of 1/256 of a pixel when converting.
571 // A float has only 24 bits of precision, so we cannot
572 // make the tolerance too small. For example, a tolerance
573 // of 2^-8 means that the tolerance will be significant
574 // for float values up to 2^16 or 65536.0. But a
575 // tolerance of 2^-16 would only be significant for
576 // float values up to 2^8 or 256.0. A small tolerance
577 // disappears into insignificance when added to a large float.
578 float tol = 0.00390625; // 1.0/256.0
579 tol = (x >= 0 ? tol : -tol);
580 return static_cast<int>(x + tol);
581}
