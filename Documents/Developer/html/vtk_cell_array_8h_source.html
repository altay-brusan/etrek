820 template <typename ArrayT>
821 struct VisitState
822 {
823 using ArrayType = ArrayT;
824 using ValueType = typename ArrayType::ValueType;
825 using CellRangeType = decltype(vtk::DataArrayValueRange<1>(std::declval<ArrayType>()));
826
827 // We can't just use is_same here, since binary compatible representations
828 // (e.g. int and long) are distinct types. Instead, ensure that ValueType
829 // is a signed integer the same size as vtkIdType.
830 // If this value is true, ValueType pointers may be safely converted to
831 // vtkIdType pointers via reinterpret cast.
832 static constexpr bool ValueTypeIsSameAsIdType = std::is_integral<ValueType>::value &&
833 std::is_signed<ValueType>::value && (sizeof(ValueType) == sizeof(vtkIdType));

1216 // Encapsulates storage of the internal arrays as a discriminated union
1217 // between 32-bit and 64-bit storage.
1218 struct Storage
1219 {
1220 // Union type that switches 32 and 64 bit array storage
1221 union ArraySwitch
1222 {
1223 ArraySwitch() = default; // handled by Storage
1224 ~ArraySwitch() = default; // handle by Storage
1225 VisitState<ArrayType32>* Int32;
1226 VisitState<ArrayType64>* Int64;
1227 };
1228
1229 Storage()
1230 {
1231#ifdef VTK_USE_MEMKIND
1232 this->Arrays =
1233 static_cast<ArraySwitch*>(vtkObjectBase::GetCurrentMallocFunction()(sizeof(ArraySwitch)));
1234#else
1235 this->Arrays = new ArraySwitch;
1236#endif
1237
1238 // Default can be changed, to save memory
1239 if (vtkCellArray::GetDefaultStorageIs64Bit())
1240 {
1241 this->Arrays->Int64 = new VisitState<ArrayType64>;
1242 this->StorageIs64Bit = true;
1243 }
1244 else
1245 {
1246 this->Arrays->Int32 = new VisitState<ArrayType32>;
1247 this->StorageIs64Bit = false;
1248 }
1249
1250#ifdef VTK_USE_MEMKIND
1251 if (vtkObjectBase::GetUsingMemkind())
1252 {
1253 this->IsInMemkind = true;
1254 }
1255#else
1256 (void)this->IsInMemkind; // comp warning workaround
1257#endif
1258 }
1259
1260 ~Storage()
1261 {
1262 if (this->StorageIs64Bit)
1263 {
1264 this->Arrays->Int64->~VisitState();
1265 delete this->Arrays->Int64;
1266 }
1267 else
1268 {
1269 this->Arrays->Int32->~VisitState();
1270 delete this->Arrays->Int32;
1271 }
1272#ifdef VTK_USE_MEMKIND
1273 if (this->IsInMemkind)
1274 {
1275 vtkObjectBase::GetAlternateFreeFunction()(this->Arrays);
1276 }
1277 else
1278 {
1279 free(this->Arrays);
1280 }
1281#else
1282 delete this->Arrays;
1283#endif
1284 }
1285
1286 // Switch the internal arrays to be 32-bit. Any old data is lost. Returns
1287 // true if the storage changes.
1288 bool Use32BitStorage()
1289 {
1290 if (!this->StorageIs64Bit)
1291 {
1292 return false;
1293 }
1294
1295 this->Arrays->Int64->~VisitState();
1296 delete this->Arrays->Int64;
1297 this->Arrays->Int32 = new VisitState<ArrayType32>;
1298 this->StorageIs64Bit = false;
1299
1300 return true;
1301 }
1302
1303 // Switch the internal arrays to be 64-bit. Any old data is lost. Returns
1304 // true if the storage changes.
1305 bool Use64BitStorage()
1306 {
1307 if (this->StorageIs64Bit)
1308 {
1309 return false;
1310 }
1311
1312 this->Arrays->Int32->~VisitState();
1313 delete this->Arrays->Int32;
1314 this->Arrays->Int64 = new VisitState<ArrayType64>;
1315 this->StorageIs64Bit = true;
1316
1317 return true;
1318 }
