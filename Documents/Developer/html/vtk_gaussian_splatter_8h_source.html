239 void SetScalar(vtkIdType idx, double dist2, double* sPtr)
240 {
241 double v =
242 (this->*SampleFactor)(this->S) * std::exp(this->ExponentFactor * (dist2) / (this->Radius2));
244
245 if (!this->Visited[idx])
246 {
247 this->Visited[idx] = 1;
248 *sPtr = v;
249 }
250 else
251 {
252 switch (this->AccumulationMode)
253 {
254 case VTK_ACCUMULATION_MODE_MIN:
255 if (*sPtr > v)
256 {
257 *sPtr = v;
258 }
259 break;
260 case VTK_ACCUMULATION_MODE_MAX:
261 if (*sPtr < v)
262 {
263 *sPtr = v;
264 }
265 break;
266 case VTK_ACCUMULATION_MODE_SUM:
267 *sPtr += v;
268 break;
269 }
270 } // not first visit
271 }
