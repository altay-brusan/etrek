280 static vtkIdType GetLinearIndex(const int i, const int j, const int k, const int N1, const int N2)
281 {
282 return ((static_cast<vtkIdType>(k) * N2 + j) * N1 + i);
283 }

292 static void GetStructuredCoordinates(
293 const vtkIdType idx, const int N1, const int N2, int& i, int& j, int& k)
294 {
295 vtkIdType N12 = N1 * N2;
296 k = static_cast<int>(idx / N12);
297 j = static_cast<int>((idx - k * N12) / N1);
298 i = static_cast<int>(idx - k * N12 - j * N1);
299 }

336inline vtkIdType vtkStructuredData::GetNumberOfPoints(const int ext[6], int)
337{
338 return static_cast<vtkIdType>(ext[1] - ext[0] + 1) * static_cast<vtkIdType>(ext[3] - ext[2] + 1) *
339 static_cast<vtkIdType>(ext[5] - ext[4] + 1);
340}

342//------------------------------------------------------------------------------
343inline vtkIdType vtkStructuredData::GetNumberOfCells(const int ext[6], int)
344{
345 int dims[3];
346 vtkStructuredData::GetDimensionsFromExtent(ext, dims);
347
348 // if any of the dimensions is 0, then there are no cells
349 const int cellDims[3] = { dims[0] != 0 ? vtkStructuredData::Max(dims[0] - 1, 1) : 0,
350 dims[1] != 0 ? vtkStructuredData::Max(dims[1] - 1, 1) : 0,
351 dims[2] != 0 ? vtkStructuredData::Max(dims[2] - 1, 1) : 0 };
352
353 // Note, when we compute the result below, we statically cast to vtkIdType to
354 // ensure the compiler will generate a 32x32=64 instruction.
355 return static_cast<vtkIdType>(cellDims[0]) * static_cast<vtkIdType>(cellDims[1]) *
356 static_cast<vtkIdType>(cellDims[2]);
357}

361 const int nodeExtent[6], int cellExtent[6], int)
362{
363 cellExtent[0] = nodeExtent[0];
364 cellExtent[2] = nodeExtent[2];
365 cellExtent[4] = nodeExtent[4];
366
367 cellExtent[1] = vtkStructuredData::Max(nodeExtent[0], nodeExtent[1] - 1);
368 cellExtent[3] = vtkStructuredData::Max(nodeExtent[2], nodeExtent[3] - 1);
369 cellExtent[5] = vtkStructuredData::Max(nodeExtent[4], nodeExtent[5] - 1);
