46// help function to set scaling values
47static inline void setScaleValues(Uint16 data[],
48 const Uint16 min,
49 const Uint16 max)
50{
51 Uint16 remainder = max % min;
52 Uint16 step0 = max / min;
53 Uint16 step1 = max / min;
54 if (remainder > OFstatic_cast(Uint16, min / 2))
55 {
56 remainder = min - remainder;
57 ++step0;
58 } else
59 ++step1;
60 const double count = OFstatic_cast(double, min) / (OFstatic_cast(double, remainder) + 1);
61 Uint16 i;
62 double c = count;
63 for (i = 0; i < min; ++i)
64 {
65 if ((i >= OFstatic_cast(Uint16, c)) && (remainder > 0))
66 {
67 --remainder;
68 c += count;
69 data[i] = step1;
70 }
71 else
72 data[i] = step0;
73 }
74}

76// cubic value interpolation using Catmull-Rom formula.
77// the interpolated pixel lies between the second and the third original pixels
78static inline double cubicValue(const double v1,
79 const double v2,
80 const double v3,
81 const double v4,
82 const double dD,
83 const double minVal,
84 const double maxVal)
85{
86 double dVal = 0.5 * ((((-v1 + 3 * v2 - 3 * v3 + v4) * dD + (2 * v1 - 5 * v2 + 4 * v3 - v4)) * dD + (-v1 + v3)) * dD + (v2 + v2));
87 return (dVal < minVal) ? minVal : ((dVal > maxVal) ? maxVal : dVal);
88}

187 void scaleData(const T *src[],
188 T *dest[],
189 const int interpolate,
190 const T value = 0)
191 {
192 if ((src != NULL) && (dest != NULL))
193 {
194 DCMIMGLE_TRACE("Col/Rows: " << Columns << " " << Rows << OFendl
195 << "Left/Top: " << Left << " " << Top << OFendl
196 << "Src X/Y: " << this->Src_X << " " << this->Src_Y << OFendl
197 << "Dest X/Y: " << this->Dest_X << " " << this->Dest_Y);
198 if ((Left + OFstatic_cast(signed long, this->Src_X) <= 0) || (Top + OFstatic_cast(signed long, this->Src_Y) <= 0) ||
199 (Left >= OFstatic_cast(signed long, Columns)) || (Top >= OFstatic_cast(signed long, Rows)))
200 { // no image to be displayed
201 DCMIMGLE_DEBUG("clipping area is fully outside the image boundaries");
202 this->fillPixel(dest, value); // ... fill bitmap
203 }
204 else if ((this->Src_X == this->Dest_X) && (this->Src_Y == this->Dest_Y)) // no scaling
205 {
206 if ((Left == 0) && (Top == 0) && (Columns == this->Src_X) && (Rows == this->Src_Y))
207 this->copyPixel(src, dest); // copying
208 else if ((Left >= 0) && (OFstatic_cast(Uint16, Left + this->Src_X) <= Columns) &&
209 (Top >= 0) && (OFstatic_cast(Uint16, Top + this->Src_Y) <= Rows))
210 clipPixel(src, dest); // clipping
211 else
212 clipBorderPixel(src, dest, value); // clipping (with border)
213 }
214 else if ((interpolate == 1) && (this->Bits <= MAX_INTERPOLATION_BITS))
215 interpolatePixel(src, dest); // interpolation (pbmplus)
216 else if ((interpolate == 4) && (this->Dest_X >= this->Src_X) && (this->Dest_Y >= this->Src_Y) &&
217 (this->Src_X >= 3) && (this->Src_Y >= 3))
218 bicubicPixel(src, dest); // bicubic magnification
219 else if ((interpolate >= 3) && (this->Dest_X >= this->Src_X) && (this->Dest_Y >= this->Src_Y) &&
220 (this->Src_X >= 2) && (this->Src_Y >= 2))
221 bilinearPixel(src, dest); // bilinear magnification
222 else if ((interpolate >= 1) && (this->Dest_X >= this->Src_X) && (this->Dest_Y >= this->Src_Y))
223 expandPixel(src, dest); // interpolated expansion (c't)
224 else if ((interpolate >= 1) && (this->Src_X >= this->Dest_X) && (this->Src_Y >= this->Dest_Y))
225 reducePixel(src, dest); // interpolated reduction (c't)
226 else if ((interpolate >= 1) && (this->Bits <= MAX_INTERPOLATION_BITS))
227 interpolatePixel(src, dest); // interpolation (pbmplus), fallback
228 else if ((this->Dest_X % this->Src_X == 0) && (this->Dest_Y % this->Src_Y == 0))
229 replicatePixel(src, dest); // replication
230 else if ((this->Src_X % this->Dest_X == 0) && (this->Src_Y % this->Dest_Y == 0))
231 suppressPixel(src, dest); // suppression
232 else
233 scalePixel(src, dest); // general scaling
234 }
235 }

257 void clipPixel(const T *src[],
258 T *dest[])
259 {
260 DCMIMGLE_DEBUG("using clip image to specified area algorithm");
261 const unsigned long x_feed = Columns - this->Src_X;
262 const unsigned long y_feed = OFstatic_cast(unsigned long, Rows - this->Src_Y) * OFstatic_cast(unsigned long, Columns);
263 Uint16 x;
264 Uint16 y;
265 const T *p;
266 T *q;
267 for (int j = 0; j < this->Planes; ++j)
268 {
269 p = src[j] + OFstatic_cast(unsigned long, Top) * OFstatic_cast(unsigned long, Columns) + Left;
270 q = dest[j];
271 for (unsigned long f = this->Frames; f != 0; --f)
272 {
273 for (y = this->Dest_Y; y != 0; --y)
274 {
275 for (x = this->Dest_X; x != 0; --x)
276 *(q++) = *(p++);
277 p += x_feed;
278 }
279 p += y_feed;
280 }
281 }

290 void clipBorderPixel(const T *src[],
291 T *dest[],
292 const T value)
293 {
294 DCMIMGLE_DEBUG("using clip image to specified area and add border algorithm");
295 const Uint16 s_left = (Left > 0) ? OFstatic_cast(Uint16, Left) : 0;
296 const Uint16 s_top = (Top > 0) ? OFstatic_cast(Uint16, Top) : 0;
297 const Uint16 d_left = (Left < 0 ? OFstatic_cast(Uint16, -Left) : 0);
298 const Uint16 d_top = (Top < 0) ? OFstatic_cast(Uint16, -Top) : 0;
299 const Uint16 d_right = (OFstatic_cast(unsigned long, this->Src_X) + OFstatic_cast(unsigned long, s_left) <
300 OFstatic_cast(unsigned long, Columns) + OFstatic_cast(unsigned long, d_left)) ?
301 (this->Src_X - 1) : (Columns + d_left - s_left - 1);
302 const Uint16 d_bottom = (OFstatic_cast(unsigned long, this->Src_Y) + OFstatic_cast(unsigned long, s_top) <
303 OFstatic_cast(unsigned long, Rows) + OFstatic_cast(unsigned long, d_top)) ?
304 (this->Src_Y - 1) : (Rows + d_top - s_top - 1);
305 const Uint16 x_count = d_right - d_left + 1;
306 const Uint16 y_count = d_bottom - d_top + 1;
307 const unsigned long s_start = OFstatic_cast(unsigned long, s_top) * OFstatic_cast(unsigned long, Columns) + s_left;
308 const unsigned long x_feed = Columns - x_count;
309 const unsigned long y_feed = OFstatic_cast(unsigned long, Rows - y_count) * Columns;
310 const unsigned long t_feed = OFstatic_cast(unsigned long, d_top) * OFstatic_cast(unsigned long, this->Src_X);
311 const unsigned long b_feed = OFstatic_cast(unsigned long, this->Src_Y - d_bottom - 1) * OFstatic_cast(unsigned long, this->Src_X);
312
313 /*
314 * The approach is to divide the destination image in up to four areas outside the source image
315 * plus one area for the source image. The for and while loops are scanning linearly over the
316 * destination image and setting the appropriate value depending on the current area. This is
317 * different from most of the other algorithms in this toolkit where the source image is scanned
318 * linearly.
319 */
320 Uint16 x;
321 Uint16 y;
322 unsigned long i;
323 const T *p;
324 T *q;
325 for (int j = 0; j < this->Planes; ++j)
326 {
327 p = src[j] + s_start;
328 q = dest[j];
329 for (unsigned long f = this->Frames; f != 0; --f)
330 {
331 for (i = t_feed; i != 0; --i) // top
332 *(q++) = value;
333 for (y = y_count; y != 0; --y) // middle part:
334 {
335 x = 0;
336 while (x < d_left) // - left
337 {
338 *(q++) = value;
339 ++x;
340 }
341 while (x <= d_right) // - middle
342 {
343 *(q++) = *(p++);
344 ++x;
345 }
346 while (x < this->Src_X) // - right
347 {
348 *(q++) = value;
349 ++x;
350 }
351 p += x_feed;
352 }
353 for (i = b_feed; i != 0; --i) // bottom
354 *(q++) = value;
355 p += y_feed;
356 }
357 }
358 }

366 void replicatePixel(const T *src[],
367 T *dest[])
368 {
369 DCMIMGLE_DEBUG("using replicate pixel scaling algorithm without interpolation");
370 const Uint16 x_factor = this->Dest_X / this->Src_X;
371 const Uint16 y_factor = this->Dest_Y / this->Src_Y;
372 const unsigned long x_feed = Columns;
373 const unsigned long y_feed = OFstatic_cast(unsigned long, Rows - this->Src_Y) * OFstatic_cast(unsigned long, Columns);
374 const T *sp;
375 Uint16 x;
376 Uint16 y;
377 Uint16 dx;
378 Uint16 dy;
379 const T *p;
380 T *q;
381 T value;
382 for (int j = 0; j < this->Planes; ++j)
383 {
384 sp = src[j] + OFstatic_cast(unsigned long, Top) * OFstatic_cast(unsigned long, Columns) + Left;
385 q = dest[j];
386 for (unsigned long f = this->Frames; f != 0; --f)
387 {
388 for (y = this->Src_Y; y != 0; --y)
389 {
390 for (dy = y_factor; dy != 0; --dy)
391 {
392 for (x = this->Src_X, p = sp; x != 0; --x)
393 {
394 value = *(p++);
395 for (dx = x_factor; dx != 0; --dx)
396 *(q++) = value;
397 }
398 }
399 sp += x_feed;
400 }
401 sp += y_feed;
402 }
403 }

449 void scalePixel(const T *src[],
450 T *dest[])
451 {
452 DCMIMGLE_DEBUG("using free scaling algorithm without interpolation");
453 const Uint16 xmin = (this->Dest_X < this->Src_X) ? this->Dest_X : this->Src_X; // minimum width
454 const Uint16 ymin = (this->Dest_Y < this->Src_Y) ? this->Dest_Y : this->Src_Y; // minimum height
455 Uint16 *x_step = new Uint16[xmin];
456 Uint16 *y_step = new Uint16[ymin];
457 Uint16 *x_fact = new Uint16[xmin];
458 Uint16 *y_fact = new Uint16[ymin];
459
460 /*
461 * Approach: If one pixel line has to be added or removed it is taken from the middle of the image (1/2).
462 * For two lines it is at 1/3 and 2/3 of the image and so on. It sounds easy but it was a hard job ;-)
463 */
464
465 if ((x_step != NULL) && (y_step != NULL) && (x_fact != NULL) && (y_fact != NULL))
466 {
467 Uint16 x;
468 Uint16 y;
469 if (this->Dest_X < this->Src_X)
470 setScaleValues(x_step, this->Dest_X, this->Src_X);
471 else if (this->Dest_X > this->Src_X)
472 setScaleValues(x_fact, this->Src_X, this->Dest_X);
473 if (this->Dest_X <= this->Src_X)
474 OFBitmanipTemplate<Uint16>::setMem(x_fact, 1, xmin); // initialize with default values
475 if (this->Dest_X >= this->Src_X)
476 OFBitmanipTemplate<Uint16>::setMem(x_step, 1, xmin); // initialize with default values
477 x_step[xmin - 1] += Columns - this->Src_X; // skip to next line
478 if (this->Dest_Y < this->Src_Y)
479 setScaleValues(y_step, this->Dest_Y, this->Src_Y);
480 else if (this->Dest_Y > this->Src_Y)
481 setScaleValues(y_fact, this->Src_Y, this->Dest_Y);
482 if (this->Dest_Y <= this->Src_Y)
483 OFBitmanipTemplate<Uint16>::setMem(y_fact, 1, ymin); // initialize with default values
484 if (this->Dest_Y >= this->Src_Y)
485 OFBitmanipTemplate<Uint16>::setMem(y_step, 1, ymin); // initialize with default values
486 y_step[ymin - 1] += Rows - this->Src_Y; // skip to next frame
487 const T *sp;
488 Uint16 dx;
489 Uint16 dy;
490 const T *p;
491 T *q;
492 T value;
493 for (int j = 0; j < this->Planes; ++j)
494 {
495 sp = src[j] + OFstatic_cast(unsigned long, Top) * OFstatic_cast(unsigned long, Columns) + Left;
496 q = dest[j];
497 for (unsigned long f = 0; f < this->Frames; ++f)
498 {
499 for (y = 0; y < ymin; ++y)
500 {
501 for (dy = 0; dy < y_fact[y]; ++dy)
502 {
503 for (x = 0, p = sp; x < xmin; ++x)
504 {
505 value = *p;
506 for (dx = 0; dx < x_fact[x]; ++dx)
507 *(q++) = value;
508 p += x_step[x];
509 }
510 }
511 sp += OFstatic_cast(unsigned long, y_step[y]) * OFstatic_cast(unsigned long, Columns);
512 }
513 }
514 }
515 }
516 delete[] x_step;
517 delete[] y_step;
518 delete[] x_fact;
519 delete[] y_fact;
520 }

528 void interpolatePixel(const T *src[],
529 T *dest[])
530 {
531 DCMIMGLE_DEBUG("using scaling algorithm with interpolation from pbmplus toolkit");
532 if ((this->Src_X != Columns) || (this->Src_Y != Rows))
533 {
534 DCMIMGLE_ERROR("interpolated scaling and clipping at the same time not implemented ... ignoring clipping region");
535 this->Src_X = Columns; // temporarily removed 'const' for 'Src_X' in class 'DiTransTemplate'
536 this->Src_Y = Rows; // ... 'Src_Y' ...
537 }
538
539 /*
540 * based on scaling algorithm from "Extended Portable Bitmap Toolkit" (pbmplus10dec91)
541 * (adapted to be used with signed pixel representation, inverse images - mono1,
542 * various bit depths, multi-frame and multi-plane/color images)
543 */
544
545 Uint16 x;
546 Uint16 y;
547 const T *p;
548 T *q;
549 const T *sp = NULL; // initialization avoids compiler warning
550 const T *fp;
551 T *sq;
552
553 const unsigned long sxscale = OFstatic_cast(unsigned long, (OFstatic_cast(double, this->Dest_X) / OFstatic_cast(double, this->Src_X)) * SCALE_FACTOR);
554 const unsigned long syscale = OFstatic_cast(unsigned long, (OFstatic_cast(double, this->Dest_Y) / OFstatic_cast(double, this->Src_Y)) * SCALE_FACTOR);
555 const signed long maxvalue = DicomImageClass::maxval(this->Bits - isSigned());
556
557 T *xtemp = new T[this->Src_X];
558 signed long *xvalue = new signed long[this->Src_X];
559
560 if ((xtemp == NULL) || (xvalue == NULL))
561 {
562 DCMIMGLE_ERROR("can't allocate temporary buffers for interpolation scaling");
563 this->clearPixel(dest);
564 } else {
565 for (int j = 0; j < this->Planes; ++j)
566 {
567 fp = src[j];
568 sq = dest[j];
569 for (unsigned long f = this->Frames; f != 0; --f)
570 {
571 for (x = 0; x < this->Src_X; ++x)
572 xvalue[x] = HALFSCALE_FACTOR;
573 unsigned long yfill = SCALE_FACTOR;
574 unsigned long yleft = syscale;
575 int yneed = 1;
576 int ysrc = 0;
577 for (y = 0; y < this->Dest_Y; ++y)
578 {
579 if (this->Src_Y == this->Dest_Y)
580 {
581 sp = fp;
582 for (x = this->Src_X, p = sp, q = xtemp; x != 0; --x)
583 *(q++) = *(p++);
584 fp += this->Src_X;
585 }
586 else
587 {
588 while (yleft < yfill)
589 {
590 if (yneed && (ysrc < OFstatic_cast(int, this->Src_Y)))
591 {
592 sp = fp;
593 fp += this->Src_X;
594 ++ysrc;
595 }
596 for (x = 0, p = sp; x < this->Src_X; ++x)
597 xvalue[x] += yleft * OFstatic_cast(signed long, *(p++));
598 yfill -= yleft;
599 yleft = syscale;
600 yneed = 1;
601 }
602 if (yneed && (ysrc < OFstatic_cast(int, this->Src_Y)))
603 {
604 sp = fp;
605 fp += this->Src_X;
606 ++ysrc;
607 yneed = 0;
608 }
609 signed long v;
610 for (x = 0, p = sp, q = xtemp; x < this->Src_X; ++x)
611 {
612 v = xvalue[x] + yfill * OFstatic_cast(signed long, *(p++));
613 v /= SCALE_FACTOR;
614 *(q++) = OFstatic_cast(T, (v > maxvalue) ? maxvalue : v);
615 xvalue[x] = HALFSCALE_FACTOR;
616 }
617 yleft -= yfill;
618 if (yleft == 0)
619 {
620 yleft = syscale;
621 yneed = 1;
622 }
623 yfill = SCALE_FACTOR;
624 }
625 if (this->Src_X == this->Dest_X)
626 {
627 for (x = this->Dest_X, p = xtemp, q = sq; x != 0; --x)
628 *(q++) = *(p++);
629 sq += this->Dest_X;
630 }
631 else
632 {
633 signed long v = HALFSCALE_FACTOR;
634 unsigned long xfill = SCALE_FACTOR;
635 unsigned long xleft;
636 int xneed = 0;
637 q = sq;
638 for (x = 0, p = xtemp; x < this->Src_X; ++x, ++p)
639 {
640 xleft = sxscale;
641 while (xleft >= xfill)
642 {
643 if (xneed)
644 {
645 ++q;
646 v = HALFSCALE_FACTOR;
647 }
648 v += xfill * OFstatic_cast(signed long, *p);
649 v /= SCALE_FACTOR;
650 *q = OFstatic_cast(T, (v > maxvalue) ? maxvalue : v);
651 xleft -= xfill;
652 xfill = SCALE_FACTOR;
653 xneed = 1;
654 }
655 if (xleft > 0)
656 {
657 if (xneed)
658 {
659 ++q;
660 v = HALFSCALE_FACTOR;
661 xneed = 0;
662 }
663 v += xleft * OFstatic_cast(signed long, *p);
664 xfill -= xleft;
665 }
666 }
667 if (xfill > 0)
668 v += xfill * OFstatic_cast(signed long, *(--p));
669 if (!xneed)
670 {
671 v /= SCALE_FACTOR;
672 *q = OFstatic_cast(T, (v > maxvalue) ? maxvalue : v);
673 }
674 sq += this->Dest_X;
675 }
676 }
677 }
678 }
679 }
680 delete[] xtemp;
681 delete[] xvalue;
682 }

690 void expandPixel(const T *src[],
691 T *dest[])
692 {
693 DCMIMGLE_DEBUG("using expand pixel scaling algorithm with interpolation from c't magazine");
694 const double x_factor = OFstatic_cast(double, this->Src_X) / OFstatic_cast(double, this->Dest_X);
695 const double y_factor = OFstatic_cast(double, this->Src_Y) / OFstatic_cast(double, this->Dest_Y);
696 const unsigned long f_size = OFstatic_cast(unsigned long, Rows) * OFstatic_cast(unsigned long, Columns);
697 const T *sp;
698 double bx, ex;
699 double by, ey;
700 int bxi, exi;
701 int byi, eyi;
702 unsigned long offset;
703 double value, sum;
704 double x_part, y_part;
705 double l_factor, r_factor;
706 double t_factor, b_factor;
707 int xi;
708 int yi;
709 Uint16 x;
710 Uint16 y;
711 const T *p;
712 T *q;
713
714 /*
715 * based on scaling algorithm from "c't - Magazin fuer Computertechnik" (c't 11/94)
716 * (adapted to be used with signed pixel representation, inverse images - mono1,
717 * various bit depths, multi-frame and multi-plane/color images, combined clipping/scaling)
718 */
719
720 for (int j = 0; j < this->Planes; ++j)
721 {
722 sp = src[j] + OFstatic_cast(unsigned long, Top) * OFstatic_cast(unsigned long, Columns) + Left;
723 q = dest[j];
724 for (unsigned long f = 0; f < this->Frames; ++f)
725 {
726 for (y = 0; y < this->Dest_Y; ++y)
727 {
728 by = y_factor * OFstatic_cast(double, y);
729 ey = y_factor * (OFstatic_cast(double, y) + 1.0);
730 if (ey > this->Src_Y)
731 {
732#ifdef DEBUG // this output is only useful for debugging purposes
733 DCMIMGLE_TRACE(" limiting value of 'ey' to 'Src_Y': " << ey << " -> " << this->Src_Y);
734#endif
735 // see reducePixel()
736 ey = this->Src_Y;
737 }
738 byi = OFstatic_cast(int, by);
739 eyi = OFstatic_cast(int, ey);
740 if (OFstatic_cast(double, eyi) == ey)
741 {
742#ifdef DEBUG // this output is only useful for debugging purposes

744#endif

746 }
747 y_part = OFstatic_cast(double, eyi) / y_factor;

749 t_factor = (OFstatic_cast(double, y) + 1.0) - y_part;

751 {
752 value = 0;
753 bx = x_factor * OFstatic_cast(double, x);

755 if (ex > this->Src_X)

759#endif

761 ex = this->Src_X;
762 }

764 exi = OFstatic_cast(int, ex);

766 {

768 DCMIMGLE_TRACE(" decreasing value of 'exi' by 1: " << exi << " -> " << (exi - 1));
769#endif
770 --exi;
771 }

773 l_factor = x_part - OFstatic_cast(double, x);

776 for (yi = byi; yi <= eyi; ++yi)
777 {
778 p = sp + offset + bxi;
779 for (xi = bxi; xi <= exi; ++xi)
780 {

782 if (bxi != exi)
783 {

785 sum *= l_factor;
786 else
787 sum *= r_factor;
788 }
789 if (byi != eyi)
790 {
791 if (yi == byi)
792 sum *= b_factor;
793 else
794 sum *= t_factor;
795 }
796 value += sum;
797 }
798 offset += Columns;
799 }
800 *(q++) = OFstatic_cast(T, value + 0.5);
801 }
802 }
803 sp += f_size;
804 }

814 void reducePixel(const T *src[],
815 T *dest[])
816 {
817 DCMIMGLE_DEBUG("using reduce pixel scaling algorithm with interpolation from c't magazine");
818 const double x_factor = OFstatic_cast(double, this->Src_X) / OFstatic_cast(double, this->Dest_X);
819 const double y_factor = OFstatic_cast(double, this->Src_Y) / OFstatic_cast(double, this->Dest_Y);
820 const double xy_factor = x_factor * y_factor;
821 const unsigned long f_size = OFstatic_cast(unsigned long, Rows) * OFstatic_cast(unsigned long, Columns);
822 const T *sp;
823 double bx, ex;
824 double by, ey;
825 int bxi, exi;
826 int byi, eyi;
827 unsigned long offset;
828 double value, sum;
829 double l_factor, r_factor;
830 double t_factor, b_factor;
831 int xi;
832 int yi;
833 Uint16 x;
834 Uint16 y;
835 const T *p;
836 T *q;
837
838 /*
839 * based on scaling algorithm from "c't - Magazin fuer Computertechnik" (c't 11/94)
840 * (adapted to be used with signed pixel representation, inverse images - mono1,
841 * various bit depths, multi-frame and multi-plane/color images, combined clipping/scaling)
842 */
843
844 for (int j = 0; j < this->Planes; ++j)
845 {
846 sp = src[j] + OFstatic_cast(unsigned long, Top) * OFstatic_cast(unsigned long, Columns) + Left;
847 q = dest[j];
848 for (unsigned long f = 0; f < this->Frames; ++f)
849 {
850 for (y = 0; y < this->Dest_Y; ++y)
851 {
852 by = y_factor * OFstatic_cast(double, y);
853 ey = y_factor * (OFstatic_cast(double, y) + 1.0);
854 if (ey > this->Src_Y)
855 {
856#ifdef DEBUG // this output is only useful for debugging purposes
857 DCMIMGLE_TRACE(" limiting value of 'ey' to 'Src_Y': " << ey << " -> " << this->Src_Y);
858#endif
859 // yes, this can happen due to rounding, e.g. double(943) / double(471) * double(471)
860 // is something like 943.00000000000011368683772161602974 and then, the eyi == ey check
861 // fails to bring eyi back into range!
862 ey = this->Src_Y;
863 }
864 byi = OFstatic_cast(int, by);
865 eyi = OFstatic_cast(int, ey);
866 if (OFstatic_cast(double, eyi) == ey)
867 {
868#ifdef DEBUG // this output is only useful for debugging purposes
869 DCMIMGLE_TRACE(" decreasing value of 'eyi' by 1: " << eyi << " -> " << (eyi - 1));
870#endif
871 --eyi;
872 }
873 b_factor = 1 + OFstatic_cast(double, byi) - by;
874 t_factor = ey - OFstatic_cast(double, eyi);
875 for (x = 0; x < this->Dest_X; ++x)
876 {
877 value = 0;
878 bx = x_factor * OFstatic_cast(double, x);
879 ex = x_factor * (OFstatic_cast(double, x) + 1.0);
880 if (ex > this->Src_X)
881 {
882#ifdef DEBUG // this output is only useful for debugging purposes
883 DCMIMGLE_TRACE(" limiting value of 'ex' to 'Src_X': " << ex << " -> " << this->Src_X);
884#endif
885 // see above comment
886 ex = this->Src_X;
887 }
888 bxi = OFstatic_cast(int, bx);
889 exi = OFstatic_cast(int, ex);
890 if (OFstatic_cast(double, exi) == ex)
891 {
892#ifdef DEBUG // this output is only useful for debugging purposes
893 DCMIMGLE_TRACE(" decreasing value of 'exi' by 1: " << exi << " -> " << (exi - 1));
894#endif
895 --exi;
896 }
897 l_factor = 1 + OFstatic_cast(double, bxi) - bx;
898 r_factor = ex - OFstatic_cast(double, exi);
899 offset = OFstatic_cast(unsigned long, byi) * OFstatic_cast(unsigned long, Columns);
900 for (yi = byi; yi <= eyi; ++yi)
901 {
902 p = sp + offset + bxi;
903 for (xi = bxi; xi <= exi; ++xi)
904 {
905 sum = OFstatic_cast(double, *(p++)) / xy_factor;
906 if (xi == bxi)
907 sum *= l_factor;
908 else if (xi == exi)
909 sum *= r_factor;
910 if (yi == byi)
911 sum *= b_factor;
912 else if (yi == eyi)
913 sum *= t_factor;
914 value += sum;
915 }
916 offset += Columns;
917 }
918 *(q++) = OFstatic_cast(T, value + 0.5);
919 }
920 }
921 sp += f_size;
922 }
923 }

931 void bilinearPixel(const T *src[],
932 T *dest[])
933 {
934 DCMIMGLE_DEBUG("using magnification algorithm with bilinear interpolation contributed by Eduard Stanescu");
935 const double x_factor = OFstatic_cast(double, this->Src_X) / OFstatic_cast(double, this->Dest_X);
936 const double y_factor = OFstatic_cast(double, this->Src_Y) / OFstatic_cast(double, this->Dest_Y);
937 const unsigned long f_size = OFstatic_cast(unsigned long, Rows) * OFstatic_cast(unsigned long, Columns);
938 const unsigned long l_offset = OFstatic_cast(unsigned long, this->Src_Y - 1) * OFstatic_cast(unsigned long, this->Dest_X);
939 Uint16 x;
940 Uint16 y;
941 T *pD;
942 T *pCurrTemp;
943 const T *pCurrSrc;
944 Uint16 nSrcIndex;
945 double dOff;
946 T *pT;
947 const T *pS;
948 const T *pF;
949
950 // buffer used for storing temporarily the interpolated lines
951 T *pTemp = new T[OFstatic_cast(unsigned long, this->Src_Y) * OFstatic_cast(unsigned long, this->Dest_X)];
952 if (pTemp == NULL)
953 {
954 DCMIMGLE_ERROR("can't allocate temporary buffer for interpolation scaling");
955 this->clearPixel(dest);
956 } else {
957
958 /*
959 * based on scaling algorithm contributed by Eduard Stanescu
960 * (adapted to be used with signed pixel representation, inverse images - mono1,
961 * various bit depths, multi-frame multi-plane/color images, combined clipping/scaling)
962 */
963
964 for (int j = 0; j < this->Planes; ++j)
965 {
966 pF = src[j] + OFstatic_cast(unsigned long, Top) * OFstatic_cast(unsigned long, Columns) + Left;
967 pD = dest[j];
968 for (unsigned long f = this->Frames; f != 0; --f)
969 {
970 pT = pCurrTemp = pTemp;
971 pS = pCurrSrc = pF;
972 // first, interpolate the columns:
973 // column 0, just copy the source data column 0
974 for (y = this->Src_Y; y != 0; --y)
975 {
976 *(pCurrTemp) = *(pCurrSrc);
977 pCurrSrc += Columns;
978 pCurrTemp += this->Dest_X;
979 }
980 pCurrSrc = pS;
981 nSrcIndex = 0;
982 // column 1 to column Dest_X - 1
983 for (x = 1; x < this->Dest_X - 1; ++x)
984 {
985 pCurrTemp = ++pT;
986 dOff = x * x_factor - nSrcIndex;
987 dOff = (1.0 < dOff) ? 1.0 : dOff;
988 for (y = 0; y < this->Src_Y; ++y)
989 {
990 // use floating points in order to avoid possible integer overflow
991 const double v1 = OFstatic_cast(double, *(pCurrSrc));
992 const double v2 = OFstatic_cast(double, *(pCurrSrc + 1));
993 *(pCurrTemp) = OFstatic_cast(T, v1 + (v2 - v1) * dOff);
994 pCurrSrc += Columns;
995 pCurrTemp += this->Dest_X;
996 }
997 // don't go beyond the source data
998 if ((nSrcIndex < this->Src_X - 2) && (x * x_factor >= nSrcIndex + 1))
999 {
1000 pS++;
1001 nSrcIndex++;
1002 }
1003 pCurrSrc = pS;
1004 }
1005 pCurrTemp = ++pT;
1006 // last column, just copy the source data column Src_X
1007 for (y = this->Src_Y; y != 0; --y)
1008 {
1009 *(pCurrTemp) = *(pCurrSrc);
1010 pCurrSrc += Columns;
1011 pCurrTemp += this->Dest_X;
1012 }
1013 // now the columns are interpolated in temp buffer, so interpolate the lines
1014 pT = pCurrTemp = pTemp;
1015 // line 0, just copy the temp buffer line 0
1016 for (x = this->Dest_X; x != 0; --x)
1017 *(pD++) = *(pCurrTemp++);
1018 nSrcIndex = 0;
1019 pCurrTemp = pTemp;
1020 for (y = 1; y < this->Dest_Y - 1; ++y)
1021 {
1022 dOff = y * y_factor - nSrcIndex;
1023 dOff = (1.0 < dOff) ? 1.0 : dOff;
1024 for (x = this->Dest_X; x != 0; --x)
1025 {
1026 // use floating points in order to avoid possible integer overflow
1027 const double v1 = OFstatic_cast(double, *(pCurrTemp));
1028 const double v2 = OFstatic_cast(double, *(pCurrTemp + this->Dest_X));
1029 *(pD++) = OFstatic_cast(T, v1 + (v2 - v1) * dOff);
1030 pCurrTemp++;
1031 }
1032 // don't go beyond the source data
1033 if ((nSrcIndex < this->Src_Y - 2) && (y * y_factor >= nSrcIndex + 1))
1034 {
1035 pT += this->Dest_X;
1036 nSrcIndex++;
1037 }
1038 pCurrTemp = pT;
1039 }
1040 // the last line, just copy the temp buffer line Src_X
1041 pCurrTemp = pTemp + l_offset;
1042 for (x = this->Dest_X; x != 0; --x)
1043 *(pD++) = *(pCurrTemp++);
1044 // skip to next frame
1045 pF += f_size;
1046 }
1047 }
1048 }
1049 delete[] pTemp;
1050 }
