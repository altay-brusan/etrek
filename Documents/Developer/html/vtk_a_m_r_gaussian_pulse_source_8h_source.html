140 double ComputePulseAt(const double x, const double y, const double z)
141 {
142 double xyz[3];
143 xyz[0] = x;
144 xyz[1] = y;
145 xyz[2] = z;
146 return (this->ComputePulseAt(xyz));
147 }
148 double ComputePulseAt(double pt[3])
149 {
150 double pulse = 0.0;
151 double r = 0.0;
152 for (int i = 0; i < this->Dimension; ++i)
153 {
154 double d = pt[i] - this->PulseOrigin[i];
155 double d2 = d * d;
156 double L2 = this->PulseWidth[i] * this->PulseWidth[i];
157 r += d2 / L2;
158 }
159 pulse = this->PulseAmplitude * std::exp(-r);
160 return (pulse);
161 }
