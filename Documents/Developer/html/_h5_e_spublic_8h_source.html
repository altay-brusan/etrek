43/* Asynchronous operation status */
44typedef enum H5ES_status_t {
45 H5ES_STATUS_IN_PROGRESS, /* Operation(s) have not yet completed */
46 H5ES_STATUS_SUCCEED, /* Operation(s) have completed, successfully */
47 H5ES_STATUS_CANCELED, /* Operation(s) has been canceled */
48 H5ES_STATUS_FAIL /* An operation has completed, but failed */
49} H5ES_status_t;

51/* Information about operations in an event set */
52typedef struct H5ES_op_info_t {
53 /* API call info */
54 const char *api_name; /* Name of HDF5 API routine called */
55 char * api_args; /* "Argument string" for arguments to HDF5 API routine called */
56
57 /* Application info */
58 const char *app_file_name; /* Name of source file where the HDF5 API routine was called */
59 const char *app_func_name; /* Name of function where the HDF5 API routine was called */
60 unsigned app_line_num; /* Line # of source file where the HDF5 API routine was called */
61
62 /* Operation info */
63 uint64_t op_ins_count; /* Counter of operation's insertion into event set */
64 uint64_t op_ins_ts; /* Timestamp for when the operation was inserted into the event set */
65 uint64_t op_exec_ts; /* Timestamp for when the operation began execution */
66 uint64_t op_exec_time; /* Execution time for operation (in ns) */
67} H5ES_op_info_t;
