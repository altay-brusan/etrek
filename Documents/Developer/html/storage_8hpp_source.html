84 virtual int put(MemoryBuffer& bb) override
85 {
86 auto log = get_logger();
87 std::string filename;
88 int fh = open_random(filename);
89
90 log->debug("FileStorage::put(): {}; buffer size: {}", filename, bb.size());
91
92 size_t sz = bb.buffer.size();
93#if defined(_WIN32)
94 using r_type = int;
95 r_type written = _write(fh, &bb.buffer[0], static_cast<unsigned int>(sz));
96#else
97 using r_type = ssize_t;
98 r_type written = write(fh, &bb.buffer[0], sz);
99#endif
100 if (written < static_cast<r_type>(sz) || written == r_type(-1))
101 log->warn("Could not write the full buffer to {}: written = {}; size = {}", filename, written, sz);
102 io::utils::close(fh);
103 bb.wipe();
104
105#if 0 // double-check the written file size: only for extreme debugging
106 FILE* fp = fopen(filename.c_str(), "r");
107 fseek(fp, 0L, SEEK_END);
108 int fsz = ftell(fp);
109 if (fsz != sz)
110 log->warn("file size doesn't match the buffer size, {} vs {}", fsz, sz);
111 fclose(fp);
112#endif
113
114 return make_file_record(filename, sz);
115 }

214 int make_file_record(const std::string& filename, size_t sz)
215 {
216 int res = (*count_.access())++;
217 FileRecord fr = { sz, filename };
218 (*filenames_.access())[res] = fr;
219
220 // keep track of sizes
221 critical_resource<size_t>::accessor cur = current_size_.access();
222 *cur += sz;
223 critical_resource<size_t>::accessor max = max_size_.access();
224 if (*cur > *max)
225 *max = *cur;
226
227 return res;
228 }
