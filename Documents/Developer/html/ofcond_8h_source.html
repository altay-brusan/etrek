175class DCMTK_OFSTD_EXPORT OFCondition

178
186 OFCondition(unsigned short aModule, unsigned short aCode, OFStatus aStatus, const char *aText)
187 : theCondition()
188 , ownsText(OFTrue)
189 {
190 theCondition.theModule = aModule;
191 theCondition.theCode = aCode;
192 theCondition.theStatus = aStatus;
193 /* Be nice when someone dares to pass in NULL */
194 if (aText != NULL) {
195 theCondition.theText = strdup(aText);
196 ownsText = OFTrue;
197 } else {
198 theCondition.theText = "";
199 ownsText = OFFalse;
200 }
201 }
202
206 OFCondition(const OFConditionConst& aConst = EC_Normal)
207 : theCondition(aConst)
208 , ownsText(OFFalse)
209 {
210 }
211
215 OFCondition(const OFCondition& arg)
216 : theCondition(arg.theCondition)
217 , ownsText(arg.ownsText)
218 {
219 // Do we need our own copy of the text?
220 if (ownsText)
221 {
222 theCondition.theText = strdup(theCondition.theText);
223 }
224 }
225
226// suppress spurious gcc 12 warning about use-after-free.
227// this is a known bug in gcc.
228#include DCMTK_DIAGNOSTIC_PUSH
229#include DCMTK_DIAGNOSTIC_IGNORE_USE_AFTER_FREE
230
232 ~OFCondition()
233 {
234 if (ownsText)
235 {
236 free(OFconst_cast(char *, theCondition.theText)); // cast away const
237 }
238 }
239
240#include DCMTK_DIAGNOSTIC_POP
241
246 OFCondition& operator=(const OFCondition& arg)
247 {
248 if (&arg != this)
249 {
250 if (ownsText)
251 {
252 free(OFconst_cast(char *, theCondition.theText)); // cast away const
253 }
254 theCondition = arg.theCondition;
255 ownsText = arg.ownsText;
256 if (ownsText)
257 {
258 theCondition.theText = strdup(arg.theCondition.theText);
259 }
260 }
261 return *this;
262 }

312 inline OFBool good() const

321 inline OFBool bad() const

327#ifdef OFCONDITION_IMPLICIT_BOOL_CONVERSION
328 /* Implicit conversion from OFCondition to bool might
329 * not always be a good idea since it can hide unwanted constructs.
330 * Therefore, we disable this operator by default.
331 * @remark this method is only available if DCMTK is compiled with
332 * implicit conversion from OFCondition to bool being enabled (see macro
333 * OFCONDITION_IMPLICIT_BOOL_CONVERSION)
334 */
335
339 inline operator OFBool() const
340 {
341 return good();
342 }
343#endif
