288//------------------------------------------------------------------------------
289inline vtkIdList* vtkCellIterator::GetSerializedCellFaces()
290{
291 if (!this->CheckCache(FacesFlag))
292 {
293 this->FetchFaces();
294 this->SetCache(FacesFlag);
295 }
296 // Export Legacy Format
297 vtkNew<vtkIdTypeArray> tmp;
298 this->Faces->ExportLegacyFormat(tmp);
299 this->LegacyFacesContainer->Initialize();
300 this->LegacyFacesContainer->InsertNextId(this->Faces->GetNumberOfCells());
301 for (vtkIdType idx = 0; idx < tmp->GetNumberOfValues(); ++idx)
302 {
303 this->LegacyFacesContainer->InsertNextId(tmp->GetValue(idx));
304 }
305 return this->LegacyFacesContainer;
306}

339 case VTK_QUADRATIC_LINEAR_QUAD:
340 case VTK_BIQUADRATIC_TRIANGLE:
341 case VTK_CUBIC_LINE:

344 case VTK_PARAMETRIC_SURFACE:

351 case VTK_LAGRANGE_CURVE:
352 case VTK_LAGRANGE_TRIANGLE:
353 case VTK_LAGRANGE_QUADRILATERAL:
354 case VTK_BEZIER_CURVE:
355 case VTK_BEZIER_TRIANGLE:
356 case VTK_BEZIER_QUADRILATERAL:
357 return 0;

359 case VTK_TETRA:

383 case VTK_TRIQUADRATIC_HEXAHEDRON:
384 case VTK_HIGHER_ORDER_HEXAHEDRON:
385 case VTK_PARAMETRIC_HEX_REGION:
386 case VTK_BIQUADRATIC_QUADRATIC_HEXAHEDRON:
387 case VTK_LAGRANGE_HEXAHEDRON:
388 case VTK_BEZIER_HEXAHEDRON:
389 return 6;
390
391 case VTK_PENTAGONAL_PRISM:
392 return 7;
393
394 case VTK_HEXAGONAL_PRISM:
395 return 8;
396
397 case VTK_POLYHEDRON: // Need to look these up
398 if (!this->CheckCache(FacesFlag))
399 {
400 this->FetchFaces();
401 this->SetCache(FacesFlag);
402 }
403 return this->Faces->GetNumberOfCells();
404
405 default:
406 vtkGenericWarningMacro("Unknown cell type: " << this->CellType);
407 break;
408 }

410 return 0;
