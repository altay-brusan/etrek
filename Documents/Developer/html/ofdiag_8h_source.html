278// readable shorthands for compiler version checks
279#define DCMTK_DIAGNOSTIC_MIN_GCC_VERSION(MAJOR, MINOR, PATCH)\
280(\
281 defined(__GNUC__) &&\
282 (\
283 __GNUC__ > MAJOR ||\
284 (\
285 __GNUC__ == MAJOR &&\
286 (\
287 __GNUC_MINOR__ > MINOR ||\
288 (\
289 __GNUC_MINOR == MINOR &&\
290 __GNUC_PATCHLEVEL__ >= PATCH\
291 )\
292 )\
293 )\
294 )\
295)
296
