249template <typename T_Units, OFBool T_DefinedGroup>
250OFCondition CMR_SRNumericMeasurementValueWithUnits<T_Units, T_DefinedGroup>::setValue(const OFString &numericValue,
251 const T_Units &measurementUnit,
252 const CID42_NumericValueQualifier &valueQualifier,
253 const OFBool check)
254{
255 OFCondition result = EC_Normal;
256 if (check)
257 {
258 /* only check if at least one of the two values is non-empty */
259 if (!numericValue.empty() || !measurementUnit.hasSelectedValue())
260 {
261 /* check whether the passed values are valid */
262 result = checkNumericValue(numericValue);
263 if (result.good())
264 result = checkMeasurementUnit(measurementUnit);
265 }
266 if (result.good())
267 result = checkNumericValueQualifier(valueQualifier);
268 }
269 /* set the value (without checking all the details again) */
270 if (result.good())
271 result = CMR_SRNumericMeasurementValue::setValue(numericValue, measurementUnit, valueQualifier, OFFalse /*check*/);
272 return result;
273}

280 /* first, check given measurement unit for validity (if not disabled) */

289template <typename T_Units, OFBool T_DefinedGroup>
290OFCondition CMR_SRNumericMeasurementValueWithUnits<T_Units, T_DefinedGroup>::checkMeasurementUnit(const DSRCodedEntryValue &measurementUnit) const
291{
292 /* first, check coded entry for basic validity */
293 OFCondition result = CMR_SRNumericMeasurementValue::checkMeasurementUnit(measurementUnit);
294 /* then, check whether coded entry is allowed (if defined context group) */
295 if (result.good() && T_DefinedGroup)
296 {
297 const T_Units contextGroup;
298 if (!contextGroup.hasCodedEntry(measurementUnit))
299 result = SR_EC_CodedEntryNotInContextGroup;
300 }
301 return result;
302}
