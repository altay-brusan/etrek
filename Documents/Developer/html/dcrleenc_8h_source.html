96 inline void add(unsigned char ch)
97 {
98 if (! fail_) // if fail_ is true, just ignore input
99 {
100 // if the current byte equals the last byte read
101 // (which is initialized with the "impossible" value -1),
102 // just increase the repeat counter
103 if (OFstatic_cast(int, ch) == RLE_prev_) RLE_pcount_++;
104 else
105 {
106 // byte is different from last byte read.
107 // flush replicate run if necessary
108 switch (RLE_pcount_)
109 {
110 case 0:
111 // happens only after construction or flush()
112 break;
113 case 2:
114 // two bytes in repeat buffer. Convert to literal run
115 RLE_buff_[RLE_bindex_++] = OFstatic_cast(unsigned char, RLE_prev_);
116 // no break. Fall-through into next case statement is intended.
117 case 1:
118 // one (or two) bytes in repeat buffer. Convert to literal run
119 RLE_buff_[RLE_bindex_++] = OFstatic_cast(unsigned char, RLE_prev_);
120 break;
121 default:
122 // more than two bytes in repeat buffer. Convert to replicate run
123 if (RLE_bindex_ > 1)
124 {
125 // there is a literal run in the buffer that must be flushed
126 // before the replicate run. Flush literal run now.
127 RLE_buff_[0] = OFstatic_cast(unsigned char, RLE_bindex_-2);
128 move(RLE_bindex_);
129 }
130 // this is the byte value for the repeat run
131 RLE_buff_[1] = OFstatic_cast(unsigned char, RLE_prev_);
132 // write as many repeat runs as necessary
133 for (; RLE_pcount_>0; RLE_pcount_-=128)
134 {
135 // different PackBit schemes exist. The original from which
136 // this code is derived used 0x80 | (RLE_pcount_ - 1)
137 // to represent replicate runs.
138 // DICOM instead uses 257 - RLE_pcount_
139 if (RLE_pcount_ > 128) RLE_buff_[0] = 0x81;
140 else RLE_buff_[0] = OFstatic_cast(unsigned char, 257 - RLE_pcount_);
141 move(2);
142 }
143 // now the buffer is guaranteed to be empty
144 RLE_buff_[0] = 0;
145 RLE_bindex_ = 1;
146 break;
147 }
148
149 // if we have 128 or more bytes in the literal run, flush buffer
150 if (RLE_bindex_ > 129)
151 {
152 RLE_buff_[0] = 127;
153 move(129);
154 RLE_bindex_ -= 128;
155 if (RLE_bindex_ > 1)
156 RLE_buff_[1] = RLE_buff_[129];
157 if (RLE_bindex_ > 2)
158 RLE_buff_[2] = RLE_buff_[130];
159 }
160
161 // current byte is stored in RLE_prev_, RLE_pcount_ is 1.
162 RLE_prev_ = ch;
163 RLE_pcount_ = 1;
164 }
165 }
166 }

187 inline void flush()
188 {
189 if (! fail_) // if fail_ is true, do nothing
190 {
191 // if there are max 1 bytes in the repeat counter, convert to literal run
192 if (RLE_pcount_ < 2)
193 {
194 for (; RLE_pcount_>0; --RLE_pcount_) RLE_buff_[RLE_bindex_++] = OFstatic_cast(unsigned char, RLE_prev_);
195 }
196
197 // if we have 128 or more bytes in the literal run, flush buffer
198 if (RLE_bindex_ > 129)
199 {
200 RLE_buff_[0] = 127;
201 move(129);
202 RLE_bindex_ -= 128;
203 if (RLE_bindex_ > 1)
204 RLE_buff_[1] = RLE_buff_[129];
205 if (RLE_bindex_ > 2)
206 RLE_buff_[2] = RLE_buff_[130];
207 }
208
209 // if there is still a literal run in the buffer, flush literal run
210 if (RLE_bindex_ > 1)
211 {
212 RLE_buff_[0] = OFstatic_cast(unsigned char, RLE_bindex_-2);
213 move(RLE_bindex_);
214 }
215
216 // if there is a remaining repeat run, flush this one as well
217 if (RLE_pcount_ >= 2)
218 {
219 RLE_buff_[1] = OFstatic_cast(unsigned char, RLE_prev_);
220 // write as many repeat runs as necessary
221 for (; RLE_pcount_>0; RLE_pcount_-=128)
222 {
223 // different PackBit schemes exist. The original from which
224 // this code is derived used 0x80 | (RLE_pcount_ - 1)
225 // to represent replicate runs.
226 // DICOM instead uses 257 - RLE_pcount_
227 if (RLE_pcount_ > 128) RLE_buff_[0] = 0x81;
228 else RLE_buff_[0] = OFstatic_cast(unsigned char, 257 - RLE_pcount_);
229 move(2);
230 }
231 }
232
233 // now the buffer is guaranteed to be empty, re-initialize
234 RLE_buff_[0] = 0;
235 RLE_prev_ = -1;
236 RLE_pcount_ = 0;
237 RLE_bindex_ = 1;
238 }
239 }

267 inline void write(void *target) const
268 {
269 if ((!fail_) && target)
270 {
271 unsigned char *current = NULL;
272 unsigned char *target8 = OFstatic_cast(unsigned char *, target);
273 OFListConstIterator(unsigned char *) first = blockList_.begin();
274 OFListConstIterator(unsigned char *) last = blockList_.end();
275 while (first != last)
276 {
277 current = *first;
278 memcpy(target8, current, DcmRLEEncoder_BLOCKSIZE);
279 target8 += DcmRLEEncoder_BLOCKSIZE;
280 ++first;
281 }
282 if (offset_ > 0)
283 {
284 memcpy(target8, currentBlock_, offset_);
285 }
286
287 // pad to even number of bytes if necessary
288 if (pad_ && ((blockList_.size() * DcmRLEEncoder_BLOCKSIZE + offset_) & 1))
289 {
290 target8 += offset_;
291 *target8 = 0;
292 }
293 }
294 }

300 inline void write(DcmEncoderOutputStream& os) const
301 {
302 if (!fail_)
303 {
304 OFListConstIterator(unsigned char *) first = blockList_.begin();
305 OFListConstIterator(unsigned char *) last = blockList_.end();
306 while (first != last)
307 {
308 os.write(*first, DcmRLEEncoder_BLOCKSIZE);
309 ++first;
310 }
311 if (offset_ > 0)
312 {
313 os.write(currentBlock_, offset_);
314 }
315
316 // pad to even number of bytes if necessary
317 if (pad_ && ((blockList_.size() * DcmRLEEncoder_BLOCKSIZE + offset_) & 1))
318 {
319 unsigned char c = 0;
320 os.write(&c, 1);
321 }
322 }

338 inline void move(size_t numberOfBytes)
339 {
340 size_t i=0;
341 while (i < numberOfBytes)
342 {
343 if (offset_ == DcmRLEEncoder_BLOCKSIZE)
344 {
345 blockList_.push_back(currentBlock_);
346 currentBlock_ = new unsigned char[DcmRLEEncoder_BLOCKSIZE];
347 offset_ = 0;
348 if (! currentBlock_) // out of memory
349 {
350 fail_ = 1;
351 break; // exit while loop
352 }
353 }
354 currentBlock_[offset_++] = RLE_buff_[i++];
355 }
356 }
