399// In-lined for performance
400inline int vtkIncrementalOctreeNode::GetChildIndex(const double point[3])
401{
402 // Children[0]->MaxBounds[] is exactly the center point of this node.
403 return int(point[0] > this->Children[0]->MaxBounds[0]) +
404 ((int(point[1] > this->Children[0]->MaxBounds[1])) << 1) +
405 ((int(point[2] > this->Children[0]->MaxBounds[2])) << 2);
406}

408// In-lined for performance
409inline vtkTypeBool vtkIncrementalOctreeNode::ContainsPoint(const double pnt[3])
410{
411 return (
412 (this->MinBounds[0] < pnt[0] && pnt[0] <= this->MaxBounds[0] && this->MinBounds[1] < pnt[1] &&
413 pnt[1] <= this->MaxBounds[1] && this->MinBounds[2] < pnt[2] && pnt[2] <= this->MaxBounds[2])
414 ? 1
415 : 0);
416}

418// In-lined for performance
419inline vtkTypeBool vtkIncrementalOctreeNode::ContainsPointByData(const double pnt[3])
420{
421 return ((this->MinDataBounds[0] <= pnt[0] && pnt[0] <= this->MaxDataBounds[0] &&
422 this->MinDataBounds[1] <= pnt[1] && pnt[1] <= this->MaxDataBounds[1] &&
423 this->MinDataBounds[2] <= pnt[2] && pnt[2] <= this->MaxDataBounds[2])
424 ? 1
425 : 0);
426}

428// In-lined for performance
429inline int vtkIncrementalOctreeNode::ContainsDuplicatePointsOnly(const double pnt[3])
430{
431 return ((this->MinDataBounds[0] == pnt[0] && pnt[0] == this->MaxDataBounds[0] &&
432 this->MinDataBounds[1] == pnt[1] && pnt[1] == this->MaxDataBounds[1] &&
433 this->MinDataBounds[2] == pnt[2] && pnt[2] == this->MaxDataBounds[2])
434 ? 1
435 : 0);
436}

438// In-lined for performance
439inline void vtkIncrementalOctreeNode::UpdateCounterAndDataBounds(const double point[3])
440{
441 this->NumberOfPoints++;
442
443 this->MinDataBounds[0] = (point[0] < this->MinDataBounds[0]) ? point[0] : this->MinDataBounds[0];
444 this->MinDataBounds[1] = (point[1] < this->MinDataBounds[1]) ? point[1] : this->MinDataBounds[1];
445 this->MinDataBounds[2] = (point[2] < this->MinDataBounds[2]) ? point[2] : this->MinDataBounds[2];
446 this->MaxDataBounds[0] = (point[0] > this->MaxDataBounds[0]) ? point[0] : this->MaxDataBounds[0];
447 this->MaxDataBounds[1] = (point[1] > this->MaxDataBounds[1]) ? point[1] : this->MaxDataBounds[1];
448 this->MaxDataBounds[2] = (point[2] > this->MaxDataBounds[2]) ? point[2] : this->MaxDataBounds[2];
449}

451// In-lined for performance
452inline int vtkIncrementalOctreeNode::UpdateCounterAndDataBoundsRecursively(
453 const double point[3], int nHits, int updateData, vtkIncrementalOctreeNode* endNode)
454{
455 int updated = this->UpdateCounterAndDataBounds(point, nHits, updateData);
456
457 return ((this->Parent == endNode)
458 ? updated
459 : this->Parent->UpdateCounterAndDataBoundsRecursively(point, nHits, updated, endNode));
460}
