146#define vtkInformationKeySetStringMacro(name) \
147 virtual void Set##name(const char* _arg) \
148 { \
149 if (this->name == nullptr && _arg == nullptr) \
150 { \
151 return; \
152 } \
153 if (this->name && _arg && (!strcmp(this->name, _arg))) \
154 { \
155 return; \
156 } \
157 delete[] this->name; \
158 if (_arg) \
159 { \
160 size_t n = strlen(_arg) + 1; \
161 char* cp1 = new char[n]; \
162 const char* cp2 = (_arg); \
163 this->name = cp1; \
164 do \
165 { \
166 *cp1++ = *cp2++; \
167 } while (--n); \
168 } \
169 else \
170 { \
171 this->name = nullptr; \
172 } \
173 }
