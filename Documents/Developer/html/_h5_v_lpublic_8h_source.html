63/* Flags to return from H5VLquery_optional API and 'opt_query' callbacks */
64/* Note: Operations which access multiple objects' data or metadata in a
65 * container should be registered as file-level optional operations.
66 * (e.g. "H5Dwrite_multi" takes a list of datasets to write data to, so
67 * a VOL connector that implemented it should register it as an optional
68 * file operation, and pass-through VOL connectors that are stacked above
69 * the connector that registered it should assume that dataset elements
70 * for _any_ dataset in the file could be written to)
71 */
72#define H5VL_OPT_QUERY_SUPPORTED 0x0001 /* VOL connector supports this operation */
73#define H5VL_OPT_QUERY_READ_DATA 0x0002 /* Operation reads data for object */
74#define H5VL_OPT_QUERY_WRITE_DATA 0x0004 /* Operation writes data for object */
75#define H5VL_OPT_QUERY_QUERY_METADATA 0x0008 /* Operation reads metadata for object */
76#define H5VL_OPT_QUERY_MODIFY_METADATA 0x0010 /* Operation modifies metadata for object */
77#define H5VL_OPT_QUERY_COLLECTIVE \
78 0x0020 /* Operation is collective (operations without this flag are assumed to be independent) */
79#define H5VL_OPT_QUERY_NO_ASYNC 0x0040 /* Operation may NOT be executed asynchronously */
80#define H5VL_OPT_QUERY_MULTI_OBJ 0x0080 /* Operation involves multiple objects */
