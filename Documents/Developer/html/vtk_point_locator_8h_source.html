268 void GetBucketIndices(const double* x, int ijk[3]) const
269 {
270 // Compute point index. Make sure it lies within range of locator.
271 vtkIdType tmp0 = static_cast<vtkIdType>(((x[0] - this->BX) * this->FX));
272 vtkIdType tmp1 = static_cast<vtkIdType>(((x[1] - this->BY) * this->FY));
273 vtkIdType tmp2 = static_cast<vtkIdType>(((x[2] - this->BZ) * this->FZ));
274
275 ijk[0] = tmp0 < 0 ? 0 : (tmp0 >= this->XD ? this->XD - 1 : tmp0);
276 ijk[1] = tmp1 < 0 ? 0 : (tmp1 >= this->YD ? this->YD - 1 : tmp1);
277 ijk[2] = tmp2 < 0 ? 0 : (tmp2 >= this->ZD ? this->ZD - 1 : tmp2);
278 }
279
280 vtkIdType GetBucketIndex(const double* x) const
281 {
282 int ijk[3];
283 this->GetBucketIndices(x, ijk);
284 return ijk[0] + ijk[1] * this->XD + ijk[2] * this->SliceSize;
285 }
