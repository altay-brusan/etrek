124template<typename T>
125class OFdefault_optional_traits
126{
127#ifndef DOXYGEN
128 // types for detecting T's default constructibility via sfinae
129 struct no_type {};
130 struct yes_type {double d;};
131#ifdef HAVE_DEFAULT_CONSTRUCTOR_DETECTION_VIA_SFINAE
132 // helper class to create an argument out of size_t
133 template<size_t>
134 struct consume{};
135 // sfinae overload working for default constructible Xs
136 template<typename X>
137 static yes_type sfinae(consume<sizeof *new X>*);
138#elif defined(_MSC_VER)
139#if _MSC_VER < 1700
140 // Workaround bug in Visual Studio.
141 // On these broken compilers, the argument is not evaluated
142 // unless we require them to evaluate it for choosing which
143 // specialization should be instantiated.
144 template<size_t,size_t>
145 struct consume{};
146 template<size_t X>
147 struct consume<X,X> { typedef void type; };
148 // sfinae overload working for value-initializable Xs, that's as
149 // close as we get on these broken compilers
150 template<typename X>
151 static yes_type sfinae(typename consume<sizeof X(),sizeof X()>::type*);
152#else
153 // Visual Studio 2012 is completely broken, but it has std::is_default_constructible
154 // Note: this tests constructibility, but not if WE can construct this.
155 template<typename X>
156 static yes_type sfinae(typename OFenable_if<std::is_default_constructible<X>::value>::type*);
157#endif // _MSC_VER
158#endif // HAVE_DEFAULT_CONSTRUCTOR_DETECTION_VIA_SFINAE
159 // most general sfinae overload, chosen only if everything else fails
160 template<typename X>
161 static no_type sfinae(...);
162
163public:
164 struct is_default_constructible
165 : OFintegral_constant<OFBool,sizeof(sfinae<T>(OFnullptr)) == sizeof(yes_type)> {};
166
167#ifdef HAVE_CXX11
168
169 template<typename... Args>
170 void construct( Args&&... args )
171 { new (content()) T( std::forward<Args>( args )... ); m_State = OFTrue; }
172
173#else // C++11
174
175#ifdef OFalign
176 void construct() { new (content()) T; m_State = OFTrue; }
177 template<typename X>
178 void construct( const X& x ) { new (content()) T( x ); m_State = OFTrue; }
179#else
180 void construct() { m_pContent = new T; }
181 template<typename X>
182 void construct( const X& x ) { m_pContent = new T( x ); }
183#endif
184
185#endif // NOT C++11
186
187#ifdef OFalign
188 // State and content are stored in the same data-array to
189 // optimize alignment and so on. Mutable, since the content
190 // itself can be modified even if it can't be (re-)assigned.
191 OFdefault_optional_traits() : m_State( OFFalse ) {}
192 void destroy() { OFstatic_cast( T*, content() )->~T(); m_State = OFFalse; }
193 OFBool state() const { return m_State; }
194 void* content() const { return m_Content; }
195 OFalign_typename(Uint8[sizeof(T)],T) mutable m_Content;
196 OFBool m_State;
197#else
198 // Allocate content on the heap.
199 OFdefault_optional_traits() : m_pContent( OFnullptr ) {}
200 void destroy() { delete m_pContent; m_pContent = OFnullptr; }
201 OFBool state() const { return m_pContent; }
202 void* content() const { return m_pContent; }
203 T* m_pContent;
204#endif
205
206#endif // NOT DOXYGEN
207};

790 // Move assignment
791 OFoptional& operator=( OFoptional&& rhs )
792 {
793 // Prevent self-assignment
794 if( &rhs != this )
795 {
796 // if both objects are engaged, move content
797 // and kill rhs
798 if( OFoptional_traits<T>::state() == rhs.state() )
799 {
800 if( OFoptional_traits<T>::state() )
801 {
802 (**this) = std::move( *rhs );
803 rhs.destroy();
804 }
805 }
806 else if( OFoptional_traits<T>::state() ) // suicide if engaged and rhs isn't
807 {
808 OFoptional_traits<T>::destroy();
809 }
810 else // if rhs is engaged and we aren't, swap states, move contents and kill rhs.
811 {
812 OFoptional_traits<T>::construct( std::move( *rhs ) );
813 rhs.destroy();
814 }
815 }

903 // Swap the contents with another OFoptional object.
904 void swap( OFoptional& rhs )
905 {
906 // if both objects are engaged, the contents are swapped.
907 if( OFoptional_traits<T>::state() == rhs.state() )
908 {
909 if( OFoptional_traits<T>::state() ) OFswap( **this, *rhs );
910 }
911 else
912 {
913 // if we are engaged and rhs isn't, move assign
914 // our contents to rhs.
915 if( OFoptional_traits<T>::state() )
916 {
917 rhs.construct( OFmove( **this ) );
918 OFoptional_traits<T>::destroy();
919 }
920 else // else move assign rhs' contents to us
921 {
922 OFoptional_traits<T>::construct( OFmove( *rhs ) );
923 rhs.destroy();
924 }
925 }
