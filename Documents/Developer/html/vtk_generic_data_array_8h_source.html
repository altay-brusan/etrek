439// Adds an implementation of NewInstanceInternal() that returns an AoS
440// (unmapped) VTK array, if possible. This allows the pipeline to copy and
441// propagate the array when the array data is not modifiable. Use this in
442// combination with vtkAbstractTypeMacro or vtkAbstractTemplateTypeMacro
443// (instead of vtkTypeMacro) to avoid adding the default NewInstance
444// implementation.
445#define vtkAOSArrayNewInstanceMacro(thisClass) \
446protected: \
447 vtkObjectBase* NewInstanceInternal() const override \
448 { \
449 if (vtkDataArray* da = vtkDataArray::CreateDataArray(thisClass::VTK_DATA_TYPE)) \
450 { \
451 return da; \
452 } \
453 return thisClass::New(); \
454 } \
455 \
456public:
457
458#endif
