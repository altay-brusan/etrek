31 ZoneConnectivity(std::string name, int owner_zone, std::string donor_name, int donor_zone,
32 const Ioss::IJK_t p_transform, const Ioss::IJK_t range_beg,
33 const Ioss::IJK_t range_end, const Ioss::IJK_t donor_beg,

46 ZoneConnectivity(std::string name, int owner_zone, std::string donor_name, int donor_zone,

63 // Return number of nodes in the connection shared with the donor zone.
64 IOSS_NODISCARD size_t get_shared_node_count() const
65 {
66 size_t snc = 1;
67 for (int i = 0; i < 3; i++) {
68 snc *= (std::abs(m_ownerRangeEnd[i] - m_ownerRangeBeg[i]) + 1);
69 }
70 return snc;
71 }

106 // The following are all subsetted down to the portion that is actually on this zone
107 // This can be different than m_ownerRange and m_donorRange in a parallel run if the
108 // decomposition splits the connection. In a serial run, they are the same.
109 //
110 // 1 of ijk should be the same for rangeBeg and rangeEnd defining a surface.
111 Ioss::IJK_t m_ownerRangeBeg{}; // ijk triplet defining beginning of range on this zone
112 Ioss::IJK_t m_ownerRangeEnd{}; // ijk triplet defining end of range on this zone
113 Ioss::IJK_t m_ownerOffset{}; // ijk triplet with offset of this zone. Used to convert
114 // rangeBeg and rangeEnd global indices to local indices
115 Ioss::IJK_t m_donorRangeBeg{}; // ijk triplet defining beginning of range on the connected zone
116 Ioss::IJK_t m_donorRangeEnd{}; // ijk triplet defining end of range on the connected zone
117 Ioss::IJK_t m_donorOffset{}; // ijk triplet with offset of the donor zone. Used to convert
118 // donorRangeBeg and End global indices to local indices
119
120 size_t m_ownerGUID{}; // globally-unique id of owner
121 size_t m_donorGUID{}; // globally-unique id of donor
122
123 // NOTE: Shared nodes are "owned" by the zone with the lowest zone id.
124 int m_ownerZone{}; // "id" of zone that owns this connection
125 int m_donorZone{}; // "id" of zone that is donor (or other side) of this connection
126 int m_ownerProcessor{-1}; // processor that owns the owner zone
127 int m_donorProcessor{-1}; // processor that owns the donor zone
128 bool m_sameRange{false}; // True if owner and donor range should always match...(special use
129 // during decomp)
130 // True if it is the "lower" zone id in the connection. Uses adam unless both have same adam.
131 bool m_ownsSharedNodes{false}; // Deprecate soon
132
133 // True if this zc is created due to processor decompositions in a parallel run
134 mutable bool m_fromDecomp{false};
135
136 bool m_isActive{true}; // True if non-zero range. That is, it has at least one face
137
138 private:
139 bool equal_(const Ioss::ZoneConnectivity &rhs, bool quiet) const;
140 };
