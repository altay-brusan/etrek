28// If VTK_USE_PTHREADS is defined, then pthread_create() will be
29// used to create multiple threads
30
31// If VTK_USE_PTHREADS is defined, then the multithreaded
32// function is of type void *, and returns nullptr
33// Otherwise the type is void which is correct for WIN32
34
35// Defined in vtkThreads.h:
36// VTK_MAX_THREADS
37// VTK_THREAD_RETURN_VALUE
38// VTK_THREAD_RETURN_TYPE
39
40#ifdef VTK_USE_PTHREADS
41typedef void* (*vtkThreadFunctionType)(void*);
42typedef pthread_t vtkThreadProcessIDType;
43// #define VTK_THREAD_RETURN_VALUE nullptr
44// #define VTK_THREAD_RETURN_TYPE void *
45typedef pthread_t vtkMultiThreaderIDType;
46#endif
47
48#ifdef VTK_USE_WIN32_THREADS
49typedef vtkWindowsLPTHREAD_START_ROUTINE vtkThreadFunctionType;
50typedef vtkWindowsHANDLE vtkThreadProcessIDType;
51// #define VTK_THREAD_RETURN_VALUE 0
52// #define VTK_THREAD_RETURN_TYPE DWORD __stdcall
53typedef vtkWindowsDWORD vtkMultiThreaderIDType;
54#endif
55
56#if !defined(VTK_USE_PTHREADS) && !defined(VTK_USE_WIN32_THREADS)
57typedef void (*vtkThreadFunctionType)(void*);
58typedef int vtkThreadProcessIDType;
59// #define VTK_THREAD_RETURN_VALUE
60// #define VTK_THREAD_RETURN_TYPE void
61typedef int vtkMultiThreaderIDType;
62#endif
