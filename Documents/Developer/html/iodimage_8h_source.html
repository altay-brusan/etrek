249 OFCondition readIntegerImagePixel(DcmItem& dataset)
250 {
251 Uint16 allocated = 0;
252 Uint16 representation = 0;
253 // Pixel data is not managed right now by Image Pixel Module class.
254 if (/* dataset.tagExists(DCM_PixelData) && */ dataset.findAndGetUint16(DCM_BitsAllocated, allocated).good()
255 && dataset.findAndGetUint16(DCM_PixelRepresentation, representation).good())
256 switch ((allocated > 8 ? 2 : 0) | (representation ? 1 : 0))
257 {
258 case 0:
259 return readImagePixel<Uint8>(dataset);
260 case 1:
261 return readImagePixel<Sint8>(dataset);
262 case 2:
263 return readImagePixel<Uint16>(dataset);
264 case 3:
265 return readImagePixel<Sint16>(dataset);
266 default:
267 break;
268 }
269 return IOD_EC_InvalidPixelData;
270 }
