146 // Used to control how the merge tolerance value is interpreted.
147 // ABSOLUTE_TOLERANCE is a tolerance expressed in world coordinates;
148 // RELATIVE_TO_TOLERANCE is a tolerance relative to the projection
149 // tolerance (i.e., MergeTolerance is a scale factor on Tolerance);
150 // RELATIVE_TO_MIN_EDGE_LENGTH is a tolerance relative to the minimum edge
151 // length of the tool/imprint mesh; and RELATIVE_TO_AVERAGE_EDGE_LENGTH is
152 // a tolerance based on the average edge length of all imprint mesh edges.
153 enum MergeTolType
154 {
155 ABSOLUTE_TOLERANCE = 0,
156 RELATIVE_TO_PROJECTION_TOLERANCE = 1,
157 RELATIVE_TO_MIN_EDGE_LENGTH = 2,
158 RELATIVE_TO_AVERAGE_EDGE_LENGTH = 3
159 };

186 // Used to control the relationship between the merge tolerance
187 // (MergeTolerance) and projection tolerance (Tolerance) as these need to
188 // be treated consistently to minimize numerical issues. The projection
189 // tolerance addresses projection of the imprint onto the target, while
190 // the merge tolerance is about computing accurate in-plane edge-edge
191 // intersections.
192 enum ToleranceStrategy
193 {
194 DECOUPLED_TOLERANCES = 0,
195 LINKED_TOLERANCES = 1
196 };
