12inline void send_req(AuxBlock*, // local block (unused)
13 const diy::Master::ProxyWithLink& cp, // communication proxy for neighbor blocks
14 std::set<int>& procs) // processes to query
15{
16 // send requests for work info to sample_procs
17 int v = 1; // any message will do
18 for (auto proc_iter = procs.begin(); proc_iter != procs.end(); proc_iter++)

20 int gid = *proc_iter;
21 int proc = *proc_iter;
22 diy::BlockID dest_block = {gid, proc};

31{

48// get work information from a random sample of processes
49inline void exchange_sample_work_info(diy::Master& master, // the real master with multiple blocks per process
50 diy::Master& aux_master, // auxiliary master with 1 block per process for communicating between procs
51 float sample_frac, // fraction of procs to sample 0.0 < sample_size <= 1.0
52 const WorkInfo& my_work_info, // my process' work info
53 std::vector<WorkInfo>& sample_work_info) // (output) vector of sorted sample work info, sorted by increasing total work per process
54{
55 auto nprocs = master.communicator().size(); // global number of procs
56 auto my_proc = master.communicator().rank(); // rank of my proc
57
58 // pick a random sample of processes, w/o duplicates, and excluding myself
59 int nsamples = static_cast<int>(sample_frac * (nprocs - 1));
60 std::set<int> sample_procs;
61 for (auto i = 0; i < nsamples; i++)
62 {
63 int rand_proc;
64 do
65 {
66 std::uniform_int_distribution<> distrib(0, nprocs - 1); // inclusive
67 rand_proc = distrib(master.mt_gen);
68 } while (sample_procs.find(rand_proc) != sample_procs.end() || rand_proc == my_proc);
69 sample_procs.insert(rand_proc);
70 }
71
72 // rexchange requests for work info
73 std::vector<int> req_procs; // requests for work info received from these processes
74 aux_master.foreach([&](AuxBlock* b, const diy::Master::ProxyWithLink& cp)
75 { send_req(b, cp, sample_procs); });
76 aux_master.exchange(true); // true = remote
77 aux_master.foreach([&](AuxBlock* b, const diy::Master::ProxyWithLink& cp)
78 { recv_req(b, cp, req_procs); });
79
80 // send work info
81 int work_info_tag = 0;
82 std::vector<diy::mpi::request> reqs(req_procs.size());
83 for (auto i = 0; i < req_procs.size(); i++)
84 reqs[i] = mpi::detail::isend(MPI_Comm(master.communicator()), req_procs[i], work_info_tag, &my_work_info, sizeof(WorkInfo), MPI_BYTE);
85
86 // receive work info
87 sample_work_info.resize(nsamples);
88 for (auto i = 0; i < nsamples; i++)
89 mpi::detail::recv(MPI_Comm(master.communicator()), diy::mpi::any_source, work_info_tag, &sample_work_info[i], sizeof(WorkInfo), MPI_BYTE);
90
91 // ensure all the send requests cleared
92 for (auto i = 0; i < req_procs.size(); i++)
93 reqs[i].wait();
94
95 // sort sample_work_info by proc_work
96 std::sort(sample_work_info.begin(), sample_work_info.end(),
97 [&](WorkInfo& a, WorkInfo& b) { return a.proc_work < b.proc_work; });
98}

100// send block
101inline void send_block(AuxBlock*, // local block (unused)
102 const diy::Master::ProxyWithLink& cp, // communication proxy for neighbor blocks
103 diy::Master& master, // real master with multiple blocks per process
104 const std::vector<WorkInfo>& sample_work_info, // sampled work info
105 const WorkInfo& my_work_info, // my work info
106 float quantile) // quantile cutoff above which to move blocks (0.0 - 1.0)
107{
108 MoveInfo move_info = {-1, -1, -1};
109
110 // my rank's position in the sampled work info, sorted by proc_work
111 int my_work_idx = sample_work_info.size(); // index where my work would be in the sample_work
112 for (auto i = 0; i < sample_work_info.size(); i++)
113 {
114 if (my_work_info.proc_work < sample_work_info[i].proc_work)
115 {
116 my_work_idx = i;
117 break;
118 }
119 }
120
121 // send my heaviest block if it passes the quantile cutoff
122 if (my_work_idx >= quantile * sample_work_info.size())
123 {
124 // pick the destination process to be the mirror image of my work location in the samples
125 // ie, the heavier my process, the lighter the destination process
126 int target = sample_work_info.size() - my_work_idx;
127
128 auto src_work_info = my_work_info;
129 auto dst_work_info = sample_work_info[target];
130
131 // sanity check that the move makes sense
132 if (src_work_info.proc_work - dst_work_info.proc_work > src_work_info.top_work && // improve load balance
133 src_work_info.proc_rank != dst_work_info.proc_rank && // not self
134 src_work_info.nlids > 1) // don't leave a proc with no blocks
135 {
136
137 move_info.move_gid = my_work_info.top_gid;
138 move_info.src_proc = my_work_info.proc_rank;
139 move_info.dst_proc = sample_work_info[target].proc_rank;
140
141 // destination in aux_master, where gid = proc
142 diy::BlockID dest_block = {move_info.dst_proc, move_info.dst_proc};
143
144 // enqueue the gid of the moving block
145 cp.enqueue(dest_block, move_info.move_gid);
146
147 // enqueue the block
148 void* send_b = master.block(master.lid(move_info.move_gid));
149 diy::MemoryBuffer bb;
150 master.saver()(send_b, bb);
151 cp.enqueue(dest_block, bb.buffer);
152
153 // enqueue the link for the block
154 diy::Link* send_link = master.link(master.lid(move_info.move_gid));
155 diy::LinkFactory::save(bb, send_link);
156 cp.enqueue(dest_block, bb.buffer);
157
158 // remove the block from the master
159 int move_lid = master.lid(move_info.move_gid);
160 master.destroyer()(master.release(move_lid));
161 }
162 }
163}

166inline void recv_block(AuxBlock*, // local block (unused)
167 const diy::Master::ProxyWithLink& cp, // communication proxy for neighbor blocks
168 diy::Master& master) // real master with multiple blocks per process
169{
170 std::vector<int> incoming_gids;
171 cp.incoming(incoming_gids);
172
173 // for anything incoming, dequeue data received in the last exchange
174 for (int i = 0; i < incoming_gids.size(); i++)
175 {
176 int gid = incoming_gids[i];
177 if (cp.incoming(gid).size())
178 {
179 // dequeue the gid of the moving block
180 int move_gid;
181 cp.dequeue(gid, move_gid);
182
183 // dequeue the block
184 void* recv_b = master.creator()();
185 diy::MemoryBuffer bb;
186 cp.dequeue(gid, bb.buffer);
187 master.loader()(recv_b, bb);
188
189 // dequeue the link
190 diy::Link* recv_link;
191 cp.dequeue(gid, bb.buffer);
192 recv_link = diy::LinkFactory::load(bb);
193
194 // add block to the master
195 master.add(move_gid, recv_b, recv_link);
196 }
