876inline const char* vtkProperty::GetInterpolationAsString()
877{
878 if (this->Interpolation == VTK_FLAT)
879 {
880 return "Flat";
881 }
882 else if (this->Interpolation == VTK_GOURAUD)
883 {
884 return "Gouraud";
885 }
886 else if (this->Interpolation == VTK_PHONG)
887 {
888 return "Phong";
889 }

891 {
892 return "Physically based rendering";
893 }
