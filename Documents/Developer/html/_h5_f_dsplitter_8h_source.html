34/* ----------------------------------------------------------------------------
35 * Structure: H5FD_spliiter_vfd_config_t
36 *
37 * One-stop shopping for configuring a Splitter VFD (rather than many
38 * parameters passed into H5Pset/get functions).
39 *
40 * magic (int32_t)
41 * Semi-unique number, used to sanity-check that a given pointer is
42 * likely (or not) to be this structure type. MUST be first.
43 * If magic is not H5FD_SPLITTER_MAGIC, the structure (and/or pointer to)
44 * must be considered invalid.
45 *
46 * version (unsigned int)
47 * Version number of this structure -- informs component membership.
48 * If not H5FD_CURR_SPLITTER_VFD_CONFIG_VERSION, the structure (and/or
49 * pointer to) must be considered invalid.
50 *
51 * rw_fapl_id (hid_t)
52 * Library-given identification number of the Read/Write channel driver
53 * File Access Property List.
54 * The driver must support read/write access.
55 * Must be set to H5P_DEFAULT or a valid FAPL ID.
56 *
57 * wo_fapl_id (hid_t)
58 * Library-given identification number of the Read/Write channel driver
59 * File Access Property List.
60 * The driver feature flags must include H5FD_FEAT_DEFAULT_VFD_COMPAITBLE.
61 * Must be set to H5P_DEFAULT or a valid FAPL ID.
62 *
63 * wo_file_path (char[H5FD_SPLITTER_PATH_MAX + 1])
64 * String buffer for the Write-Only channel target file.
65 * Must be null-terminated, cannot be empty.
66 *
67 * log_file_path (char[H5FD_SPLITTER_PATH_MAX + 1])
68 * String buffer for the Splitter VFD logging output.
69 * Must be null-terminated.
70 * If null, no logfile is created.
71 *
72 * ignore_wo_errors (hbool_t)
73 * Toggle flag for how judiciously to respond to errors on the Write-Only
74 * channel.
75 *
76 * ----------------------------------------------------------------------------
77 */
78typedef struct H5FD_splitter_vfd_config_t {
79 int32_t magic;
80 unsigned int version;
81 hid_t rw_fapl_id;
82 hid_t wo_fapl_id;
83 char wo_path[H5FD_SPLITTER_PATH_MAX + 1];
84 char log_file_path[H5FD_SPLITTER_PATH_MAX + 1];
85 hbool_t ignore_wo_errs;
86} H5FD_splitter_vfd_config_t;
