549 // Points derived from vtkPointSet.
550 // Attribute data (i.e., point and cell data (i.e., scalars, vectors, normals, tcoords)
551 // derived from vtkDataSet.
552
553 // The heart of the data representation. The points are managed by the
554 // superclass vtkPointSet. A cell is defined by its connectivity (i.e., the
555 // point ids that define the cell) and the cell type, represented by the
556 // Connectivity and Types arrays.
557 // Finally, when certain topological information is needed (e.g.,
558 // all the cells that use a point), the cell links array is built.
559 vtkSmartPointer<vtkCellArray> Connectivity;
560 vtkSmartPointer<vtkAbstractCellLinks> Links;
561 vtkSmartPointer<vtkUnsignedCharArray> Types;
