139 inline void rotateLeft(const T *src[],
140 T *dest[])
141 {
142 if ((src != NULL) && (dest != NULL))
143 {
144 Uint16 x;
145 Uint16 y;
146 const T *p;
147 T *q;
148 T *r;
149 const unsigned long count = OFstatic_cast(unsigned long, this->Dest_X) * OFstatic_cast(unsigned long, this->Dest_Y);
150 for (int j = 0; j < this->Planes; ++j)
151 {
152 p = src[j];
153 r = dest[j];
154 for (unsigned long f = this->Frames; f != 0; --f)
155 {
156 r += count;
157 for (x = this->Dest_X; x != 0; --x)
158 {
159 q = r - x;
160 for (y = this->Dest_Y; y != 0; --y)
161 {
162 *q = *p++;
163 q -= this->Dest_X;
164 }
165 }
166 }
167 }
168 }

319 inline void rotateTopDown(T *data[])
320 {
321 unsigned long i;
322 T *p;
323 T *q;
324 T t;
325 T *s;
326 const unsigned long count = OFstatic_cast(unsigned long, this->Dest_X) * OFstatic_cast(unsigned long, this->Dest_Y);
327 for (int j = 0; j < this->Planes; ++j)
328 {
329 s = data[j];
330 for (unsigned long f = this->Frames; f != 0; --f)
331 {
332 p = s;
333 q = s + count;
334 for (i = count / 2; i != 0; --i)
335 {
336 t = *p;
337 *p++ = *--q;
338 *q = t;
339 }
340 s += count;
341 }
342 }
