440 // The get_field and put_field functions are just a wrapper around the
441 // pure virtual get_field_internal and put_field_internal functions,
442 // but this lets me add some debug/checking/common code to the
443 // functions without having to do it in the calling code or in the
444 // derived classes code. This also fulfills the heuristic that a
445 // public interface should not contain pure virtual functions.
446 template <typename T>
447 int64_t get_field(const T *reg, const Field &field, void *data, size_t data_size) const
448 {
449 IOSS_FUNC_ENTER(m_);
450 verify_and_log(reg, field, 1);
451 int64_t retval = get_field_internal(reg, field, data, data_size);
452 if (get_nan_detection()) {
453 verify_field_data(reg, field, Ioss::Field::InOut::INPUT, data);
454 }
455 verify_and_log(nullptr, field, 1);
456 return retval;
457 }
458
459 template <typename T>
460 int64_t put_field(const T *reg, const Field &field, void *data, size_t data_size) const
461 {
462 IOSS_FUNC_ENTER(m_);
463 verify_and_log(reg, field, 0);
464 if (get_nan_detection()) {
465 verify_field_data(reg, field, Ioss::Field::InOut::OUTPUT, data);
466 }
467 int64_t retval = put_field_internal(reg, field, data, data_size);
468 verify_and_log(nullptr, field, 0);
469 return retval;
470 }
471
472 template <typename T>
473 int64_t get_zc_field(const T *reg, const Field &field, void **data, size_t *data_size) const
474 {
475 IOSS_FUNC_ENTER(m_);
476 verify_and_log(reg, field, 1);
477 int64_t retval = get_zc_field_internal(reg, field, data, data_size);
478 if (get_nan_detection()) {
479 verify_field_data(reg, field, Ioss::Field::InOut::INPUT, data);
480 }
481 verify_and_log(nullptr, field, 1);
482 return retval;
483 }
