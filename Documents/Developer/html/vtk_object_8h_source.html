297 template <class T>
298 class vtkClassMemberHandlerPointer
299 {
300 public:
301 void operator=(vtkObjectBase* o)
302 {
303 // The cast is needed in case "o" has multi-inheritance,
304 // to offset the pointer to get the vtkObjectBase.
305 if ((this->VoidPointer = dynamic_cast<T*>(o)) == nullptr)
306 {
307 // fallback to just using its vtkObjectBase as-is.
308 this->VoidPointer = o;
309 }
310 this->WeakPointer = o;
311 this->UseWeakPointer = true;
312 }
313 void operator=(void* o)
314 {
315 this->VoidPointer = o;
316 this->WeakPointer = nullptr;
317 this->UseWeakPointer = false;
318 }
319 T* GetPointer()
320 {
321 if (this->UseWeakPointer && !this->WeakPointer.GetPointer())
322 {
323 return nullptr;
324 }
325 return static_cast<T*>(this->VoidPointer);
326 }
327
328 private:
329 vtkWeakPointerBase WeakPointer;
330 void* VoidPointer;
331 bool UseWeakPointer;
332 };

339 template <class T>
340 class vtkClassMemberCallback : public vtkClassMemberCallbackBase
341 {
342 vtkClassMemberHandlerPointer<T> Handler;
343 void (T::*Method1)();
344 void (T::*Method2)(vtkObject*, unsigned long, void*);
345 bool (T::*Method3)(vtkObject*, unsigned long, void*);
346
347 public:
348 vtkClassMemberCallback(T* handler, void (T::*method)())
349 {
350 this->Handler = handler;
351 this->Method1 = method;
352 this->Method2 = nullptr;
353 this->Method3 = nullptr;
354 }
355
356 vtkClassMemberCallback(T* handler, void (T::*method)(vtkObject*, unsigned long, void*))
357 {
358 this->Handler = handler;
359 this->Method1 = nullptr;
360 this->Method2 = method;
361 this->Method3 = nullptr;
362 }
363
364 vtkClassMemberCallback(T* handler, bool (T::*method)(vtkObject*, unsigned long, void*))
365 {
366 this->Handler = handler;
367 this->Method1 = nullptr;
368 this->Method2 = nullptr;
369 this->Method3 = method;
370 }
371 ~vtkClassMemberCallback() override = default;
372
373 // Called when the event is invoked
374 bool operator()(vtkObject* caller, unsigned long event, void* calldata) override
375 {
376 T* handler = this->Handler.GetPointer();
377 if (handler)
378 {
379 if (this->Method1)
380 {
381 (handler->*this->Method1)();
382 }
383 else if (this->Method2)
384 {
385 (handler->*this->Method2)(caller, event, calldata);
386 }
387 else if (this->Method3)
388 {
389 return (handler->*this->Method3)(caller, event, calldata);
390 }
391 }
392 return false;
393 }
394 };
