24template <typename T>
25int RegressionTestWithImageName(
26 int argc, char* argv[], T* rw, const std::string& img, double thresh = 10.)
27{
28 vtkNew<vtkTesting> testing;
29 bool isImgPath = false;
30 for (int i = 0; i < argc; ++i)
31 {
32 if (!strcmp(argv[i], "-V"))
33 {
34 isImgPath = true;
35 }
36 else if (isImgPath)
37 {
38 isImgPath = false;
39 std::vector<std::string> components;
40 std::string originalImage = argv[i];
41 vtksys::SystemTools::SplitPath(originalImage, components);
42 // Substitute image filename for last component;
43 components.back() = img;
44 std::string tryme = vtksys::SystemTools::JoinPath(components);
45 testing->AddArgument(tryme.c_str());
46 continue;
47 }
48
49 testing->AddArgument(argv[i]);
50 }
51
52 if (testing->IsInteractiveModeSpecified())
53 {
54 return vtkTesting::DO_INTERACTOR;
55 }
56
57 if (testing->IsValidImageSpecified())
58 {
59 testing->SetRenderWindow(rw);
60 return testing->RegressionTest(thresh, cout);
61 }
62
63 return vtkTesting::NOT_RUN;
64}
