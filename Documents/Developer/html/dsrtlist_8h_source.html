104 OFBool operator==(const DSRListOfItems<T> &lst) const
105 {
106 /* first check whether the size of both lists is equal */
107 OFBool result = (ItemList.size() == lst.ItemList.size());
108 /* then iterate over all list entries (if any) */
109 if (result && !ItemList.empty())
110 {
111 /* since OFList does not implement a comparison operator we need the following */
112 const OFLIST_TYPENAME OFListConstIterator(T) endPos = ItemList.end();
113 OFLIST_TYPENAME OFListConstIterator(T) iterator = ItemList.begin();
114 OFLIST_TYPENAME OFListConstIterator(T) lstIter = lst.ItemList.begin();

116 result = (*iterator == *lstIter);

118 lstIter++;
119 } while (result && (iterator != endPos));
120 }
