587{

589 (gridIdx < static_cast<int>(this->NumberOfGrids)));
590 assert("pre: Neighbor grid index is out-of-bounds!" && (NeighborGridIdx >= 0) &&
591 (NeighborGridIdx < static_cast<int>(this->NumberOfGrids)));

594 assert("pre: Neighboring grid pair does not exist in hash!" &&

601//------------------------------------------------------------------------------
602inline void vtkStructuredGridConnectivity::GetGhostedExtent(
603 int* ghostedExtent, int GridExtent[6], int minIdx, int maxIdx, int N)
604{
605 assert("pre: Number of ghost layers must be N >= 1" && (N >= 1));
606 assert("pre: ghosted extent pointer is nullptr" && ghostedExtent != nullptr);
607
608 ghostedExtent[minIdx] = GridExtent[minIdx] - N;
609 ghostedExtent[maxIdx] = GridExtent[maxIdx] + N;
610
611 // Clamp the ghosted extent to be within the WholeExtent
612 ghostedExtent[minIdx] = (ghostedExtent[minIdx] < this->WholeExtent[minIdx])
613 ? this->WholeExtent[minIdx]
614 : ghostedExtent[minIdx];
615 ghostedExtent[maxIdx] = (ghostedExtent[maxIdx] > this->WholeExtent[maxIdx])
616 ? this->WholeExtent[maxIdx]
617 : ghostedExtent[maxIdx];
618}

789//------------------------------------------------------------------------------
790inline void vtkStructuredGridConnectivity::DetermineNeighborOrientation(
791 int idx, int A[2], int B[2], int overlap[2], int orient[3])
792{
793 assert("pre: idx is out-of-bounds" && (idx >= 0) && (idx < 3));
794
795 // A. Non-overlapping cases
796 if (overlap[0] == overlap[1])
797 {
798 if (A[1] == B[0])
799 {
800 orient[idx] = vtkStructuredNeighbor::HI;
801 }
802 else if (A[0] == B[1])
803 {
804 orient[idx] = vtkStructuredNeighbor::LO;
805 }
806 else
807 {
808 orient[idx] = vtkStructuredNeighbor::UNDEFINED;
809 assert("ERROR: Code should not reach here!" && false);
810 }
811 } // END non-overlapping cases
812 // B. Sub-set cases
813 else if (this->IsSubset(A, B))
814 {
815 if ((A[0] == B[0]) && (A[1] == B[1]))
816 {
817 orient[idx] = vtkStructuredNeighbor::ONE_TO_ONE;
818 }
819 else if (this->StrictlyInsideBounds(A[0], B[0], B[1]) &&
820 this->StrictlyInsideBounds(A[1], B[0], B[1]))
821 {
822 orient[idx] = vtkStructuredNeighbor::SUBSET_BOTH;
823 }
824 else if (A[0] == B[0])
825 {
826 orient[idx] = vtkStructuredNeighbor::SUBSET_HI;
827 }
828 else if (A[1] == B[1])
829 {
830 orient[idx] = vtkStructuredNeighbor::SUBSET_LO;
831 }
832 else
833 {
834 orient[idx] = vtkStructuredNeighbor::UNDEFINED;
835 assert("ERROR: Code should not reach here!" && false);
836 }
837 }
838 // C. Super-set cases
839 else if (this->IsSubset(B, A))
840 {
841 orient[idx] = vtkStructuredNeighbor::SUPERSET;
842 }
843 // D. Partially-overlapping (non-subset) cases
844 else if (!(this->IsSubset(A, B) || this->IsSubset(A, B)))
845 {
846 if (this->InBounds(A[0], B[0], B[1]))
847 {
848 orient[idx] = vtkStructuredNeighbor::LO;
849 }
850 else if (this->InBounds(A[1], B[0], B[1]))
851 {
852 orient[idx] = vtkStructuredNeighbor::HI;
853 }
854 else
855 {
856 orient[idx] = vtkStructuredNeighbor::UNDEFINED;
857 assert("ERROR: Code should not reach here!" && false);
858 }
859 }
860 else
861 {
862 orient[idx] = vtkStructuredNeighbor::UNDEFINED;
863 assert("ERROR: Code should not reach here!" && false);
864 }
865}
