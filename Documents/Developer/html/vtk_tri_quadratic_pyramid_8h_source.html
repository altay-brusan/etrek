206// Return the center of the tri-quadratic pyramid in parametric coordinates.
207//
208inline int vtkTriQuadraticPyramid::GetParametricCenter(double pcoords[3])
209{
210 pcoords[0] = pcoords[1] = 0.5;
211 // This is different compared to the last node, because the last node
212 // is the centroid of the nodes 0-4, and not the centroid of the nodes 0-17.
213 // So pcoords[2] is defined as followed to pass the requirement of TestGenericCell
214 pcoords[2] = 283.0 / 456.0;
215 return 0;
216}
