42template <class A>
43A SafeDivision(A a, A b)
44{
45 // Avoid overflow
46 if ((b < static_cast<A>(1)) && (a > b * std::numeric_limits<A>::max()))
47 {
48 return std::numeric_limits<A>::max();
49 }
50
51 // Avoid underflow
52 if ((a == static_cast<A>(0)) ||
53 ((b > static_cast<A>(1)) && (a < b * std::numeric_limits<A>::min())))
54 {
55 return static_cast<A>(0);
56 }
57
58 // safe to do the division
59 return (a / b);
60}

67template <class A>
68bool NearlyEqual(A a, A b, A tol = std::numeric_limits<A>::epsilon())
69{
70 A absdiff = fabs(a - b);
71 A d1 = vtkMathUtilities::SafeDivision<A>(absdiff, fabs(a));
72 A d2 = vtkMathUtilities::SafeDivision<A>(absdiff, fabs(b));
73
74 return ((d1 <= tol) || (d2 <= tol));
75}

100template <class A>
101void UpdateRangeImpl(A& min0, A& max0, const A& value)
102{
103 // need temporaries to handle const/non const ref mismatch
104 if (value < min0)
105 {
106 min0 = value;
107 max0 = max0 < value ? value : max0;
108 }
109 else if (value > max0)
110 {
111 min0 = min0 > value ? value : min0;
112 max0 = value;
113 }
114}

116template <class A> // Non floating point implementation not caring about NaN
117void UpdateRange(A& min0, A& max0, const A& value,
118 typename std::enable_if<!std::is_floating_point<A>::value>::type* = nullptr)
119{
120 UpdateRangeImpl<A>(min0, max0, value);
121}

123template <class A> // Floating point implementation specifically considering NaN
124void UpdateRange(A& min0, A& max0, const A& value,
125 typename std::enable_if<std::is_floating_point<A>::value>::type* = nullptr)
126{
127 if (!std::isnan(value))
128 {
129 UpdateRangeImpl<A>(min0, max0, value);
130 }
131}
