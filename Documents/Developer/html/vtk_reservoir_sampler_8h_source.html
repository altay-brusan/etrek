58 const std::vector<Integer>& operator()(Integer kk, vtkAbstractArray* array) const
59 {
60 VTK_THREAD_LOCAL static std::vector<Integer> data;
61 if (!array)
62 {
63 throw std::invalid_argument("Null arrays are disallowed.");
64 }
65 if (array->GetNumberOfTuples() > std::numeric_limits<Integer>::max())
66 {
67 throw std::invalid_argument("Array size would overflow integer type.");
68 }
69 this->GenerateSample(kk, array->GetNumberOfTuples(), data);
70 return data;
71 }
72
73protected:
74 void GenerateSample(Integer kk, Integer nn, std::vector<Integer>& data) const
75 {
76 if (nn < kk)
77 {
78 kk = nn;
79 }
80 if (kk < 0)
81 {
82 throw std::invalid_argument(
83 "You must choose a non-negative number of values from a proper sequence.");
84 }
85 data.resize(kk);
86 if (kk == 0)
87 {
88 return;
89 }
90 // I. Fill the output with the first kk values.
91 Integer ii;
92 for (ii = 0; ii < kk; ++ii)
93 {
94 data[ii] = ii;
95 }
96 if (kk == nn)
97 {
98 return;
99 }
100
101 std::mt19937 generator(vtkReservoirSampler::RandomSeed());
102 std::uniform_real_distribution<> unitUniform(0., 1.);
103 std::uniform_int_distribution<Integer> randomIndex(0, kk - 1);
104 double w = exp(log(unitUniform(generator)) / kk);
105
106 while (true)
107 {
108 double delta = floor(log(unitUniform(generator)) / log(1.0 - w)) + 1.0;
109 if (delta < 0.0 || delta > static_cast<double>(vtkTypeTraits<Integer>::Max()))
110 {
111 // If delta overflows the size of the integer, we are done.
112 break;
113 }
114 Integer intDelta = static_cast<Integer>(delta);
115 // Be careful here since delta may be large and nn may be
116 // at or near numeric_limits<Integer>::max().
117 if (nn - ii > intDelta)
118 {
119 Integer jj = randomIndex(generator);
120#if 0
121 std::cout << " i " << ii << " δ " << intDelta << " w " << w << " → j " << jj << "\n";
122#endif
123 ii += intDelta;
124 data[jj] = ii;
125 w *= exp(log(unitUniform(generator)) / kk);
126 }
127 else
128 {
129 // Adding delta to ii would step beyond our sequence size,
130 // so we are done.
131 break;
132 }
133 }
134 if (Monotonic)
135 {
136 std::sort(data.begin(), data.end());
137 }
138 }
139};
