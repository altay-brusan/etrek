27 // These static asserts only fire when the function calling CheckTypes is
28 // used. Thus, this smart pointer class may still be used as a member variable
29 // with a forward declared T, so long as T is defined by the time the calling
30 // function is used.
31 template <typename U = T>
32 static void CheckTypes() noexcept
33 {
34 static_assert(vtk::detail::IsComplete<T>::value,
35 "vtkSmartPointer<T>'s T type has not been defined. Missing "
36 "include?");
37 static_assert(vtk::detail::IsComplete<U>::value,
38 "Cannot store an object with undefined type in "
39 "vtkSmartPointer. Missing include?");
40 static_assert(std::is_base_of<T, U>::value,
41 "Argument type is not compatible with vtkSmartPointer<T>'s "
42 "T type.");
43 static_assert(std::is_base_of<vtkObjectBase, T>::value,
44 "vtkSmartPointer can only be used with subclasses of "
45 "vtkObjectBase.");
46 }

257 // Work-around for HP and IBM overload resolution bug. Since
258 // NullPointerOnly is a private type the only pointer value that can
259 // be passed by user code is a null pointer. This operator will be
260 // chosen by the compiler when comparing against null explicitly and
261 // avoid the bogus ambiguous overload error.
262#if defined(__HP_aCC) || defined(__IBMCPP__)
263#define VTK_SMART_POINTER_DEFINE_OPERATOR_WORKAROUND(op) \
264 bool operator op(NullPointerOnly*) const { return ::operator op(*this, 0); }
265
266private:
267 class NullPointerOnly
268 {
269 };

275 VTK_SMART_POINTER_DEFINE_OPERATOR_WORKAROUND(<=)
276 VTK_SMART_POINTER_DEFINE_OPERATOR_WORKAROUND(>)
277 VTK_SMART_POINTER_DEFINE_OPERATOR_WORKAROUND(>=)
278#undef VTK_SMART_POINTER_DEFINE_OPERATOR_WORKAROUND
279#endif
