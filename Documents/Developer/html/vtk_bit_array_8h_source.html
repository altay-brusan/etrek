369inline void vtkBitArray::SetValue(vtkIdType id, int value)
370{
371 this->Array[id / 8] =
372 static_cast<unsigned char>((value != 0) ? (this->Array[id / 8] | (0x80 >> id % 8))
373 : (this->Array[id / 8] & (~(0x80 >> id % 8))));
374 this->DataChanged();
375}
376
377inline void vtkBitArray::InsertValue(vtkIdType id, int i)
378{
379 if (id >= this->Size)
380 {
381 if (!this->ResizeAndExtend(id + 1))
382 {
383 return;
384 }
385 }
386 this->Array[id / 8] =
387 static_cast<unsigned char>((i != 0) ? (this->Array[id / 8] | (0x80 >> id % 8))
388 : (this->Array[id / 8] & (~(0x80 >> id % 8))));
389 if (id > this->MaxId)
390 {
391 this->MaxId = id;
392 this->InitializeUnusedBitsInLastByte();
393 }
394 this->DataChanged();
395}
