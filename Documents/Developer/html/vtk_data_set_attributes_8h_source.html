116 // ----------- ghost points and ghost cells -------------------------------------------
117 // The following bit fields are consistent with VisIt ghost zones specification
118 // For details, see http://www.visitusers.org/index.php?title=Representing_ghost_data
119
120 enum CellGhostTypes
121 {
122 DUPLICATECELL = 1, // the cell is present on multiple processors
123 HIGHCONNECTIVITYCELL = 2, // the cell has more neighbors than in a regular mesh
124 LOWCONNECTIVITYCELL = 4, // the cell has less neighbors than in a regular mesh
125 REFINEDCELL = 8, // other cells are present that refines it.
126 EXTERIORCELL = 16, // the cell is on the exterior of the data set
127 HIDDENCELL =
128 32 // the cell is needed to maintain connectivity, but the data values should be ignored.
129 };
130
131 enum PointGhostTypes
132 {
133 DUPLICATEPOINT = 1, // the cell is present on multiple processors
134 HIDDENPOINT =
135 2 // the point is needed to maintain connectivity, but the data values should be ignored.
136 };
137
138 // A vtkDataArray with this name must be of type vtkUnsignedCharArray.
139 // Each value must be assigned according to the bit fields described in
140 // PointGhostTypes or CellGhostType
141 static const char* GhostArrayName() { return "vtkGhostType"; }
