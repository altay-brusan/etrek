338inline void vtkContourFilter::SetContourValues(const std::vector<double>& values)
339{
340 int numContours = static_cast<int>(values.size());
341 this->SetNumberOfContours(numContours);
342 for (int i = 0; i < numContours; i++)
343 {
344 this->SetValue(i, values[i]);
345 }
346}

353inline std::vector<double> vtkContourFilter::GetContourValues()
354{
355 std::vector<double> contours;
356 int numContours = this->GetNumberOfContours();
357 contours.reserve(numContours);
358 for (int i = 0; i < numContours; i++)
359 {
360 contours.push_back(this->GetValue(i));
361 }
362 return contours;
363}
