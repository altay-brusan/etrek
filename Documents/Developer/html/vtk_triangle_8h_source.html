228inline int vtkTriangle::GetParametricCenter(double pcoords[3])
229{
230 pcoords[0] = pcoords[1] = 1.0 / 3.0;
231 pcoords[2] = 0.0;

233}

235//----------------------------------------------------------------------------
236inline void vtkTriangle::ComputeNormalDirection(
237 const double v1[3], const double v2[3], const double v3[3], double n[3])
238{
239 // order is important!!! maintain consistency with triangle vertex order
240 double ax = v3[0] - v2[0];
241 double ay = v3[1] - v2[1];
242 double az = v3[2] - v2[2];
243 double bx = v1[0] - v2[0];
244 double by = v1[1] - v2[1];
245 double bz = v1[2] - v2[2];
246
247 n[0] = (ay * bz - az * by);
248 n[1] = (az * bx - ax * bz);
249 n[2] = (ax * by - ay * bx);
250}
251
252//----------------------------------------------------------------------------
253inline void vtkTriangle::ComputeNormal(
254 const double v1[3], const double v2[3], const double v3[3], double n[3])
255{
256 vtkTriangle::ComputeNormalDirection(v1, v2, v3, n);
257
258 double length = sqrt(n[0] * n[0] + n[1] * n[1] + n[2] * n[2]);
259 if (length != 0.0)
260 {
261 n[0] /= length;
262 n[1] /= length;
263 n[2] /= length;
264 }
265}

267//----------------------------------------------------------------------------
268inline void vtkTriangle::TriangleCenter(
269 const double p1[3], const double p2[3], const double p3[3], double center[3])
270{
271 center[0] = (p1[0] + p2[0] + p3[0]) / 3.0;
272 center[1] = (p1[1] + p2[1] + p3[1]) / 3.0;
273 center[2] = (p1[2] + p2[2] + p3[2]) / 3.0;
274}
275
276//----------------------------------------------------------------------------
277inline double vtkTriangle::TriangleArea(const double p1[3], const double p2[3], const double p3[3])
278{
279 double n[3];
280 vtkTriangle::ComputeNormalDirection(p1, p2, p3, n);
281
282 return 0.5 * vtkMath::Norm(n);
283}
