101 virtual OFBool operator==( const OFOrderedSet<T> &other ) const
102 {
103 // check if both sets contain the same
104 // amount of items; if not, return OFFalse
105 if( this->num != other.num )
106 return( OFFalse );
107
108 // initialize result with OFTrue
109 OFBool result = OFTrue;
110
111 // as long as result is OFTrue go through all items in this
112 for( unsigned int i=0 ; i < this->num && result == OFTrue ; i++ )
113 {
114 // in case the current element does not equal the current
115 // element in other, result shall be set to OFFalse
116 if( *this->items[i] != *other.items[i] )
117 result = OFFalse;
118 }
119
120 // return result
121 return( result );
122 }

173 virtual void InsertAt( const T &item, unsigned int idx )
174 {
175 unsigned int i;
176
177 // in case index is greater than the index of the last item,
178 // insert the new item right behind the last item of the set
179 if( idx > this->num - 1 )
180 Insert( item );
181 else
182 {
183 // if size equals num, we need more space
184 if( this->size == this->num )
185 this->Resize( this->size * 2 );
186
187 // copy item
188 T *newItem = new T( item );
189
190 // create a new temporary array and assign all pointers correspondingly
191 T **tmp = new T*[this->size];
192
193 for( i=0 ; i<idx ; i++ )
194 tmp[i] = this->items[i];
195
196 tmp[idx] = newItem;
197
198 for( i=idx ; i < this->size - 1 ; i++ )
199 {
200 if( i < this->num )
201 tmp[i+1] = this->items[i];
202 else
203 tmp[i+1] = NULL;
204 }
205
206 // delete old array
207 delete[] this->items;
208
209 // assign new array to member variable
210 this->items = tmp;
211
212 // increase counter
213 this->num++;
214 }
215 }

221 virtual void Remove( const T &item )
222 {
223 // so far, nothing was deleted
224 OFBool itemDeleted = OFFalse;
225
226 // go through all items
227 for( unsigned int i=0 ; i < this->num && !itemDeleted ; i++ )
228 {
229 // if current item is the one which shall be deleted
230 if( *this->items[i] == item )
231 {
232 // delete item
233 delete this->items[i];
234
235 // and - so that there are no holes in the array - move all elements
236 // behind the current element up one array field; only do so in case
237 // we did _not_ delete the last item
238 if( i != this->num - 1 )
239 {
240 unsigned int j;
241 for( j=i+1 ; j < this->num ; j++ )
242 {
243 this->items[j-1] = this->items[j];
244 }
245
246 this->items[j-1] = NULL;
247 }
248 else
249 this->items[i] = NULL;
250
251 // reduce counter
252 this->num--;
253
254 // remember that an item was deleted (so that always only one item will be deleted)
255 itemDeleted = OFTrue;
256 }
257 }

342 virtual OFBool IsSupersetOf( const OFOrderedSet<T> &other ) const
343 {
344 // if this contains less or the same amount of items than other, return OFFalse
345 if( this->num <= other.num )
346 return( OFFalse );
347
348 // initialize result with OFTrue
349 OFBool result = OFTrue;
350
351 // make a copy of this
352 OFOrderedSet<T> s = *this;
353
354 // as long as result is OFTrue go through all items in other
355 for( unsigned int i=0 ; i<other.num && result == OFTrue ; i++ )
356 {
357 // in case s contains the current item of other
358 if( s.Contains( *other.items[i] ) )
359 {
360 // remove this item from s so that it will not be
361 // considered again in a later call to s.Contains()
362 s.Remove( *other.items[i] );
363 }
364 // in case s does not contain the current item of other the result is OFFalse
365 else
366 result = OFFalse;
367 }
368
369 // return result
370 return( result );
371 }

445 OFOrderedSet<T> Difference( const OFOrderedSet<T> &other ) const
446 {
447 // initialize result set
448 OFOrderedSet<T> resultSet;
449
450 // make a copy of other
451 OFOrderedSet<T> s = other;
452
453 // go through all items in this
454 for( unsigned int i=0 ; i < this->num ; i++ )
455 {
456 // if s does not contain the current item
457 if( !s.Contains( *this->items[i] ) )
458 {
459 // insert the item into the result set
460 resultSet.Insert( *this->items[i] );
461 }
462 else
463 {
464 // else remove the item from s so that it will not be
465 // considered again in a later call to s.Contains()
466 s.Remove( *this->items[i] );
467 }
468 }
469
470 // return result set
471 return( resultSet );

482 OFOrderedSet<T> SymmetricDifference( const OFOrderedSet<T> &other ) const
483 {
484 // determine s1 = this - other
485 OFOrderedSet<T> s1 = (*this).Difference( other );
486
487 // determine s2 = other - this
488 OFOrderedSet<T> s2 = other.Difference( *this );
489
490 // determine the union of s1 and s2
491 OFOrderedSet<T> resultSet = s1.Union( s2 );
492
493 // return result set
