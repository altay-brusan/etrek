412// This is a helper class hidden from the users.
413// It helps implementing the post increment operator of the actual directory
414// iterator, which, since it is an input iterator, cannot return a copy of
415// itself as a forward iterator would do, but instead only copies the 'value'
416// (the OFdirectory_entry part) of the enclosing iterator.
417class DCMTK_OFSTD_EXPORT OFdirectory_iterator_proxy
418{
419public:
420 inline OFdirectory_iterator_proxy(OFrvalue_ref(OFdirectory_iterator_proxy) rhs)
421 : m_Entry( OFmove( OFrvalue_access( rhs ).m_Entry ) ) {}
422 // add operators '->' and '*' to access the 'value', acting like an
423 // iterator would although this already is the actual entry object
424 inline const OFdirectory_entry* operator->() const { return &m_Entry; }
425 inline const OFdirectory_entry& operator*() const { return m_Entry; }
426
427private:
428 // only a directory iterator may create instances of this object, others
429 // shouldn't even know it exists
430 friend class OFdirectory_iterator;
431 OFdirectory_iterator_proxy(const OFdirectory_entry&);
432 OFdirectory_entry m_Entry;
