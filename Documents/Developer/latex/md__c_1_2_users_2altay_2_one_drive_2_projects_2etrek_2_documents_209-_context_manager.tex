It enables decoupled communication between different parts of the application without creating tight dependencies.\hypertarget{md__c_1_2_users_2altay_2_one_drive_2_projects_2etrek_2_documents_209-_context_manager_autotoc_md588}{}\doxysection{\texorpdfstring{Architecture}{Architecture}}\label{md__c_1_2_users_2altay_2_one_drive_2_projects_2etrek_2_documents_209-_context_manager_autotoc_md588}

\doxylink{requirements_The}{The} system supports two main categories of context:\+


\begin{DoxyEnumerate}
\item {\bfseries{Session Context}} ({\ttfamily ISession\+Context})
\begin{DoxyItemize}
\item Persists for the duration of a user session
\item Contains:\+ logged-\/in user, sign-\/in time, workstation name, institution info
\item Created after successful authentication
\end{DoxyItemize}
\item {\bfseries{Workflow Context}} ({\ttfamily IWorkflow\+Context})
\begin{DoxyItemize}
\item Transient state for specific workflows
\item Example:\+ {\ttfamily Examination\+Context} for Worklist → Examination flow
\item Created when user initiates a workflow (e.\+g., double-\/clicks worklist item)
\end{DoxyItemize}
