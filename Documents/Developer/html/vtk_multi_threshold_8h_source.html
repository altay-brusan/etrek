246 class NormKey
247 {
248 public:
249 int Association; // vtkDataObject::FIELD_ASSOCIATION_POINTS or
250 // vtkDataObject::FIELD_ASSOCIATION_CELLS
251 int Type; // -1 => use Name, otherwise: vtkDataSetAttributes::{SCALARS, VECTORS, TENSORS,
252 // NORMALS, TCOORDS, GLOBALIDS}
253 std::string Name; // Either empty or (when ArrayType == -1) an input array name
254 int Component; // LINFINITY_NORM, L1_NORM, L2_NORM or an integer component number
255 int AllScalars; // For Association == vtkDataObject::FIELD_ASSOCIATION_POINTS, must all points
256 // be in the interval?
257 int InputArrayIndex; // The number passed to vtkAlgorithm::SetInputArrayToProcess()
258 TupleNorm NormFunction; // A function pointer to compute the norm (or fetcht the correct
259 // component) of a tuple.
260
263 void ComputeNorm(
264 vtkIdType cellId, vtkCell* cell, vtkDataArray* array, double cellNorm[2]) const;
265
268 bool operator<(const NormKey& other) const
269 {
270 if (this->Association < other.Association)
271 return true;
272 else if (this->Association > other.Association)
273 return false;
274
275 if (this->Component < other.Component)
276 return true;
277 else if (this->Component > other.Component)
278 return false;
279
280 if ((!this->AllScalars) && other.AllScalars)
281 return true;
282 else if (this->AllScalars && (!other.AllScalars))
283 return false;
284
285 if (this->Type == -1)
286 {
287 if (other.Type == -1)
288 return this->Name < other.Name;
289 return true;
290 }
291 else
292 {
293 return this->Type < other.Type;
294 }
295 }
296 };

485inline int vtkMultiThreshold::AddNotchIntervalSet(
486 double xlo, double xhi, int assoc, const char* arrayName, int component, int allScalars)
487{
488 int band =
489 this->AddIntervalSet(xlo, xhi, CLOSED, CLOSED, assoc, arrayName, component, allScalars);
490 if (band < 0)
491 {
492 return -1;
493 }
494 return this->AddBooleanSet(NAND, 1, &band);
