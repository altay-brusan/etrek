82 void convert(const T1 *pixel,
83 const unsigned long planeSize,
84 const int bits,
85 const OFBool rgb)
86 {
87 if (this->Init(pixel))
88 {
89 const T1 offset = OFstatic_cast(T1, DicomImageClass::maxval(bits - 1));
90 // use the number of input pixels derived from the length of the 'PixelData'
91 // attribute), but not more than the size of the intermediate buffer
92 const unsigned long count = (this->InputCount < this->Count) ? this->InputCount : this->Count;
93 if (rgb) /* convert to RGB model */
94 {
95 T2 *r = this->Data[0];
96 T2 *g = this->Data[1];
97 T2 *b = this->Data[2];
98 const T2 maxvalue = OFstatic_cast(T2, DicomImageClass::maxval(bits));
99 DiPixelRepresentationTemplate<T1> rep;
100 if (bits == 8 && !rep.isSigned()) // only for unsigned 8 bit
101 {
102 Sint16 rcr_tab[256];
103 Sint16 gcb_tab[256];
104 Sint16 gcr_tab[256];
105 Sint16 bcb_tab[256];
106 const double r_const = 0.7010 * OFstatic_cast(double, maxvalue);
107 const double g_const = 0.5291 * OFstatic_cast(double, maxvalue);
108 const double b_const = 0.8859 * OFstatic_cast(double, maxvalue);
109 unsigned long l;
110 for (l = 0; l < 256; ++l)
111 {
112 rcr_tab[l] = OFstatic_cast(Sint16, 1.4020 * OFstatic_cast(double, l) - r_const);
113 gcb_tab[l] = OFstatic_cast(Sint16, 0.3441 * OFstatic_cast(double, l));
114 gcr_tab[l] = OFstatic_cast(Sint16, 0.7141 * OFstatic_cast(double, l) - g_const);
115 bcb_tab[l] = OFstatic_cast(Sint16, 1.7720 * OFstatic_cast(double, l) - b_const);
116 }
117 Sint32 sr;
118 Sint32 sg;
119 Sint32 sb;
120 if (this->PlanarConfiguration)
121 {
122/*
123 const T1 *y = pixel;
124 const T1 *cb = y + this->InputCount;
125 const T1 *cr = cb + this->InputCount;
126 for (i = count; i != 0; --i, ++y, ++cb, ++cr)
127 {
128 sr = OFstatic_cast(Sint32, *y) + OFstatic_cast(Sint32, rcr_tab[*cr]);
129 sg = OFstatic_cast(Sint32, *y) - OFstatic_cast(Sint32, gcb_tab[*cb]) - OFstatic_cast(Sint32, gcr_tab[*cr]);
130 sb = OFstatic_cast(Sint32, *y) + OFstatic_cast(Sint32, bcb_tab[*cb]);
131 *(r++) = (sr < 0) ? 0 : (sr > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sr);
132 *(g++) = (sg < 0) ? 0 : (sg > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sg);
133 *(b++) = (sb < 0) ? 0 : (sb > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sb);
134 }
135*/
136 const T1 *y = pixel;
137 const T1 *cb = y + planeSize;
138 const T1 *cr = cb + planeSize;
139 unsigned long i = count;
140 while (i != 0)
141 {
142 /* convert a single frame */
143 for (l = planeSize; (l != 0) && (i != 0); --l, --i, ++y, ++cb, ++cr)
144 {
145 sr = OFstatic_cast(Sint32, *y) + OFstatic_cast(Sint32, rcr_tab[OFstatic_cast(Uint32, *cr)]);
146 sg = OFstatic_cast(Sint32, *y) - OFstatic_cast(Sint32, gcb_tab[OFstatic_cast(Uint32, *cb)]) - OFstatic_cast(Sint32, gcr_tab[OFstatic_cast(Uint32, *cr)]);
147 sb = OFstatic_cast(Sint32, *y) + OFstatic_cast(Sint32, bcb_tab[OFstatic_cast(Uint32, *cb)]);
148 *(r++) = (sr < 0) ? 0 : (sr > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sr);
149 *(g++) = (sg < 0) ? 0 : (sg > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sg);
150 *(b++) = (sb < 0) ? 0 : (sb > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sb);
151 }
152 /* jump to next frame start (skip 2 planes) */
153 y += 2 * planeSize;
154 cb += 2 * planeSize;
155 cr += 2 * planeSize;
156 }
157 }
158 else
159 {
160 const T1 *p = pixel;
161 T1 y;
162 T1 cb;
163 T1 cr;
164 unsigned long i;
165 for (i = count; i != 0; --i)
166 {
167 y = *(p++);
168 cb = *(p++);
169 cr = *(p++);
170 sr = OFstatic_cast(Sint32, y) + OFstatic_cast(Sint32, rcr_tab[OFstatic_cast(Uint32, cr)]);
171 sg = OFstatic_cast(Sint32, y) - OFstatic_cast(Sint32, gcb_tab[OFstatic_cast(Uint32, cb)]) - OFstatic_cast(Sint32, gcr_tab[OFstatic_cast(Uint32, cr)]);
172 sb = OFstatic_cast(Sint32, y) + OFstatic_cast(Sint32, bcb_tab[OFstatic_cast(Uint32, cb)]);
173 *(r++) = (sr < 0) ? 0 : (sr > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sr);
174 *(g++) = (sg < 0) ? 0 : (sg > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sg);
175 *(b++) = (sb < 0) ? 0 : (sb > OFstatic_cast(Sint32, maxvalue)) ? maxvalue : OFstatic_cast(T2, sb);
176 }
177 }
178 }
179 else
180 {
181 if (this->PlanarConfiguration)
182 {
183/*
184 const T1 *y = pixel;
185 const T1 *cb = y + this->InputCount;
186 const T1 *cr = cb + this->InputCount;
187 for (i = count; i != 0; --i)
188 convertValue(*(r++), *(g++), *(b++), removeSign(*(y++), offset), removeSign(*(cb++), offset),
189 removeSign(*(cr++), offset), maxvalue);
190*/
191 unsigned long l;
192 unsigned long i = count;
193 const T1 *y = pixel;
194 const T1 *cb = y + planeSize;
195 const T1 *cr = cb + planeSize;
196 while (i != 0)
197 {
198 /* convert a single frame */
199 for (l = planeSize; (l != 0) && (i != 0); --l, --i)
200 {
201 convertValue(*(r++), *(g++), *(b++), removeSign(*(y++), offset), removeSign(*(cb++), offset),
202 removeSign(*(cr++), offset), maxvalue);
203 }
204 /* jump to next frame start (skip 2 planes) */
205 y += 2 * planeSize;
206 cb += 2 * planeSize;
207 cr += 2 * planeSize;
208 }
209 }
210 else
211 {
212 const T1 *p = pixel;
213 T2 y;
214 T2 cb;
215 T2 cr;
216 unsigned long i;
217 for (i = count; i != 0; --i)
218 {
219 y = removeSign(*(p++), offset);
220 cb = removeSign(*(p++), offset);
221 cr = removeSign(*(p++), offset);
222 convertValue(*(r++), *(g++), *(b++), y, cb, cr, maxvalue);
223 }
224 }
225 }
226 } else { /* retain YCbCr model */
227 const T1 *p = pixel;
228 if (this->PlanarConfiguration)
229 {
230/*
231 T2 *q;
232 // number of pixels to be skipped (only applicable if 'PixelData' contains more
233 // pixels than expected)
234 const unsigned long skip = (this->InputCount > this->Count) ? (this->InputCount - this->Count) : 0;
235 for (int j = 0; j < 3; ++j)
236 {
237 q = this->Data[j];
238 for (i = count; i != 0; --i)
239 *(q++) = removeSign(*(p++), offset);
240 // skip to beginning of next plane
241 p += skip;
242 }
243*/
244 unsigned long l;
245 unsigned long i = 0;
246 while (i < count)
247 {
248 /* store current pixel index */
249 const unsigned long iStart = i;
250 for (int j = 0; j < 3; ++j)
251 {
252 /* convert a single plane */
253 for (l = planeSize, i = iStart; (l != 0) && (i < count); --l, ++i)
254 this->Data[j][i] = removeSign(*(p++), offset);
255 }
256 }
257 }
258 else
259 {
260 int j;
261 unsigned long i;
262 for (i = 0; i < count; ++i) /* for all pixel ... */
263 for (j = 0; j < 3; ++j)
264 this->Data[j][i] = removeSign(*(p++), offset); /* ... copy planes */
265 }
266 }
267 }
268 }

279 inline void convertValue(T2 &red, T2 &green, T2 &blue, const T2 y, const T2 cb, const T2 cr, const T2 maxvalue)
280 {
281 double dr = OFstatic_cast(double, y) + 1.4020 * OFstatic_cast(double, cr) - 0.7010 * OFstatic_cast(double, maxvalue);
282 double dg = OFstatic_cast(double, y) - 0.3441 * OFstatic_cast(double, cb) - 0.7141 * OFstatic_cast(double, cr) + 0.5291 * OFstatic_cast(double, maxvalue);
283 double db = OFstatic_cast(double, y) + 1.7720 * OFstatic_cast(double, cb) - 0.8859 * OFstatic_cast(double, maxvalue);
284 red = (dr < 0.0) ? 0 : (dr > OFstatic_cast(double, maxvalue)) ? maxvalue : OFstatic_cast(T2, dr);
285 green = (dg < 0.0) ? 0 : (dg > OFstatic_cast(double, maxvalue)) ? maxvalue : OFstatic_cast(T2, dg);
286 blue = (db < 0.0) ? 0 : (db > OFstatic_cast(double, maxvalue)) ? maxvalue : OFstatic_cast(T2, db);
287 }
288};
