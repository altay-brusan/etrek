15 using OutgoingQueues = std::map<BlockID, MemoryBuffer>;

17 Proxy(Master* master__, int gid__,
18 IExchangeInfo* iexchange__ = 0):
19 gid_(gid__),
20 master_(master__),
21 iexchange_(iexchange__),
22 collectives_(&master__->collectives(gid__))
23 {
24 fill_incoming();
25
26 // move outgoing_ back into proxy, in case it's a multi-foreach round
27 if (!iexchange_)
28 for (auto& x : master_->outgoing(gid_))
29 {
30 auto access = x.second.access();
31 if (!access->empty())
32 {
33 outgoing_.emplace(x.first, access->back().move());
34 access->pop_back();
35 }
36 }
37 }
38
39 // delete copy constructor to avoid coping incoming_ and outgoing_ (plus it
40 // won't work otherwise because MemoryBuffer has a deleted copy
41 // constructor)
42 Proxy(const Proxy&) =delete;
43 Proxy(Proxy&&) =default;
44 Proxy& operator=(const Proxy&) =delete;
45 Proxy& operator=(Proxy&&) =default;
46
47 ~Proxy()
48 {
49 auto& outgoing = master_->outgoing(gid_);
50 auto& incoming = master_->incoming(gid_);
51
52 // copy out outgoing_
53 for (auto& x : outgoing_)
54 {
55 outgoing[x.first].access()->emplace_back(std::move(x.second));
56 if (iexchange_)
57 iexchange_->inc_work();
58 }
59
60 // move incoming_ back into master, in case it's a multi-foreach round
61 if (!iexchange_)
62 for (auto& x : incoming_)
63 incoming[x.first].access()->emplace_front(std::move(x.second));
64 }
65
66 void init()
67 {
68 collectives_ = &master()->collectives(gid());
69 }
70
71 int gid() const { return gid_; }
72
73 bool fill_incoming() const
74 {
75 bool exists = false;
76
77 incoming_.clear();
78
79 // fill incoming_
80 for (auto& x : master_->incoming(gid_))
81 {
82 auto access = x.second.access();
83 if (!access->empty())
84 {
85 exists = true;
86 incoming_.emplace(x.first, access->front().move());
87 access->pop_front();
88 if (iexchange_)
89 iexchange_->dec_work();
90 }
91 }
92
93 return exists;
94 }
