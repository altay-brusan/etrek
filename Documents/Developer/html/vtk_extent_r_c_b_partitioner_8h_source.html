124 void GetGhostedExtent(int ext[6], int minIdx, int maxIdx)
125 {
126 ext[minIdx] -= this->NumberOfGhostLayers;
127 ext[maxIdx] += this->NumberOfGhostLayers;
128 ext[minIdx] =
129 (ext[minIdx] < this->GlobalExtent[minIdx]) ? this->GlobalExtent[minIdx] : ext[minIdx];
130 ext[maxIdx] =
131 (ext[maxIdx] > this->GlobalExtent[maxIdx]) ? this->GlobalExtent[maxIdx] : ext[maxIdx];
132 }
