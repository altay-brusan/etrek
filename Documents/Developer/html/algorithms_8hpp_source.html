27 template<class Block, class T, class Cmp>
28 void sort(Master& master,
29 const Assigner& assigner,
30 std::vector<T> Block::* values,
31 std::vector<T> Block::* samples,
32 size_t num_samples,
33 const Cmp& cmp,
34 int k = 2,
35 bool samples_only = false)
36 {
37 bool immediate = master.immediate();
38 master.set_immediate(false);
39
40 // NB: although sorter will go out of scope, its member functions sample()
41 // and exchange() will return functors whose copies get saved inside reduce
42 detail::SampleSort<Block,T,Cmp> sorter(values, samples, cmp, num_samples);
43
44 // swap-reduce to all-gather samples
45 RegularDecomposer<DiscreteBounds> decomposer(1, interval(0,assigner.nblocks()), assigner.nblocks());
46 RegularSwapPartners partners(decomposer, k);
47 reduce(master, assigner, partners, sorter.sample(), detail::SkipIntermediate(partners.rounds()));
48
49 // all_to_all to exchange the values
50 if (!samples_only)
51 all_to_all(master, assigner, sorter.exchange(), k);
52
53 master.set_immediate(immediate);
54 }

78 template<class Block, class Point>
79 void kdtree(Master& master,
80 const Assigner& assigner,
81 int dim,
82 const ContinuousBounds& domain,
83 std::vector<Point> Block::* points,
84 size_t bins,
85 bool wrap = false)
86 {
87 if (assigner.nblocks() & (assigner.nblocks() - 1))
88 throw std::runtime_error(fmt::format("KD-tree requires a number of blocks that's a power of 2, got {}", assigner.nblocks()));
89
90 typedef diy::RegularContinuousLink RCLink;
91
92 for (int i = 0; i < static_cast<int>(master.size()); ++i)
93 {
94 RCLink* link = static_cast<RCLink*>(master.link(i));
95 *link = RCLink(dim, domain, domain);
96
97 if (wrap) // set up the links to self
98 {
99 diy::BlockID self = { master.gid(i), master.communicator().rank() };
100 for (int j = 0; j < dim; ++j)
101 {
102 diy::Direction dir(dim,0), wrap_dir(dim,0);
103
104 // left
105 dir[j] = -1; wrap_dir[j] = -1;
106 link->add_neighbor(self);
107 link->add_bounds(domain);
108 link->add_direction(dir);
109 link->add_wrap(wrap_dir);
110
111 // right
112 dir[j] = 1; wrap_dir[j] = 1;
113 link->add_neighbor(self);
114 link->add_bounds(domain);
115 link->add_direction(dir);
116 link->add_wrap(wrap_dir);
117 }
118 }
119 }
120
121 detail::KDTreePartition<Block,Point> kdtree_partition(dim, points, bins);
122
123 detail::KDTreePartners partners(dim, assigner.nblocks(), wrap, domain);
124 reduce(master, assigner, partners, kdtree_partition);
125
126 // update master.expected to match the links
127 int expected = 0;
128 for (int i = 0; i < static_cast<int>(master.size()); ++i)
129 expected += master.link(i)->size_unique();
130 master.set_expected(expected);
131 }

196 // load balancing using collective method
197 template<class Callback>
198 void load_balance_collective(
199 diy::Master& master, // diy master
200 diy::DynamicAssigner& dynamic_assigner, // diy dynamic assigner
201 const Callback& f) // callback to get work for a block
202 {
203 // assert that master.destroyer() exists, will be needed for moving blocks
204 if (!master.destroyer())
205 {
206 fmt::print(stderr, "DIY error: Master must have a block destroyer function in order to use load balancing. Please define one.\n");
207 abort();
208 }
209
210 using Block = typename detail::block_traits<Callback>::type;
211 const LBCallback<Block>& f_ = f;
212
213 // compile my work info
214 diy::detail::WorkInfo my_work_info = { master.communicator().rank(), -1, 0, 0, static_cast<int>(master.size()) };
215 for (auto i = 0; i < master.size(); i++)
216 {
217 Block* block = static_cast<Block*>(master.block(i));
218 Work w = f_(block, master.gid(i));
219 my_work_info.proc_work += w;
220 if (my_work_info.top_gid == -1 || my_work_info.top_work < w)
221 {
222 my_work_info.top_gid = master.gid(i);
223 my_work_info.top_work = w;
224 }
225 }
226
227 // exchange info about load balance
228 std::vector<diy::detail::WorkInfo> all_work_info; // work info collected from all mpi processes
229 diy::detail::exchange_work_info(master, my_work_info, all_work_info);
230
231 // decide what to move where
232 std::vector<diy::detail::MoveInfo> all_move_info; // move info for all moves
233 diy::detail::decide_move_info(all_work_info, all_move_info);
234
235 // move blocks from src to dst proc
236 for (auto i = 0; i < all_move_info.size(); i++)
237 diy::detail::move_block(master, all_move_info[i]);
238
239 // fix links
240 diy::fix_links(master, dynamic_assigner);
241 }
