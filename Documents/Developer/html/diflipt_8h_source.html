64 if (pixel != NULL)

66 this->Planes = pixel->getPlanes();

137 inline void flipHorz(const T *src[],
138 T *dest[])
139 {
140 if ((src != NULL) && (dest != NULL))
141 {
142 Uint16 x;
143 Uint16 y;
144 const T *p;
145 T *q;
146 T *r;
147 for (int j = 0; j < this->Planes; ++j)
148 {
149 p = src[j];
150 r = dest[j];
151 for (Uint32 f = this->Frames; f != 0; --f)
152 {
153 for (y = this->Src_Y; y != 0; --y)
154 {
155 q = r + this->Dest_X;
156 for (x = this->Src_X; x != 0; --x)
157 *--q = *p++;
158 r += this->Dest_X;
159 }
160 }
161 }
162 }

170 inline void flipVert(const T *src[],
171 T *dest[])
172 {
173 if ((src != NULL) && (dest != NULL))
174 {
175 Uint16 x;
176 Uint16 y;
177 const T *p;
178 T *q;
179 T *r;
180 const unsigned long count = OFstatic_cast(unsigned long, this->Dest_X) * OFstatic_cast(unsigned long, this->Dest_Y);
181 for (int j = 0; j < this->Planes; ++j)
182 {
183 p = src[j];
184 r = dest[j];
185 for (Uint32 f = this->Frames; f != 0; --f)
186 {
187 r += count;
188 for (y = this->Src_Y; y != 0; --y)
189 {
190 q = r - this->Dest_X;
191 for (x = this->Src_X; x != 0; --x)
192 *q++ = *p++;
193 r -= this->Dest_X;
194 }
195 r += count;
196 }
197 }
198 }

315 {
316 s = data[j];
317 for (Uint32 f = this->Frames; f != 0; --f)
318 {
319 p = s;
320 q = s + count;
321 for (i = count / 2; i != 0; --i)
322 {
323 t = *p;
324 *p++ = *--q;
325 *q = t;
326 }
327 s += count;
328 }
