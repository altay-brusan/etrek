329inline const char* vtkTextProperty::GetFontFamilyAsString(int f)
330{
331 if (f == VTK_ARIAL)
332 {
333 return "Arial";
334 }
335 else if (f == VTK_COURIER)
336 {
337 return "Courier";
338 }
339 else if (f == VTK_TIMES)
340 {
341 return "Times";
342 }
343 else if (f == VTK_FONT_FILE)
344 {
345 return "File";
346 }
347 return "Unknown";
348}

370inline int vtkTextProperty::GetFontFamilyFromString(const char* f)
371{
372 if (strcmp(f, GetFontFamilyAsString(VTK_ARIAL)) == 0)
373 {
374 return VTK_ARIAL;
375 }
376 else if (strcmp(f, GetFontFamilyAsString(VTK_COURIER)) == 0)
377 {
378 return VTK_COURIER;
379 }
380 else if (strcmp(f, GetFontFamilyAsString(VTK_TIMES)) == 0)
381 {
382 return VTK_TIMES;
383 }
384 else if (strcmp(f, GetFontFamilyAsString(VTK_FONT_FILE)) == 0)
385 {
386 return VTK_FONT_FILE;
387 }
388 return VTK_UNKNOWN_FONT;
389}
