4/*
5
6The interface routines for reading and writing PLY polygon files.
7
8Greg Turk, February 1994
9
10---------------------------------------------------------------
11
12A PLY file contains a single polygonal _object_.
13
14An object is composed of lists of _elements_. Typical elements are
15vertices, faces, edges and materials.
16
17Each type of element for a given object has one or more _properties_
18associated with the element type. For instance, a vertex element may
19have as properties the floating-point values x,y,z and the three unsigned
20chars representing red, green and blue.
21
22*/

99typedef struct PlyElement
100{ /* description of an element */
101 char* name; /* element name */
102 int num; /* number of elements in this object */
103 int size; /* size of element (bytes) or -1 if variable */
104 int nprops; /* number of properties for this element */
105 PlyProperty** props; /* list of properties in the file */
106 char* store_prop; /* flags: property wanted by user? */
107 int other_offset; /* offset to un-asked-for props, or -1 if none*/
108 int other_size; /* size of other_props structure */
109} PlyElement;
110
111typedef struct PlyOtherProp
112{ /* describes other properties in an element */
113 char* name; /* element name */
114 int size; /* size of other_props */
115 int nprops; /* number of properties in other_props */
116 PlyProperty** props; /* list of properties in other_props */
117} PlyOtherProp;
118
119typedef struct OtherData
120{ /* for storing other_props for an other element */
121 void* other_props;
122} OtherData;
123
124typedef struct OtherElem
125{ /* data for one "other" element */
126 char* elem_name; /* names of other elements */
127 int elem_count; /* count of instances of each element */
128 OtherData** other_data; /* actual property data for the elements */
129 PlyOtherProp* other_props; /* description of the property data */
130} OtherElem;
131
132typedef struct PlyOtherElems
133{ /* "other" elements, not interpreted by user */
134 int num_elems; /* number of other elements */
135 OtherElem* other_list; /* list of data for other elements */
136} PlyOtherElems;

140 vtkSmartPointer<vtkResourceStream> is; /* input stream */

143 int file_type; /* ascii or binary */
144 float version; /* version number of file */
145 int nelems; /* number of elements of object */
146 PlyElement** elems; /* list of elements */
147 int num_comments; /* number of comments */
148 char** comments; /* list of comments */
149 int num_obj_info; /* number of items of object information */
150 char** obj_info; /* list of object info items */
151 PlyElement* which_elem; /* which element we're currently writing */
152 PlyOtherElems* other_elems; /* "other" elements from a PLY file */
153} PlyFile;
