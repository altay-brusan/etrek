87 void convert(const T1 *pixel,
88 const int bits,
89 const OFBool rgb)
90 {
91 if (this->Init(pixel))
92 {
93 const T1 offset = OFstatic_cast(T1, DicomImageClass::maxval(bits - 1));
94 unsigned long i;
95 const T1 *p = pixel;
96 T2 *r = this->Data[0];
97 T2 *g = this->Data[1];
98 T2 *b = this->Data[2];
99 T2 y1;
100 T2 y2;
101 T2 cb;
102 T2 cr;
103 // use the number of input pixels derived from the length of the 'PixelData'
104 // attribute), but not more than the size of the intermediate buffer
105 const unsigned long count = (this->InputCount < this->Count) ? this->InputCount : this->Count;
106 if (rgb) /* convert to RGB model */
107 {
108 const T2 maxvalue = OFstatic_cast(T2, DicomImageClass::maxval(bits));
109 for (i = count / 2; i != 0; --i)
110 {
111 y1 = removeSign(*(p++), offset);
112 y2 = removeSign(*(p++), offset);
113 cb = removeSign(*(p++), offset);
114 cr = removeSign(*(p++), offset);
115 convertValue(*(r++), *(g++), *(b++), y1, cb, cr, maxvalue);
116 convertValue(*(r++), *(g++), *(b++), y2, cb, cr, maxvalue);
117 }
118 } else { /* retain YCbCr model: YCbCr_422_full -> YCbCr_full */
119 for (i = count / 2; i != 0; --i)
120 {
121 y1 = removeSign(*(p++), offset);
122 y2 = removeSign(*(p++), offset);
123 cb = removeSign(*(p++), offset);
124 cr = removeSign(*(p++), offset);
125 *(r++) = y1;
126 *(g++) = cb;
127 *(b++) = cr;
128 *(r++) = y2;
129 *(g++) = cb;
130 *(b++) = cr;
131 }
132 }
133 }
134 }

145 inline void convertValue(T2 &red,
146 T2 &green,
147 T2 &blue,
148 const T2 y,
149 const T2 cb,
150 const T2 cr,
151 const T2 maxvalue)
152 {
153 double dr = OFstatic_cast(double, y) + 1.4020 * OFstatic_cast(double, cr) - 0.7010 * OFstatic_cast(double, maxvalue);
154 double dg = OFstatic_cast(double, y) - 0.3441 * OFstatic_cast(double, cb) - 0.7141 * OFstatic_cast(double, cr) + 0.5291 * OFstatic_cast(double, maxvalue);
155 double db = OFstatic_cast(double, y) + 1.7720 * OFstatic_cast(double, cb) - 0.8859 * OFstatic_cast(double, maxvalue);
156 red = (dr < 0.0) ? 0 : (dr > OFstatic_cast(double, maxvalue)) ? maxvalue : OFstatic_cast(T2, dr);
157 green = (dg < 0.0) ? 0 : (dg > OFstatic_cast(double, maxvalue)) ? maxvalue : OFstatic_cast(T2, dg);
158 blue = (db < 0.0) ? 0 : (db > OFstatic_cast(double, maxvalue)) ? maxvalue : OFstatic_cast(T2, db);
159 }
