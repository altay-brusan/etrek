141struct vtkHyperTreeData
142{
143 // Index of this tree in the hypertree grid
144 vtkIdType TreeIndex;
145
146 // Number of levels in the tree
147 unsigned int NumberOfLevels;
148
149 // Number of vertices in this tree (coarse and leaves)
150 vtkIdType NumberOfVertices;
151
152 // Number of nodes (non-leaf vertices) in the tree
153 vtkIdType NumberOfNodes;
154
155 // Offset start for the implicit global index mapping fixed by
156 // SetGlobalIndexStart after create a tree.
157 // If you don't choose implicit global index mapping then this
158 // value is -1. Then, you must to descrieb explicit global index
159 // mapping by using then SetGlobalIndexFromLocal for each cell
160 // in tree.
161 // The extra cost is equivalent to the cost of a field of values
162 // of cells.
163 vtkIdType GlobalIndexStart;
164};
