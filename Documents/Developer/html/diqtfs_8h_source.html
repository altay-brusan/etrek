64 inline void adjust(DcmQuantPixel& px, long col, long maxval)
65 {
66 long sr = px.getRed() + thisrerr[col + 1] / DcmQuantFloydSteinbergScale;
67 long sg = px.getGreen() + thisgerr[col + 1] / DcmQuantFloydSteinbergScale;
68 long sb = px.getBlue() + thisberr[col + 1] / DcmQuantFloydSteinbergScale;
69 if ( sr < 0 ) sr = 0;
70 else if ( sr > OFstatic_cast(long, maxval) ) sr = maxval;
71 if ( sg < 0 ) sg = 0;
72 else if ( sg > OFstatic_cast(long, maxval) ) sg = maxval;
73 if ( sb < 0 ) sb = 0;
74 else if ( sb > OFstatic_cast(long, maxval) ) sb = maxval;
75 px.assign(OFstatic_cast(DcmQuantComponent, sr), OFstatic_cast(DcmQuantComponent, sg), OFstatic_cast(DcmQuantComponent, sb));
76 }

83 inline void propagate(const DcmQuantPixel& px, const DcmQuantPixel& mapped, long col)
84 {
85 long err;
86
87 /* Propagate Floyd-Steinberg error terms. */
88 if ( fs_direction )
89 {
90 err = ( OFstatic_cast(long, px.getRed()) - OFstatic_cast(long, mapped.getRed()) ) * DcmQuantFloydSteinbergScale;
91 thisrerr[col + 2] += ( err * 7 ) / 16;
92 nextrerr[col ] += ( err * 3 ) / 16;
93 nextrerr[col + 1] += ( err * 5 ) / 16;
94 nextrerr[col + 2] += ( err ) / 16;
95 err = ( OFstatic_cast(long, px.getGreen()) - OFstatic_cast(long, mapped.getGreen()) ) * DcmQuantFloydSteinbergScale;
96 thisgerr[col + 2] += ( err * 7 ) / 16;
97 nextgerr[col ] += ( err * 3 ) / 16;
98 nextgerr[col + 1] += ( err * 5 ) / 16;
99 nextgerr[col + 2] += ( err ) / 16;
100 err = ( OFstatic_cast(long, px.getBlue()) - OFstatic_cast(long, mapped.getBlue()) ) * DcmQuantFloydSteinbergScale;
101 thisberr[col + 2] += ( err * 7 ) / 16;
102 nextberr[col ] += ( err * 3 ) / 16;
103 nextberr[col + 1] += ( err * 5 ) / 16;
104 nextberr[col + 2] += ( err ) / 16;
105 }
106 else
107 {
108 err = ( OFstatic_cast(long, px.getRed()) - OFstatic_cast(long, mapped.getRed()) ) * DcmQuantFloydSteinbergScale;
109 thisrerr[col ] += ( err * 7 ) / 16;
110 nextrerr[col + 2] += ( err * 3 ) / 16;
111 nextrerr[col + 1] += ( err * 5 ) / 16;
112 nextrerr[col ] += ( err ) / 16;
113 err = ( OFstatic_cast(long, px.getGreen()) - OFstatic_cast(long, mapped.getGreen()) ) * DcmQuantFloydSteinbergScale;
114 thisgerr[col ] += ( err * 7 ) / 16;
115 nextgerr[col + 2] += ( err * 3 ) / 16;
116 nextgerr[col + 1] += ( err * 5 ) / 16;
117 nextgerr[col ] += ( err ) / 16;
118 err = ( OFstatic_cast(long, px.getBlue()) - OFstatic_cast(long, mapped.getBlue()) ) * DcmQuantFloydSteinbergScale;

120 nextberr[col + 2] += ( err * 3 ) / 16;

133 inline void startRow(long& col, long& limitcol)
134 {
135 for (unsigned long c = 0; c < columns + 2; ++c)
136 nextrerr[c] = nextgerr[c] = nextberr[c] = 0;
137
138 if (fs_direction)
139 {
140 col = 0;
141 limitcol = columns;
142 }
143 else
144 {
145 col = columns - 1;
146 limitcol = -1;
147 }
148 }

154 inline void finishRow()
155 {
156 temperr = thisrerr;
157 thisrerr = nextrerr;
158 nextrerr = temperr;
159 temperr = thisgerr;
160 thisgerr = nextgerr;
161 nextgerr = temperr;
162 temperr = thisberr;
163 thisberr = nextberr;
164 nextberr = temperr;
165 fs_direction = ! fs_direction;
166 }
