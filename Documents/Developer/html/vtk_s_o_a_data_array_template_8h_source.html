80 void GetTypedTuple(vtkIdType tupleIdx, ValueType* tuple) const
81 {
82 if (this->StorageType == StorageTypeEnum::SOA)
83 {
84 for (size_t cc = 0; cc < this->Data.size(); cc++)
85 {
86 tuple[cc] = this->Data[cc]->GetBuffer()[tupleIdx];
87 }
88 }
89 else
90 {
91 ValueType* buffer = this->AoSData->GetBuffer();
92 std::copy(buffer + tupleIdx * this->GetNumberOfComponents(),
93 buffer + (tupleIdx + 1) * this->GetNumberOfComponents(), tuple);
94 }
95 }

100 void SetTypedTuple(vtkIdType tupleIdx, const ValueType* tuple)
101 {
102 if (this->StorageType == StorageTypeEnum::SOA)
103 {
104 for (size_t cc = 0; cc < this->Data.size(); ++cc)
105 {
106 this->Data[cc]->GetBuffer()[tupleIdx] = tuple[cc];
107 }
108 }
109 else
110 {
111 ValueType* buffer = this->AoSData->GetBuffer();
112 std::copy(tuple, tuple + this->GetNumberOfComponents(),
113 buffer + tupleIdx * this->GetNumberOfComponents());
114 }
115 }

125 }
126 return this->AoSData->GetBuffer()[tupleIdx * this->GetNumberOfComponents() + comp];
127 }
