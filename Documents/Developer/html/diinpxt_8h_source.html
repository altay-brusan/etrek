184 int determineMinMax()
185 {
186 if (Data != NULL)
187 {
188 DCMIMGLE_DEBUG("determining minimum and maximum pixel values for input data");
189 T2 *p = Data;
190 unsigned long i;
191 const double absrange = getAbsMaxRange();
192 const unsigned long ocnt = (absrange <= 10000000.0) ? OFstatic_cast(unsigned long, absrange) : 0 /* no LUT */;
193 Uint8 *lut = NULL;
194 if ((sizeof(T2) <= 2) && (ocnt > 0) && (Count > 3 * ocnt)) // optimization criteria
195 {
196 lut = new Uint8[ocnt];
197 if (lut != NULL)
198 {
199 DCMIMGLE_DEBUG("using optimized routine with additional LUT");
200 OFBitmanipTemplate<Uint8>::zeroMem(lut, ocnt);
201 Uint8 *q = lut - OFstatic_cast(T2, getAbsMinimum());
202 for (i = Count; i != 0; --i) // fill lookup table
203 *(q + *(p++)) = 1;
204 q = lut;
205 for (i = 0; i < ocnt; ++i) // search for minimum
206 {
207 if (*(q++) != 0)
208 {
209 MinValue[0] = OFstatic_cast(T2, OFstatic_cast(double, i) + getAbsMinimum());
210 break;
211 }
212 }
213 q = lut + ocnt;
214 for (i = ocnt; i != 0; --i) // search for maximum
215 {
216 if (*(--q) != 0)
217 {
218 MaxValue[0] = OFstatic_cast(T2, OFstatic_cast(double, i - 1) + getAbsMinimum());
219 break;
220 }
221 }
222 if (Count >= PixelCount) // use global min/max value
223 {
224 MinValue[1] = MinValue[0];
225 MaxValue[1] = MaxValue[0];
226 } else { // calculate min/max for selected range
227 OFBitmanipTemplate<Uint8>::zeroMem(lut, ocnt);
228 p = Data + PixelStart;
229 q = lut - OFstatic_cast(T2, getAbsMinimum());
230 for (i = PixelCount; i != 0; --i) // fill lookup table
231 *(q + *(p++)) = 1;
232 q = lut;
233 for (i = 0; i < ocnt; ++i) // search for minimum
234 {
235 if (*(q++) != 0)
236 {
237 MinValue[1] = OFstatic_cast(T2, OFstatic_cast(double, i) + getAbsMinimum());
238 break;
239 }
240 }
241 q = lut + ocnt;
242 for (i = ocnt; i != 0; --i) // search for maximum
243 {
244 if (*(--q) != 0)
245 {
246 MaxValue[1] = OFstatic_cast(T2, OFstatic_cast(double, i - 1) + getAbsMinimum());
247 break;
248 }
249 }
250 }
251 }
252 }
253 if (lut == NULL) // use conventional method
254 {
255 T2 value = *p;
256 MinValue[0] = value;
257 MaxValue[0] = value;
258 for (i = Count; i > 1; --i)
259 {
260 value = *(++p);
261 if (value < MinValue[0])
262 MinValue[0] = value;
263 else if (value > MaxValue[0])
264 MaxValue[0] = value;
265 }
266 if (Count <= PixelCount) // use global min/max value
267 {
268 MinValue[1] = MinValue[0];
269 MaxValue[1] = MaxValue[0];
270 } else { // calculate min/max for selected range
271 p = Data + PixelStart;
272 value = *p;
273 MinValue[1] = value;
274 MaxValue[1] = value;
275 for (i = PixelCount; i > 1; --i)
276 {
277 value = *(++p);
278 if (value < MinValue[1])
279 MinValue[1] = value;
280 else if (value > MaxValue[1])
281 MaxValue[1] = value;
282 }
283 }
284 }
285 delete[] lut;
286 return 1;
287 }
288 return 0;
289 }

363 void convert(const DiDocument *document,
364 const Uint16 bitsAllocated,
365 const Uint16 bitsStored,
366 const Uint16 highBit,
367 DcmFileCache *fileCache,
368 Uint32 &fragment)
369 {
370 T1 *pixel = NULL;
371 OFBool deletePixel = OFFalse;
372 Uint32 lengthBytes = 0;
373 DcmPixelData *pixelData = document->getPixelData();
374 const Uint16 bitsof_T1 = bitsof(T1);
375 const Uint16 bitsof_T2 = bitsof(T2);
376 const OFBool uncompressed = pixelData->canWriteXfer(EXS_LittleEndianExplicit, EXS_Unknown);
377 /* check whether to use partial read */
378 if ((document->getFlags() & CIF_UsePartialAccessToPixelData) && (PixelCount > 0) && (bitsAllocated % 8 == 0))
379 {
380 /* Bits Allocated is always a multiple of 8 (see above), same for bits of T1 */
381 const Uint32 byteFactor = bitsAllocated / 8;
382 const Uint32 bytes_T1 = bitsof_T1 / 8;
383 const Uint32 count_T1 = (byteFactor == bytes_T1) ? PixelCount : (PixelCount * byteFactor + bytes_T1 - 1) / bytes_T1;
384#ifdef DEBUG
385 DCMIMGLE_TRACE("PixelCount: " << PixelCount << ", byteFactor: " << byteFactor << ", bytes_T1: " << bytes_T1 << ", count_T1: " << count_T1);
386#endif
387 /* allocate temporary buffer, even number of bytes required for getUncompressedFrame() */
388 const Uint32 extraByte = ((sizeof(T1) == 1) && (count_T1 & 1)) ? 1 : 0;
389#ifdef HAVE_STD__NOTHROW
390 /* use a non-throwing new here (if available) because the allocated buffer can be huge */
391 pixel = new (std::nothrow) T1[count_T1 + extraByte];
392#else
393 /* make sure that the pointer is set to NULL in case of error */
394 try
395 {
396 pixel = new T1[count_T1 + extraByte];
397 }
398 catch (STD_NAMESPACE bad_alloc const &)
399 {
400 pixel = NULL;
401 }
402#endif
403 if (pixel != NULL)
404 {
405 if (uncompressed)
406 {
407 DCMIMGLE_DEBUG("using partial read access to uncompressed pixel data");
408 const Uint32 offset = PixelStart * byteFactor;
409 const Uint32 bufSize = PixelCount * byteFactor;
410 const OFCondition status = pixelData->getPartialValue(pixel, offset, bufSize, fileCache);
411 if (status.good())
412 {
413 PixelStart = 0;
414 lengthBytes = bufSize;
415 } else {
416 DCMIMGLE_ERROR("can't access partial value from byte offset " << offset << " to "
417 << (offset + bufSize - 1) << ": " << status.text());
418 }
419 } else {
420 DCMIMGLE_DEBUG("using partial read access to compressed pixel data");
421 OFCondition status = EC_IllegalCall;
422 OFString decompressedColorModel;
423 const Uint32 fsize = FrameSize * byteFactor;
424 for (Uint32 frame = 0; frame < NumberOfFrames; ++frame)
425 {
426 /* make sure that the buffer always has an even number of bytes as required for getUncompressedFrame() */
427 const Uint32 bufSize = (fsize & 1) ? fsize + 1 : fsize;
428 status = pixelData->getUncompressedFrame(document->getDataset(), FirstFrame + frame, fragment,
429 OFreinterpret_cast(Uint8 *, pixel) + lengthBytes, bufSize, decompressedColorModel, fileCache);
430 if (status.good())
431 {
432 DCMIMGLE_TRACE("successfully decompressed frame " << FirstFrame + frame);
433 lengthBytes += fsize;
434 } else {
435 DCMIMGLE_ERROR("can't decompress frame " << FirstFrame + frame << ": " << status.text());
436 break;
437 }
438 }
439 if (status.good())
440 PixelStart = 0;
441 /* check whether color model changed during decompression */
442 if (!decompressedColorModel.empty() && (decompressedColorModel != document->getPhotometricInterpretation()))
443 {
444 DCMIMGLE_WARN("Photometric Interpretation of decompressed pixel data deviates from original image: "
445 << decompressedColorModel);
446 }
447 }
448 deletePixel = OFTrue;
449 } else
450 DCMIMGLE_DEBUG("cannot allocate memory buffer for 'pixel' in DiInputPixelTemplate::convert()");
451 } else {
452 DCMIMGLE_DEBUG("reading uncompressed pixel data completely into memory");
453 /* always access complete pixel data */
454 lengthBytes = getPixelData(pixelData, pixel);
455 }
456 if ((pixel != NULL) && (lengthBytes > 0))
457 {
458 const Uint32 length_T1 = lengthBytes / sizeof(T1);
459 /* need to split 'length' in order to avoid integer overflow for large pixel data */
460 const Uint32 length_B1 = lengthBytes / bitsAllocated;
461 const Uint32 length_B2 = lengthBytes % bitsAllocated;
462// # old code: Count = ((lengthBytes * 8) + bitsAllocated - 1) / bitsAllocated;
463 Count = 8 * length_B1 + (8 * length_B2 + bitsAllocated - 1) / bitsAllocated;
464 unsigned long i;
465#ifdef HAVE_STD__NOTHROW
466 /* use a non-throwing new here (if available) because the allocated buffer can be huge */
467 Data = new (std::nothrow) T2[Count];
468#else
469 /* make sure that the pointer is set to NULL in case of error */
470 try
471 {
472 Data = new T2[Count];
473 }
474 catch (STD_NAMESPACE bad_alloc const &)
475 {
476 Data = NULL;
477 }
478#endif
479 if (Data != NULL)
480 {
481 DCMIMGLE_TRACE("Input length: " << lengthBytes << " bytes, Pixel count: " << Count
482 << " (" << PixelCount << "), In: " << bitsof_T1 << " bits, Out: " << bitsof_T2
483 << " bits (" << (this->isSigned() ? "signed" : "unsigned") << ")");
484 const T1 *p = pixel;
485 T2 *q = Data;
486 if (bitsof_T1 == bitsAllocated) // case 1: equal 8/16 bit
487 {
488 if (bitsStored == bitsAllocated)
489 {
490 DCMIMGLE_DEBUG("convert input pixel data: case 1a (single copy)");
491 for (i = Count; i != 0; --i)
492 *(q++) = OFstatic_cast(T2, *(p++));
493 }
494 else /* bitsStored < bitsAllocated */
495 {
496 T1 mask = 0;
497 for (i = 0; i < bitsStored; ++i)
498 mask |= OFstatic_cast(T1, 1 << i);
499 const T2 sign = 1 << (bitsStored - 1);
500 T2 smask = 0;
501 for (i = bitsStored; i < bitsof_T2; ++i)
502 smask |= OFstatic_cast(T2, 1 << i);
503 const Uint16 shift = highBit + 1 - bitsStored;
504 if (shift == 0)
505 {
506 DCMIMGLE_DEBUG("convert input pixel data: case 1b (mask & sign)");
507 for (i = length_T1; i != 0; --i)
508 *(q++) = expandSign(OFstatic_cast(T2, *(p++) & mask), sign, smask);
509 }
510 else /* shift > 0 */
511 {
512 DCMIMGLE_DEBUG("convert input pixel data: case 1c (shift & mask & sign)");
513 for (i = length_T1; i != 0; --i)
514 *(q++) = expandSign(OFstatic_cast(T2, (*(p++) >> shift) & mask), sign, smask);
515 }
516 }
517 }
518 else if ((bitsof_T1 > bitsAllocated) && (bitsof_T1 % bitsAllocated == 0)) // case 2: divisor of 8/16 bit
519 {
520 const Uint16 times = bitsof_T1 / bitsAllocated;
521 T1 mask = 0;
522 for (i = 0; i < bitsStored; ++i)
523 mask |= OFstatic_cast(T1, 1 << i);
524 Uint16 j;
525 T1 value;
526 if ((bitsStored == bitsAllocated) && (bitsStored == bitsof_T2))
527 {
528 if (times == 2)
529 {
530 DCMIMGLE_DEBUG("convert input pixel data: case 2a (simple mask)");
531 for (i = length_T1; i != 0; --i, ++p)
532 {
533 *(q++) = OFstatic_cast(T2, *p & mask);
534 *(q++) = OFstatic_cast(T2, *p >> bitsAllocated);
535 }
536 /* check for additional input pixel (in case of odd length when using partial access) */
537 if (length_T1 * 2 /* times */ < lengthBytes)
538 {
539 DCMIMGLE_TRACE("found trailing pixel at the end of odd-length input data ... copying value");
540 *(q++) = OFstatic_cast(T2, *p & mask);
541 }
542 }
543 else
544 {
545 DCMIMGLE_DEBUG("convert input pixel data: case 2b (mask)");
546 for (i = length_T1; i != 0; --i)
547 {
548 value = *(p++);
549 for (j = times; j != 0; --j)
550 {
551 *(q++) = OFstatic_cast(T2, value & mask);
552 value >>= bitsAllocated;
553 }
554 }
555 }
556 }
557 else
558 {
559 DCMIMGLE_DEBUG("convert input pixel data: case 2c (shift & mask & sign)");
560 const T2 sign = 1 << (bitsStored - 1);
561 T2 smask = 0;
562 for (i = bitsStored; i < bitsof_T2; ++i)
563 smask |= OFstatic_cast(T2, 1 << i);
564 const Uint16 shift = highBit + 1 - bitsStored;
565 for (i = length_T1; i != 0; --i)
566 {
567 value = *(p++) >> shift;
568 for (j = times; j != 0; --j)
569 {
570 *(q++) = expandSign(OFstatic_cast(T2, value & mask), sign, smask);
571 value >>= bitsAllocated;
572 }
573 }
574 }
575 }
576 else if ((bitsof_T1 < bitsAllocated) && (bitsAllocated % bitsof_T1 == 0) // case 3: multiplicand of 8/16
577 && (bitsStored == bitsAllocated))
578 {
579 DCMIMGLE_DEBUG("convert input pixel data: case 3 (multi copy)");
580 const Uint16 times = bitsAllocated / bitsof_T1;
581 Uint16 j;
582 Uint16 shift;
583 T2 value;
584 for (i = length_T1; i != 0; --i)
585 {
586 shift = 0;
587 value = OFstatic_cast(T2, *(p++));
588 for (j = times; j > 1; --j, --i)
589 {
590 shift += bitsof_T1;
591 value |= OFstatic_cast(T2, *(p++)) << shift;
592 }
593 *(q++) = value;
594 }
595 }
596 else // case 4: anything else
597 {
598 DCMIMGLE_DEBUG("convert input pixel data: case 4 (general)");
599 T2 value = 0;
600 Uint16 bits = 0;
601 Uint32 skip = highBit + 1 - bitsStored;
602 Uint32 times;
603 T1 mask[bitsof_T1];
604 mask[0] = 1;
605 for (i = 1; i < bitsof_T1; ++i)
606 mask[i] = (mask[i - 1] << 1) | 1;
607 T2 smask = 0;
608 for (i = bitsStored; i < bitsof_T2; ++i)
609 smask |= OFstatic_cast(T2, 1 << i);
610 const T2 sign = 1 << (bitsStored - 1);
611 const Uint32 gap = bitsAllocated - bitsStored;
612 i = 0;
613 while (i < length_T1)
614 {
615 if (skip < bitsof_T1)
616 {
617 if (skip + bitsStored - bits < bitsof_T1) // -++- --++
618 {
619 value |= (OFstatic_cast(T2, (*p >> skip) & mask[bitsStored - bits - 1]) << bits);
620 skip += bitsStored - bits + gap;
621 bits = bitsStored;
622 }
623 else // ++-- ++++
624 {
625 value |= (OFstatic_cast(T2, (*p >> skip) & mask[bitsof_T1 - skip - 1]) << bits);
626 bits += bitsof_T1 - OFstatic_cast(Uint16, skip);
627 skip = (bits == bitsStored) ? gap : 0;
628 ++i;
629 ++p;
630 }
631 if (bits == bitsStored)
632 {
633 *(q++) = expandSign(value, sign, smask);
634 value = 0;
635 bits = 0;
636 }
637 }
638 else
639 {
640 times = skip / bitsof_T1;
641 i += times;
642 p += times;
643 skip -= times * bitsof_T1;
644 }
645 }
646 }
647 } else
648 DCMIMGLE_DEBUG("cannot allocate memory buffer for 'Data' in DiInputPixelTemplate::convert()");
649 } else {
650 /* in case of error, reset pixel count variable */
651 Count = 0;
652 }
653 if (deletePixel)
654 {
655 /* delete temporary buffer */
656#if defined(HAVE_STD__NOTHROW) && defined(HAVE_NOTHROW_DELETE)
657 /* use a non-throwing delete (if available) */
658 operator delete[] (pixel, std::nothrow);
659#else
660 delete[] pixel;
661#endif
662 }
663 }
664
666 T2 *Data;
667
669 T2 MinValue[2];
671 T2 MaxValue[2];

677};
