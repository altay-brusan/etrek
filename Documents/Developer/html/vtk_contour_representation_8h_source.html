88class vtkContourRepresentationInternals
89{
90public:
91 std::vector<vtkContourRepresentationNode*> Nodes;
92 void ClearNodes()
93 {
94 for (unsigned int i = 0; i < this->Nodes.size(); i++)
95 {
96 for (unsigned int j = 0; j < this->Nodes[i]->Points.size(); j++)
97 {
98 delete this->Nodes[i]->Points[j];
99 }
100 this->Nodes[i]->Points.clear();
101 delete this->Nodes[i];
102 }
103 this->Nodes.clear();
104 }
105};
