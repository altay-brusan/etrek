22// determine move info from work info
23inline void decide_move_info(std::vector<WorkInfo>& all_work_info, // global work info
24 std::vector<MoveInfo>& all_move_info) // (output) move info for all moves
25{
26 all_move_info.clear();
27
28 // move all blocks with an approximation to the longest processing time first (LPTF) scheduling algorithm
29 // https://en.wikipedia.org/wiki/Longest-processing-time-first_scheduling
30 // we iteratively move the heaviest block to lightest proc
31 // constrained by only recording the heaviest block for each proc, not all blocks
32
33 // minimum proc_work priority queue, top is min proc_work
34 auto cmp = [&](WorkInfo& a, WorkInfo& b) { return a.proc_work > b.proc_work; };
35 std::priority_queue<WorkInfo, std::vector<WorkInfo>, decltype(cmp)> min_proc_work_q(cmp);
36 for (auto i = 0; i < all_work_info.size(); i++)
37 min_proc_work_q.push(all_work_info[i]);
38
39 // sort all_work_info by descending top_work
40 std::sort(all_work_info.begin(), all_work_info.end(),
41 [&](WorkInfo& a, WorkInfo& b) { return a.top_work > b.top_work; });
42
43 // walk the all_work_info vector in descending top_work order
44 // move the next heaviest block to the lightest proc
45 for (auto i = 0; i < all_work_info.size(); i++)
46 {
47 auto src_work_info = all_work_info[i]; // heaviest block
48 auto dst_work_info = min_proc_work_q.top(); // lightest proc
49
50 // sanity check that the move makes sense
51 if (src_work_info.proc_work - dst_work_info.proc_work > src_work_info.top_work && // improve load balance
52 src_work_info.proc_rank != dst_work_info.proc_rank && // not self
53 src_work_info.nlids > 1) // don't leave a proc with no blocks
54 {
55 MoveInfo move_info;
56 move_info.src_proc = src_work_info.proc_rank;
57 move_info.dst_proc = dst_work_info.proc_rank;
58 move_info.move_gid = src_work_info.top_gid;
59 all_move_info.push_back(move_info);
60
61 // update the min_proc_work priority queue
62 auto work_info = min_proc_work_q.top(); // lightest proc
63 work_info.proc_work += src_work_info.top_work;
64 if (work_info.top_work < src_work_info.top_work)
65 {
66 work_info.top_work = src_work_info.top_work;
67 work_info.top_gid = src_work_info.top_gid;
68 }
69 min_proc_work_q.pop();
70 min_proc_work_q.push(work_info);
71 }
72 }
73}

75// move one block from src to dst proc
76inline void move_block(diy::Master& master,
77 const MoveInfo& move_info)
78{
79 // sanity check that source and destination are different
80 if (move_info.src_proc == move_info.dst_proc)
81 {
82 fmt::print(stderr, "Error: move_block(): source and destination are same. This should not happen.\n");
83 abort();
84 }
85
86 if (master.communicator().rank() == move_info.src_proc)
87 {
88 diy::MemoryBuffer bb;
89
90 // move the block from src to dst proc
91 void* send_b = master.block(master.lid(move_info.move_gid));
92 master.saver()(send_b, bb);
93 master.communicator().send(move_info.dst_proc, 0, bb.buffer);
94
95 // move the link for the moving block
96 diy::Link* send_link = master.link(master.lid(move_info.move_gid));
97 diy::LinkFactory::save(bb, send_link);
98 master.communicator().send(move_info.dst_proc, 0, bb.buffer);
99
100 // remove the block from the master
101 int move_lid = master.lid(move_info.move_gid);
102 master.destroyer()(master.release(move_lid));
103 }
104 else if (master.communicator().rank() == move_info.dst_proc)
105 {
106 diy::MemoryBuffer bb;
107
108 // move the block from src to dst proc
109 void* recv_b = master.creator()();
110 master.communicator().recv(move_info.src_proc, 0, bb.buffer);
111 master.loader()(recv_b, bb);
112
113 // move the link for the moving block
114 diy::Link* recv_link;
115 master.communicator().recv(move_info.src_proc, 0, bb.buffer);
116 recv_link = diy::LinkFactory::load(bb);
117
118 // add the block to the master
119 master.add(move_info.move_gid, recv_b, recv_link);
120 }
