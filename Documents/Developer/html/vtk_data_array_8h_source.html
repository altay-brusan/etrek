714{
715 if (source)
716 {
717 switch (source->GetArrayType())
718 {
719 case AoSDataArrayTemplate:
720 case SoADataArrayTemplate:
721 case ImplicitArray:
722 case TypedDataArray:
723 case DataArray:
724 case MappedDataArray:

727 break;
