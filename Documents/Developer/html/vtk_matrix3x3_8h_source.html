190inline void vtkMatrix3x3::SetElement(int i, int j, double value)
191{
192 if (this->Element[i][j] != value)

194 this->Element[i][j] = value;
195 this->Modified();
196 }

199inline bool vtkMatrix3x3::IsIdentity()
200{
201 double* M = *this->Element;
202 if (M[0] == 1.0 && M[4] == 1.0 && M[8] == 1.0 && M[1] == 0.0 && M[2] == 0.0 && M[3] == 0.0 &&
203 M[5] == 0.0 && M[6] == 0.0 && M[7] == 0.0)
204 {
205 return true;
206 }
207 else
208 {
209 return false;
210 }
211}
