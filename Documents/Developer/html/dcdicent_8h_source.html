288 /* containment */
289
297 int contains(const DcmTagKey& key, const char *privCreator) const /* this contains key */
298 {
299 if ((getGroupRangeRestriction() == DcmDictRange_Even) &&
300 DCM_IS_ODD(key.getGroup()))
301 return OFFalse;
302 else if ((getGroupRangeRestriction() == DcmDictRange_Odd) &&
303 DCM_IS_EVEN(key.getGroup()))
304 return OFFalse;
305 else if ((getElementRangeRestriction() == DcmDictRange_Even) &&
306 DCM_IS_ODD(key.getElement()))
307 return OFFalse;
308 else if ((getElementRangeRestriction() == DcmDictRange_Odd) &&
309 DCM_IS_EVEN(key.getElement()))
310 return OFFalse;
311 else if (! privateCreatorMatch(privCreator))
312 return OFFalse;
313 else
314 {
315 const OFBool groupMatches=DCM_INRANGE(key.getGroup(), getGroup(), getUpperGroup());
316 OFBool found=groupMatches && DCM_INRANGE(key.getElement(), getElement(), getUpperElement());
317 if (!found && groupMatches && privCreator)
318 found=DCM_INRANGE(key.getElement() & 0xFF, getElement(), getUpperElement());
319 return found;
320 }
321 }
